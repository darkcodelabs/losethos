// imports all syms from the Adam Task's hash table.
#include "::/LT/OSMain/StaticAdam.HPZ"

#exe {OptOn(OPTf_STD_C);};

void main()
{
  char ch;
  unsigned short s; //example of unsigned
  int i;
  long q;
  double d;
  bool b;

  LtfClear();
  PrintF("char\t:%d\n",sizeof(ch));
  PrintF("bool\t:%d\n",sizeof(b));
  PrintF("short\t:%d\n",sizeof(s));
  PrintF("int\t:%d\n",sizeof(i));
  PrintF("long\t:%d\n",sizeof(q));
  PrintF("double\t:%d\n",sizeof(d));
}

main(); //run main() when Load()ed.
