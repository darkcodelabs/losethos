#define MAP_WIDTH                1024
#define MAP_HEIGHT                1024

#define MAP_SCALE                150
#define DISPLAY_SCALE                100
#define CONTROLS_SCALE                0.05

//$FG,2$Scaling is a mess.$FG$
#define COORDINATE_SCALE        256
#define COORDINATE_BITS                8

#define ROCK_ELEVATION                45
#define SNOW_ELEVATION                55

#define MAX_PANEL_SIZE                16

class Panel
{

};

void FSTransform(GrBitMap *base,I8 *x,I8 *y,I8 *z)
{
//$FG,2$I accidentally made the Z coordinate sign backward$FG$
//$FG,2$from the fill-poly depth buffer scheme, so it is negated at the
end.$FG$
//$FG,2$Negative Z coordinates are the ones we want plotted.$FG$
 GrRotate(base->r,x,y,z);

//$FG,2$We don't want divide by zero or make too monstrous foreground
objects.$FG$
//$FG,2$The foreground object boundry is kinda crappy looking when$FG$
//$FG,2$you point toward the ground.$FG$
 if (*z<-50) {

//$FG,2$The divide causes "foreshortening" which makes the distance go to a
vanishing point.$FG$
     *x = *x * -DISPLAY_SCALE/ *z;
     *y = *y * -DISPLAY_SCALE/ *z;
 }
 *x+=base->x;
 *y=base->y-*y;
 *z=-*z;
}


$BG,7$
void CalcNormals()
{
/*$FG,2$The idea is that there are two triangles
forming every grid square and we find the normal
vector with a curl.

i  j  k
0  1  dz2
1  0  dz1

Normal:
dz1*i + dz2*j - k

 i   j   k
-1   0  -dz3
 0  -1  -dz4

Normal:
-dz3*i - dz4*j + k


Basically, we want to combine grid
locations with the same normals.  These
are not unit vectors for one thing.

What I came-up with is dz1/dz2 and dz3/dz4.

It works pretty well.  It combines
more than it should, but that's good
because it would  result in too many
panels.

$FG$*/
 I8 i,j,k,dz1,dz2;
 for (j=0;j<MAP_HEIGHT-1;j++) {
     k=0;
     for (i=0;i<MAP_WIDTH-1;i++) {
         dz1=elevations[j][i+1]-elevations[j][i];
         dz2=elevations[j+1][i]-elevations[j][i];
         if (dz2)
             normals[j][k++]=dz1<<16/dz2;
         else {
             if (!dz1)
                 normals[j][k++]=0;
             else if (dz1>0)
                 normals[j][k++]=MAX_I4;
             else
                 normals[j][k++]=MIN_I4;
         }

         dz1=elevations[j+1][i+1]-elevations[j][i+1];
         dz2=elevations[j+1][i+1]-elevations[j+1][i];
         if (dz2)
             normals[j][k++]=dz1<<16/dz2;
         else {
             if (!dz1)
                 normals[j][k++]=0;
             else if (dz1>0)
                 normals[j][k++]=MAX_I4;
             else
                 normals[j][k++]=MIN_I4;
         }
     }
     normals[j][k++]=0;
     normals[j][k++]=0;
 }
 k=0;
 for (i=0;i<MAP_WIDTH-1;i++) {
     normals[j][k++]=0;
     normals[j][k++]=0;
 }
}

BoolI8 TestSameSlope(I8 x,I8 y,I8 w,I8 h)
{
 I8 k1,k2,xx,yy,s;
 if (!(0<=x && x+w<MAP_WIDTH && 0<=y && y+h<MAP_HEIGHT))
     return FALSE;
     //$FG,2$Works well averaging the two slopes of two triangles.$FG$
     //$FG,2$If it is too picky, we get too many panels.$FG$
 s=normals[y][x*2]+normals[y][x*2+1];
 for (k2=0;k2<h;k2++)
     for (k1=0;k1<w;k1++) {
         xx=(x+k1)*2;
         yy=y+k2;
         if (normals[yy][xx]+normals[yy][xx+1]!=s)
           return FALSE;
     }
 return TRUE;
}

void MPDoPanels(TssStruct *tss)
{
 I8 i,j,k,l,k1,k2,w,h,threshold,lo,hi;
 BoolI1 cont;
 Panel *tempp,*start_ptr=NULL,*end_ptr=NULL;
 P3I4 *p;
 BoolI1 old_preempt=Preempt(ON);
 lo=Gs->num*(MAP_HEIGHT-1)/mp.mp_cnt;
 hi=(Gs->num+1)*(MAP_HEIGHT-1)/mp.mp_cnt;
 for (threshold=8;threshold>=1;threshold--)
     for (j=lo;j<hi;j++) {
         k=0;
         for (i=0;i<MAP_WIDTH-1;i++) {
             if (!panels[j][k]) {
                 w=1;
                 h=1;
                 do {
                     cont=FALSE;
                     if (w<MAX_PANEL_SIZE && TestSameSlope(i,j,w+1,h)) {
                         w++;
                         cont=TRUE;
                     }
                     if (h<MAX_PANEL_SIZE && TestSameSlope(i,j,w,h+1)) {
                         h++;
                         cont=TRUE;
                     }
                 } while (cont);
                 if (w>=threshold || h>=threshold) {
                     tempp=MAlloc(sizeof(Panel),tss);
                     lock {tempp->num=num_panels++;}
                     l=elevations[j][i];
                     if (l<=WATER_ELEVATION*MAP_SCALE)
                         tempp->color=BLUE;
                     else {
                         if (RandI2&1) {
                             if (l<ROCK_ELEVATION*MAP_SCALE)
                                 tempp->color=LTGREEN;
                             else if (l<SNOW_ELEVATION*MAP_SCALE) {
                                 if (!(RandU2&3))
                                     tempp->color=LTGRAY;
                                 else
                                     tempp->color=LTGREEN;
                             } else {
                                 if (!(RandU2&3))
                                     tempp->color=WHITE;
                                 else
                                     tempp->color=LTGRAY;
                             }
                         } else {
                             if (l<ROCK_ELEVATION*MAP_SCALE)
                                 tempp->color=GREEN+LTGREEN<<32+ROP_DITHER;
                             else if (l<SNOW_ELEVATION*MAP_SCALE) {
                                 if (!(RandU2&3))
                                     tempp-
>color=DKGRAY+LTGRAY<<32+ROP_DITHER;
                                 else
                                     tempp-
>color=GREEN+LTGREEN<<32+ROP_DITHER;
                             } else {
                                 if (!(RandU2&3))
                                     tempp->color=LTGRAY+WHITE<<32+ROP_DITHER;
                                 else
                                     tempp-
>color=DKGRAY+LTGRAY<<32+ROP_DITHER;
                             }
                         }
                     }
                     tempp->n=4;
                     p=tempp->points=MAlloc(sizeof(P3I4)*tempp->n,tss);
                     p[0].x=MAP_SCALE*i;
                     p[0].y=MAP_SCALE*j;
                     p[0].z=elevations[j][i];
                     p[1].x=MAP_SCALE*(i+w);
                     p[1].y=MAP_SCALE*j;
                     p[1].z=elevations[j][i+w];
                     p[2].x=MAP_SCALE*(i+w);
                     p[2].y=MAP_SCALE*(j+h);
                     p[2].z=elevations[j+h][i+w];
                     p[3].x=MAP_SCALE*i;
                     p[3].y=MAP_SCALE*(j+h);
                     p[3].z=elevations[j+h][i];
                     tempp->next=start_ptr;
                     start_ptr=tempp;
                     if (!end_ptr)
                         end_ptr=tempp;
                     for (k2=0;k2<h;k2++)
                         for (k1=0;k1<w;k1++) {
                             panels[j+k2][(i+k1)*2]=tempp;
                             panels[j+k2][(i+k1)*2+1]=tempp;
                         }
                 }
             }
             k+=2;
         }
     }
 if (end_ptr) {
     CriticalSectionStart;
     end_ptr->next=panel_root;
     panel_root=start_ptr;
     CriticalSectionEnd;
 }
 Preempt(old_preempt);
 LBts(&mp.done_flags,Gs->num);
}


void InitMap()
{
//$FG,2$We make a topographic data structure "evevations[][]"$FG$
//$FG,2$and convert it to panels.  The conversion to panels leaves$FG$
//$FG,2$gaps the way I did it, but it looks good, except for sky leaking
through$FG$
//$FG,2$There is no wrong way to pick colors if it looks good, unless$FG$
//$FG,2$you wanted to do lighting.$FG$

//$FG,2$"panels[][]" holds the flat and sloped panels for each spot, while$FG$
//$FG,2$"hstep_panels[][]" holds side panels in cases of step-like
formations.$FG$
//$FG,2$They require a separate array because a single "elevation[][]" spot
can$FG$
//$FG,2$have a panel on top and a panel on the side like stairs.$FG$

 I8 i,j,l,k1,k2,x,y,xx,yy;
 MemSet(elevations,0,sizeof(elevations));
 MemSet(panels,0,sizeof(panels));
 for (i=0;i<MAP_WIDTH*MAP_HEIGHT/128;i++) {
     x=RandU4%MAP_WIDTH;
     y=RandU4%MAP_HEIGHT;
     j=1<<(RandU4%6);
     l=0;
     while (j--) {
         if (!l && RandU2<MAX_U2/4)
             l=RandU2%(j+1);
         if (l) {
             for (k1=-j;k1<=j;k1++)
                 for (k2=-j;k2<=j;k2++) {
                     xx=x+k2; yy=y+k1;
                     if (0<=xx<MAP_WIDTH &&
                             0<=yy<MAP_HEIGHT)
                         elevations[yy][xx]+=MAP_SCALE/2;
                 }
                 l--;
         }
     }
 }

 xx=x/(MAP_SCALE*COORDINATE_SCALE);
 yy=y/(MAP_SCALE*COORDINATE_SCALE);
 z+=elevations[yy][xx]*COORDINATE_SCALE;

 for (j=0;j<MAP_HEIGHT;j++)
     for (i=0;i<MAP_WIDTH;i++)
         if (elevations[j][i]<WATER_ELEVATION*MAP_SCALE)
             elevations[j][i]=WATER_ELEVATION*MAP_SCALE;


 CalcNormals;

 panel_root=NULL;
 num_panels=0;

 mp.done_flags=0;
 for (i=1;i<mp.mp_cnt;i++)
     MPSpawn(&MPDoPanels,Fs,"Do Panels",1<<i);
 MPDoPanels(Fs);
 while (mp.done_flags!=1<<mp.mp_cnt-1)
     SwapInNextTask;
 panels_processed_bitmap=MAlloc((num_panels+7)>>3);
}

void PanelsDel()
{
 Panel *tempp=panel_root,*tempp1;
 while (tempp) {
     tempp1=tempp->next;
     Free(tempp->points);
     Free(tempp);
     tempp=tempp1;
 }
 Free(panels_processed_bitmap);
}

/*$PI,"",1$





















$FG,2$Cores render strips.  The strips are wider toward the horizon by $FG,
4$span_factor$FG,2$
and overlap +/- 15%. The cores check the panel map array
and render the panel for each square and mark-it done.

The depth buffer is not locked in the graphic routines
so we get some glitches.
$FG$
*/

void MPDrawIt(TssStruct *tss,GrBitMap *base)
{
//$FG,2$This is identical to the previous routine but is run by core#1$FG$
 I8 i,j,ww,hh,*s2w,*r1,*r2,*r3,*r4, reg x1,reg y1,z1, xx,yy,
         xh,yh,zh, yh2,xh2, reg x1w, reg y1w, x1h,y1h,
         x3,y3,z3,
         cx=tss->win_pixel_width>>1,
         cy=tss->win_pixel_height>>1;
 F8 h_grid_size;
 Panel reg *tempp;

 xx=x/(MAP_SCALE*COORDINATE_SCALE);
 yy=y/(MAP_SCALE*COORDINATE_SCALE);

 //$FG,2$World to screen coordinates$FG$
 r1=GrRotZ(theta1);
 r2=GrRotX(phi);
 r3=GrRotZ(theta2);
 r4=GrMulMat(r1,r2);
 Free(base->r);
 GrSetRotMat(base,GrMulMat(r4,r3));
 Free(r1);
 Free(r2);
 Free(r3);
 Free(r4);

 //$FG,2$Screen to world coordinates$FG$

//$FG,2$This gives us the vectors for stepping through the grid in$FG$
//$FG,2$the direction the plane is facing. we step horizontally and
vertically$FG$
//$FG,2$and use the reciprocal slope principle        y=mx+b and y=(-1/m)x+b
are perpendicular.$FG$

 r1=GrRotZ(-theta2);
 r2=GrRotX(-phi);
 r3=GrRotZ(-theta1);
 r4=GrMulMat(r1,r2);
 s2w=GrMulMat(r4,r3);
 xh=0;
 yh=0;
 zh=-256;
 GrRotate(s2w,&xh,&yh,&zh);

//$FG,2$The layer for core#1 is not cleared automatically$FG$
//$FG,2$it is persistent.  I have carefully syncronized to the update$FG$
//$FG,2$cycle initiated by core#0 to prevent flicker.$FG$

 Free(r1);
 Free(r2);
 Free(r3);
 Free(r4);

 base->flags|=BMF_TRANSFORMATION;
 base->transform=&FSTransform;
 base->x=cx;
 base->y=cy;

//$FG,2$base->x and the translation part of base->r are identical in
effect,$FG$
//$FG,2$but we use both to show-off.        We could add offsets together and
use one or the other.$FG$

 x1=-x>>COORDINATE_BITS;
 y1=-y>>COORDINATE_BITS;
 z1=-z>>COORDINATE_BITS;
 GrRotate(base->r,&x1,&y1,&z1);
 GrSetTranslation(base->r,x1,y1,z1);

 //$FG,2$This is a refinement.$FG$
 if (Abs(phi*180/pi)>90) {
     x3=0;
     y3=-cy;
     z3=0;
     GrRotate(s2w,&x3,&y3,&z3);
     xx+=x3;
     yy+=y3;
 }

 h_grid_size=4.0*v_grid_size*(Gs->num+span_factor)/span_factor;
 if (Gs->num&1) { //$FG,2$alternate left-right,right-left$FG$
     yh2=-yh;
     xh2=-xh;
 } else {
     yh2=yh;
     xh2=xh;
 }

 ww=h_grid_size;
 hh=1.30*v_grid_size; //$FG,2$Cores overlap 15% on each side of strip.$FG$
 //$FG,2$Calc starting point.$FG$
 x1h=xx<<8+0.5*yh2*h_grid_size+xh*v_grid_size*(Gs->num+1.15);
 y1h=yy<<8-0.5*xh2*h_grid_size+yh*v_grid_size*(Gs->num+1.15);

 xh=-xh; //$FG,2$Back to front to help with depth.$FG$
 yh=-yh;

 //$FG,2$Take half steps to cover whole grid.$FG$
 xh>>=1;  yh>>=1;
 xh2>>=1; yh2>>=1;
 ww<<=1;
 hh<<=1;

 for (j=0;j<hh;j++) {
     x1w=x1h;
     y1w=y1h;
     for (i=0;i<ww;i++) {
         x1=x1w>>8; y1=y1w>>8;
         if (0<=x1<MAP_WIDTH && 0<=y1<MAP_HEIGHT) {
                 if ((tempp=panels[y1][x1*2]) && !LBts
(panels_processed_bitmap,tempp->num)) {
                     base->color=tempp->color;
                     GrFillPolygon3(base,tempp->n,tempp->points);
                 }
         }
         x1w-=yh2;
         y1w+=xh2;
     }
     x1h+=xh;
     y1h+=yh;
 }
 Free(s2w);
 mp.update_done_t[Gs->num]=t;
}

void APFlightSimTask(TssStruct *master_tss)
{
 GrBitMap *base=GrAlias(main_base,master_tss);
 Preempt(ON);
 while (!mp.app_done) {
     LBts(&Fs->task_flags,TSSf_IDLE);
     while (!LBtr(&mp.do_update,Gs->num) && !mp.app_done)
         SwapInNextTask;
     LBtr(&Fs->task_flags,TSSf_IDLE);
     if (!mp.app_done)
         MPDrawIt(master_tss,base);
 }

//$FG,2$We made an alias of this we don't want freed.$FG$
 base->depth_buffer=NULL;

 GrDel(base);
 LBts(&mp.done_flags,Gs->num);
}

void BSPFlightSim(GrBitMap *base)
{
 F8 t0,last_t;
 I8  i,xx,yy,height,
         cx=Fs->win_pixel_width>>1,
         cy=Fs->win_pixel_height>>1;

 t0=t;
 MemSet(panels_processed_bitmap,0,(num_panels+7)>>3);
 MemSet(mp.update_done_t,0,sizeof(F8)*MP_MAX_PROCESSORS);
 base->color=LTCYAN;
 GrBox(base,0,0,GR_WIDTH,GR_HEIGHT);
 GrDepthBufferReset(base);

 //$FG,2$Signal all cores to start update$FG$
 mp.do_update=1<<mp.mp_cnt-1-1;

 xx=x/(MAP_SCALE*COORDINATE_SCALE);
 yy=y/(MAP_SCALE*COORDINATE_SCALE);
 height=z/COORDINATE_SCALE-elevations[yy][xx];
 if (height<0) {
     if (!crashed) {
         music_mute=1;
         Sound(1000); //Signal plane crashed
         crashed=TRUE;
     }
 } else {
     if (crashed) {
         Sound(0);
         music_mute=0;
         crashed=FALSE;
     }
 }

 //$FG,2$Wait for all cores to start$FG$
 while (mp.do_update)
     SwapInNextTask;

 MPDrawIt(Fs,base);

 //$FG,2$Wait for all cores to complete.$FG$
 LBts(&Fs->task_flags,TSSf_IDLE);
 for (i=1;i<mp.mp_cnt;i++)
     while (!mp.update_done_t[i])
         SwapInNextTask;
 LBtr(&Fs->task_flags,TSSf_IDLE);

 base->pen_width=2;
 base->color=BLACK;
 base->flags&=~BMF_TRANSFORMATION;
 GrLine3(base,cx+5,cy,0,cx-5,cy,0);
 GrLine3(base,cx,cy+5,0,cx,cy-5,0);
 GrPrintF(base,0,0,"Theta1:%5.1f Phi:%5.1f Theta2:%5.1f Box:%5.1f",
         theta1*180/pi,phi*180/pi,theta2*180/pi,v_grid_size);
 GrPrintF(base,0,FONT_HEIGHT,"x:%5.1f y:%5.1f z:%5.1f height:%3d",
         ToF8(x)/COORDINATE_SCALE,ToF8(y)/COORDINATE_SCALE,ToF8(z)
/COORDINATE_SCALE,
         height);

 last_t=mp.update_done_t[0];
 for (i=1;i<mp.mp_cnt;i++)
     last_t=Max(last_t,mp.update_done_t[i]);

 //$FG,2$Adjust size of updated grid.$FG$
 v_grid_size  *=0.90+0.10*(0.65/win_max_refresh)/(last_t-t0);
 //$FG,2$Regulate how much grid expands in the distance.$FG$
 if (mp.mp_cnt>1)
     span_factor*=0.90+0.10*(mp.update_done_t[mp.mp_cnt-1]-t0)/(mp.
update_done_t[0]-t0);

 WinSync;
}

void AnimateTask()
{
//$FG,2$This just steadily moves the airplane forward.$FG$
 I8 *r1,*r2,*r3,*r4,*s2w,x1,y1,z1;
 while (TRUE) {
 //$FG,2$Screen to world coordinates$FG$
     r1=GrRotZ(-theta2);
     r2=GrRotX(-phi);
     r3=GrRotZ(-theta1);
     r4=GrMulMat(r1,r2);
     s2w=GrMulMat(r4,r3);

     x1=0;y1=0;z1=-speed*COORDINATE_SCALE;
     GrRotate(s2w,&x1,&y1,&z1);
     x+=x1;
     y+=y1;
     z+=z1;
     Free(r1);
     Free(r2);
     Free(r3);
     Free(r4);
     Free(s2w);
     Sleep(1);
 }
}

void MPEnd()
{
 mp.done_flags=0;
 mp.app_done=TRUE;
 //$FG,2$Wait for all cores to exit$FG$
 while (mp.done_flags!=1<<mp.mp_cnt-1-1)
     SwapInNextTask;
}

void EndTaskCB()
{
 MPEnd;
 Exit;
}

void FlightSim(I8 _mp_cnt=MP_MAX_PROCESSORS)
{
 I8 ch,sc;
 F8 pp,tt1,tt2;
 TssStruct *animate_tss,*song_tss;
 main_base=GrNew(BMT_COLOR4,GR_WIDTH,GR_HEIGHT);

 PushSettings; //See $LK,"PushSettings","MN:PushSettings"$
 //$FG,2$We really can't afford to run at 60Hz.        This is how often
VGA$FG$
 //$FG,2$memory is updated, not related to actual hardware refresh rates.$FG$

 WordStat(OFF);

 PushMenu(
 "File {"
 "  Abort(,CH_CTRLQ);"
 "  Exit(,CH_ESC);"
 "}"
 "Play {"
 "  Down(,,SC_CURSOR_UP);"
 "  Up(,,SC_CURSOR_DOWN);"
 "  Left(,,SC_CURSOR_LEFT);"
 "  Right(,,SC_CURSOR_RIGHT);"
         "}"
         );
 WinMax;
 coutln "Initializing...";
 song_tss=Spawn(&SongTask,0,"FSim2 Song",Fs);

 MemSet(&mp,0,sizeof(MPCtrl));
 if (_mp_cnt<=mp_cnt)
     mp.mp_cnt=_mp_cnt;
 else
     mp.mp_cnt=mp_cnt;
 InitMap;
 animate_tss=Spawn(&AnimateTask,0,"FSim2 Animate",Fs);

 GrAllocDepthBuffer(main_base);

 win_max_refresh=12.0;

 v_grid_size=64.0;
 span_factor=1.0;
 phi        =-90.0*pi/180.0;
 theta1=  0.0*pi/180.0;
 theta2=  0.0*pi/180.0;
 speed =  2.5;

 x=MAP_WIDTH>>1 *COORDINATE_SCALE*MAP_SCALE;
 y=MAP_HEIGHT>>1*COORDINATE_SCALE*MAP_SCALE;
 z=64                  *COORDINATE_SCALE*MAP_SCALE;

 for (i=1;i<mp.mp_cnt;i++)
     MPSpawn(&APFlightSimTask,Fs,"AP FlightSim",1<<i);
 Fs->end_task_cb=&EndTaskCB;

 Fs->update_win=&UpdateWin;
 try //$FG,2$in case CTRL-ALT-C is pressed.$FG$
     do {
         if (ScanKey(&ch,&sc)) {
             pp=phi; tt1=theta1; tt2=theta2;
             switch (ch) {
                 case 0:
                     switch (sc.u1[0]) {
                         case SC_CURSOR_DOWN:
                             phi        +=-CONTROLS_SCALE*Cos(tt1);
                             theta2+=-CONTROLS_SCALE*Sin(tt1)*Sin(phi);
                             break;
                         case SC_CURSOR_UP:
                             phi        -=-CONTROLS_SCALE*Cos(tt1);
                             theta2-=-CONTROLS_SCALE*Sin(tt1)*Sin(phi);
                             break;
                         case SC_CURSOR_RIGHT:
                             theta1+=CONTROLS_SCALE;
                             break;
                         case SC_CURSOR_LEFT:
                             theta1-=CONTROLS_SCALE;
                             break;
                     }
                     break;
                 }
             theta1=Unwrap(theta1);
             phi        =Unwrap(phi);
             theta2=Unwrap(theta2);
         } else
             BSPFlightSim(main_base);
     } while (ch!=CH_CTRLQ && ch!=CH_ESC);
 catch
     Fs->catch_except=TRUE;
 Kill(animate_tss);
 Kill(song_tss);
 PopSettings;
 WinSync;

 MPEnd;
 PopMenu;
 PanelsDel;
 GrDel(main_base);
}

FlightSim;
