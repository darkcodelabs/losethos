#define MPCf_DONE                0
#define MPCf_DISPATCHED                1
#define MPCf_FREE_ON_COMPLETE        2

#define MPCT_CALL                1
#define MPCT_SPAWN_TASK                2

class MPCmdStruct
{
  MPCmdStruct *next,*last;
  U8 cmd_code;
  U8 flags;
  U1 *add;
  U1 *data;
  I1 *desc;
  I8 target_cpu_mask;
  TssStruct *tss; //spawned task
  I8 handler_cpu; //CPU accepting job
  I8 result;
};

class MPCmdCtrl
{
  MPCmdStruct *next_waiting,*last_waiting;
  MPCmdStruct *next_done,*last_done;
};

asm {
////**************************PROCEDURE*************************
               ALIGN        16,OC_NOP
               USE16
MP_INIT_START::
               JMP        MP2_START2
               ALIGN        4,OC_NOP
MP2_SYS_TEMP_PTR:        DU4        0,0;

MP2_START2:
               CLI

               WBINVD

               MOV_EAX_CR0
               OR        EAX,0x60000000
               MOV_CR0_EAX

               MOV        AX,MP_VECTOR_ADDRESS/16
               MOV        DS,AX

               LGDT        U4 [MP_SYS_TEMP_PTR]

               MOV        EAX,SYS_START_CR0|0x60000000
               MOV_CR0_EAX

               DU1        0x66,0xEA;                 //JMP SYS_CS_SEL:MP_INIT_OS
               DU4        MP_INIT_OS;
               DU2        SYS_CS_SEL;

MP_INIT_END::
               USE32
MP_INIT_OS:
               MOV        AX,ZERO_DS_SEL
               MOV        DS,AX
               MOV        ES,AX
               MOV        FS,AX
               MOV        GS,AX
               MOV        SS,AX

               FINIT

@@1:        LOCK
               BTS        U4 [SYS_MP_CNT_LOCK],0
               JC        @@1

               MOV        ESI,U4 [SYS_MP_CNT]
               LOCK
               INC        U4 [SYS_MP_CNT]
               LOCK
               BTR        U4 [SYS_MP_CNT_LOCK],0


               CMP        ESI,MP_MAX_PROCESSORS
               JB        @@3
               HLT

@@3:        MOV_EAX_CR0
               AND        EAX,~0x60000000 //enable cache
               MOV_CR0_EAX

               IMUL2        ESI,CPU_STRUCT_SIZE
               ADD        ESI,U4 [SYS_CPU_STRUCTS]

               LEA        EAX,U4 CPU_START_STACK_TOP[ESI]
               MOV        ESP,EAX
               PUSH        U4 SYS_START_RFLAGS
               POPFD
               PUSH        U4 0
               CALL        INIT_EM64T
USE64
               PUSH        RSI
               CALL        CP_SET_GS_BASE
               POP        RSI
@@2:        MOV        RBX,U8 CPU_SETH_TSS[RSI]
               OR        RBX,RBX
               JZ        @@2
               MOV        U8 TSS_GS[RBX],RSI
               MOV        RAX,RBX
               CALL        SET_FS_BASE

               JMP        I4 RESTORE_CONTEXT
};

void MPInt(U8 num,U8 cpu_num=1)
{
   U4 *dl=MP_ICR_LOW,*dh=MP_ICR_HIGH;
   PushFD;
   Cli;
   while (LBts(&sys_semas[SYS_SEMA_APIC_LOCK],0))
       SwapInNextTask;
   while (*dl&0x1000)
       Pause;
   *dh=mp_apic_ids[cpu_num]<<24;
   *dl=0x4800+num;
   LBtr(&sys_semas[SYS_SEMA_APIC_LOCK],0);
   PopFD;
}

void MPEOI()
{
   U4 *dl=MP_EOI;
   while (LBts(&sys_semas[SYS_SEMA_APIC_LOCK],0))
       SwapInNextTask;
   *dl=0;
   LBtr(&sys_semas[SYS_SEMA_APIC_LOCK],0);
}

void MPNMInt()
{
   U4 *dl=MP_ICR_LOW;
   *dl=0xC4400;
}

void MPHalt()
{
   mp_cnt=1;
   MPNMInt; //Hlt All other processors
}

void MPWbInvdAll()
{
   MPIntAll(I_WBINVD);
   WbInvd;
}

void MPInitAPIC()
{ //Called by adam during start-up
   //and other cores during initialization
   //after $LK,"MPStart","MN:MPStart"$().
   U4 *d;

   d=MP_SVR;
   *d|=MPF_APIC_ENABLED;
   d=MP_APIC_ID;
   mp_apic_ids[Gs->num]=*d>>24;
   d=MP_LDR;
   *d=mp_apic_ids[Gs->num]<<24;
   d=MP_DFR;
   *d=0xF0000000;
/*
   MemSet(MP_IRR,0,0x20);
   MemSet(MP_ISR,0,0x20);
   MemSet(MP_TMR,0,0x20);
*/

   SetRAX(Gs->tr);
   asm {
               LTR        U4 RAX
   }
   if (Gs->num) {
       InitIDT;
       CallExtStr("GrInitCPU",Gs);
   }
}

void MPWaitForTask()
{
   U8 timeout=0;
   MPCmdStruct *tempm;
   Preempt(OFF);
   Sti;
   LBts(&Fs->task_flags,TSSf_IDLE);
   while (TRUE) {
       while (TRUE) {
           if (GetTimeStamp>timeout) {
               FinishOffDyingTsses;
               timeout=GetTimeStamp+time_stamp_freq>>9;
           } else if (mp_ctrl->next_waiting==mp_ctrl)
               SwapInNextTask;
           else
               break;
       }
       while (LBts(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0))
           SwapInNextTask;
       tempm=mp_ctrl->next_waiting;
       while (tempm!=mp_ctrl && !Bt(&tempm->target_cpu_mask,Gs->num))
           tempm=tempm->next;
       if (tempm!=mp_ctrl) {
           RemQue(tempm);
           LBts(&tempm->flags,MPCf_DISPATCHED);
           tempm->handler_cpu=Gs->num;
           LBtr(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0);
           LBtr(&Fs->task_flags,TSSf_IDLE);
           switch (tempm->cmd_code) {
               case MPCT_CALL:
                   tempm->result=CallInd(tempm->add,tempm->data);
                   Preempt(OFF);
                   Sti;
                   break;
               case MPCT_SPAWN_TASK:
                   if (tempm->desc)
                       tempm->tss=Spawn(tempm->add,tempm->data,tempm->desc);
                   else
                       tempm->tss=Spawn(tempm->add,tempm->data,"MP Job");
                   break;
           }
           if (Bt(&tempm->flags,MPCf_FREE_ON_COMPLETE)) {
               Free(tempm->desc);
               Free(tempm);
           } else {
               while (LBts(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0))
                   SwapInNextTask;
               InsQue(tempm,mp_ctrl->last_done);
               LBts(&tempm->flags,MPCf_DONE);
               LBtr(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0);
           }
           LBtr(&Gs->flags,CPUf_NOT_READY);
           LBts(&Fs->task_flags,TSSf_IDLE);
       } else
           LBtr(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0);
   }
}

MPCmdStruct *MPQueueJob(void *add,void *data=NULL,
             I1 *desc=NULL,
             U8 flags=1<<MPCf_FREE_ON_COMPLETE,
             BoolI1 spawn=FALSE,I8 target_cpu_mask=ALL_MASK)
{
   MPCmdStruct *tempm=AMAllocZ(sizeof(MPCmdStruct));
   if (desc)
       tempm->desc=AStrNew(desc);
   if (spawn)
       tempm->cmd_code=MPCT_SPAWN_TASK;
   else
       tempm->cmd_code=MPCT_CALL;
   tempm->add=add;
   tempm->data=data;
   tempm->target_cpu_mask=target_cpu_mask;
   tempm->flags=flags;
   PushFD;
   Cli;
   while (LBts(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0))
       SwapInNextTask;
   InsQue(tempm,mp_ctrl->last_waiting);
   LBtr(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0);
   PopFD;
   return tempm;
}

MPCmdStruct *MPJob(void *add,void *data=NULL,
             U8 flags=1<<MPCf_FREE_ON_COMPLETE,
             I8 target_cpu_mask=ALL_MASK)
//Set flags to zero if you wish to
//get the result.
{
   return MPQueueJob(add,data,NULL,flags,FALSE,target_cpu_mask);
}

I8 MPJobResult(MPCmdStruct *tempm)
{
   I8 result;
   while (!Bt(&tempm->flags,MPCf_DONE))
       SwapInNextTask;
   PushFD;
   Cli;
   while (LBts(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0))
       SwapInNextTask;
   RemQue(tempm);
   LBtr(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0);
   PopFD;
   result=tempm->result;
   Free(tempm->desc);
   Free(tempm);
   return result;
}

TssStruct *MPSpawn(void *add,I8 data=0,I1 *desc=NULL,I8 target_cpu=ALL_MASK)
{
   TssStruct *result;
   MPCmdStruct *tempm=MPQueueJob(add,data,desc,0,TRUE,target_cpu);
   while (!Bt(&tempm->flags,MPCf_DONE))
       SwapInNextTask;
   result=tempm->tss;
   PushFD;
   Cli;
   while (LBts(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0))
       SwapInNextTask;
   RemQue(tempm);
   LBtr(&sys_semas[SYS_SEMA_MP_CMD_CTRL_LOCK],0);
   PopFD;
   Free(tempm->desc);
   Free(tempm);
   return result;
}

void MPInitCPUTask()
{ //Called by multicore's seth task after $LK,"MPStart","MN:MPStart"$()
   //as the first thing a CPU does before waiting for jobs
   MPInitAPIC;
   Fs->rip=&MPWaitForTask;
   Fs->time_slice_start=GetTimeStamp;
   RestoreContext;
}

void MPStart()
{ //Called by adam during start-up
   //and can be called again to reset multicores.
   //must be called from core#0
   TssStruct *tss;
   I1 buf[128];
   U4 *d;
   MPMainStruct *mp=MP_VECTOR_ADDRESS;
   CPUStruct *c;
   BlkPool *code_bp,*data_bp,*saved_code_bp,*saved_data_bp;
   I8 shared_blks,i,old_mp_cnt;
   MPCmdStruct *tempm,*tempm1;
   RaxRbxRcxRdx ee;

   CpuId(0x1,&ee);
   if (!Bt(&ee.rdx,9))
       return;

   PushFD;
   Cli;
   if (mp_cnt>1) {
       old_mp_cnt=mp_cnt;
       MPHalt; //sets mp_cnt to 1
       BusyWait(10000);

       for (i=1;i<old_mp_cnt;i++) {
           data_bp=cpu_structs[i].data_bp;
           code_bp=cpu_structs[i].code_bp;
           if (data_bp && data_bp!=code_bp) {
               if (sys_data_bp)
                   Free2MegMemBlks(data_bp,data_bp->size>>12,sys_data_bp);
               else
                   Free2MegMemBlks(data_bp,data_bp->size>>12,sys_code_bp);
           }
           if (code_bp)
               Free2MegMemBlks(code_bp,code_bp->size>>12,sys_code_bp);
       }
 
       tempm=mp_ctrl->next_waiting;
       while (tempm!=mp_ctrl) {
           tempm1=tempm->next;
           RemQue(tempm);
           Free(tempm->desc);
           Free(tempm);
           tempm=tempm1;
       }

       tempm=mp_ctrl->next_done;
       while (tempm!=&mp_ctrl->next_done) {
           tempm1=tempm->next;
           RemQue(tempm);
           Free(tempm->desc);
           Free(tempm);
           tempm=tempm1;
       }
   }
   MemSet(&cpu_structs[1],0,sizeof(CPUStruct)*(MP_MAX_PROCESSORS-1));
   MemCpy(MP_VECTOR_ADDRESS,MP_INIT_START,MP_INIT_END-MP_INIT_START);
   mp->sys_temp_ptr=MAXGDT*16-1+gdttab><(U1 *)<<16;
   mp_cnt=1;
   mp_cnt_lock=0;

   d=MP_LVT3;
   *d=*d&0xFFFFFF00+MPN_VECTOR;
   WbInvd;

   d=MP_ICR_LOW;
   *d=0xC4500; //assert init IPI
   BusyWait(10000);

   *d=0xC4600+MPN_VECTOR; //start-up
   BusyWait(200);
   *d=0xC4600+MPN_VECTOR;$FG$

   BusyWait(100000);
   if (mp_cnt>MP_MAX_PROCESSORS)
       mp_cnt=MP_MAX_PROCESSORS;

   for (i=1;i<mp_cnt;i++) {
       c =&cpu_structs[i];
       //SPrintF not available yet.
       StrCpy(buf,"Seth Task CPU#0");
       buf[14]+=i;

       shared_blks=32; //32*2Meg blks for core data heaps
       if (sys_data_bp && sys_data_bp!=sys_code_bp)
           data_bp=Alloc2MegMemBlks(&shared_blks,sys_data_bp);
       else
           data_bp=Alloc2MegMemBlks(&shared_blks,sys_code_bp);
       BlkPoolInit(data_bp,shared_blks<<12);

       shared_blks=2; //2*2Meg blks for core code heaps
       code_bp=Alloc2MegMemBlks(&shared_blks,sys_code_bp);
       BlkPoolInit(code_bp,shared_blks<<12);

       saved_code_bp=Gs->code_bp;
       saved_data_bp=Gs->data_bp;
       Gs->code_bp=code_bp;
       Gs->data_bp=data_bp;

       tss=Spawn(&MPInitCPUTask,0,buf,NULL,NULL,DEFAULT_STACK,FALSE);
       //$LK,"TssStruct","MN:TssStruct"$ allocated of this core's seth_task's heap (Which is Adam)

       tss->in_queue_signature=TSSS_IN_QUEUE_SIGNATURE;
       InitCPUStruct(i,c);
       c->code_bp=code_bp;
       c->data_bp=data_bp;
       Gs->code_bp=saved_code_bp;
       Gs->data_bp=saved_data_bp;
       c->tr=InitRealTssStruct;
       c->seth_tss=tss;
       WbInvd;
   }
   PopFD;
}

void MPInit()
{ //Called by adam during start-up
   TssStruct *tss;
   RaxRbxRcxRdx ee;
   CPUStruct *c;
   CpuId(0x1,&ee);

   mp_cnt=1;
   mp_cnt_lock=0;

   mp_ctrl=AMAllocZ(sizeof(MPCmdCtrl));
   mp_ctrl->next_waiting=mp_ctrl->last_waiting=mp_ctrl;
   mp_ctrl->next_done=mp_ctrl->last_done=&mp_ctrl->next_done;

   mp_crash=AMAllocZ(sizeof(MPCrashStruct));

   c=cpu_structs=AMAllocZ(sizeof(CPUStruct)*MP_MAX_PROCESSORS);
   MemCpy(c,&sys_temp_cpu0_struct,sizeof(CPUStruct));
   c->address=cpu_structs;
   SetGs(c);
   tss=adam_tss;
   tss->time_slice_start=GetTimeStamp;
   tss->in_queue_signature=TSSS_IN_QUEUE_SIGNATURE;
   c->seth_tss=tss;
   c->code_bp=sys_code_bp;
   c->data_bp=sys_data_bp;
   c->tr=InitRealTssStruct;
   c->idle_tss=Spawn(0,NULL,"Idle",Fs,NULL,DEFAULT_STACK,FALSE);
   if (Bt(&ee.rdx,9)) {
//Unnecessary?
//  SetMSR(IA32_APIC_BASE,APIC_BASE+0x900);
       MPInitAPIC;
   }
}

void MPCrash()
{ //Entering the debugger from another core causes an interrupt on core#0
   //Which calls this routine.
   MPEOI;
   mp_cnt=1;
   Raw(ON);
   dc_flags|=DCF_SHOW_DOLLAR;
   coutln "MP Crash CPU#",mp_crash->cpu_num," Tss:",mp_crash->tss;
   WbInvd;
   Debugger(mp_crash->msg,mp_crash->msg_num);
}
