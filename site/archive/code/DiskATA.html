U0 ATABlkSelect(LTBlkDev *bdev,U8 blk,U8 cnt)
{
  if (bdev->type!=LTBDT_ATAPI && bdev->base1)
    OutU1(bdev->base1+ATAR1_CONTROL,0x8);
  if (bdev->flags & LTBDF_EXT_SIZE) {
    OutU1(bdev->base0+ATAR0_NSECTOR,cnt.u1[1]);
    OutU1(bdev->base0+ATAR0_SECTOR,blk.u1[3]);
    OutU1(bdev->base0+ATAR0_LCYL,blk.u1[4]);
    OutU1(bdev->base0+ATAR0_HCYL,blk.u1[5]);
    OutU1(bdev->base0+ATAR0_NSECTOR,cnt);
    OutU1(bdev->base0+ATAR0_SECTOR,blk);
    OutU1(bdev->base0+ATAR0_LCYL,blk.u1[1]);
    OutU1(bdev->base0+ATAR0_HCYL,blk.u1[2]);
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  } else {
    OutU1(bdev->base0+ATAR0_NSECTOR,cnt);
    OutU1(bdev->base0+ATAR0_SECTOR,blk);
    OutU1(bdev->base0+ATAR0_LCYL,blk.u1[1]);
    OutU1(bdev->base0+ATAR0_HCYL,blk.u1[2]);
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4|blk.u1[3]);
  }
}

BoolI8 ATAWaitNotBUSY(LTBlkDev *bdev,LTDate timeout)
{
  while (TRUE) {
    if (InU1(bdev->base0+ATAR0_STATUS)&S_BUSY)
      TaskNext;
    else
      return TRUE;
    if (timeout && BootTime>timeout)
      return FALSE;
  }
}

BoolI8 ATAWaitDRQ(LTBlkDev *bdev,LTDate timeout)
{
  while (TRUE) {
    if (!(InU1(bdev->base0+ATAR0_STATUS)&S_DRQ))
      TaskNext;
    else
      return TRUE;
    if (timeout && BootTime>timeout)
      return FALSE;
  }
}

BoolI8 ATANop(LTBlkDev *bdev,LTDate timeout)
{
  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  OutU1(bdev->base0+ATAR0_FEATURE,0);
  OutU1(bdev->base0+ATAR0_CMD,ATA_NOP);
  return ATAWaitNotBUSY(bdev,timeout);
}

U0 ATACmd(LTBlkDev *bdev,U1 cmd)
{
  OutU1(bdev->base0+ATAR0_FEATURE,0);
  OutU1(bdev->base0+ATAR0_CMD,cmd);
  bdev->last_time=GetTimeStamp;
  PortNop;
}

BoolI8 ATAGetResult(LTBlkDev *bdev,U1 *buf,I8 cnt,I8 _avail,BoolI8 one_read,LTDate timeout=0)
{
  I8 avail,overflow;
  bdev->flags&=~LTBDF_LAST_WAS_WRITE;
  MemSet(buf,0,cnt);
  while (cnt) {
    if (!ATAWaitDRQ(bdev,timeout))
      return FALSE;
    if (_avail)
      avail=_avail;
    else
      avail=InU1(bdev->base0+ATAR0_HCYL)<<8+InU1(bdev->base0+ATAR0_LCYL);
    if (avail) {
      if (avail>cnt) {
        overflow=avail-cnt;
        avail=cnt;
      } else
        overflow=0;
      if (avail&2)
        RepInU2(buf,avail>>1,bdev->base0+ATAR0_DATA);
      else
        RepInU4(buf,avail>>2,bdev->base0+ATAR0_DATA);
      cnt-=avail;
      buf+=avail;
      while (overflow) {
        InU2(bdev->base0+ATAR0_DATA);
        overflow-=2;
      }
      if (one_read)
        break;
    }
  }
  return ATAWaitNotBUSY(bdev,timeout);
}

BoolI8 ATAPIWritePktWord(LTBlkDev *bdev,U2 w,LTDate timeout=0)
{
  if (!ATAWaitDRQ(bdev,timeout))
    return FALSE;
  OutU2(bdev->base0+ATAR0_DATA,BigEndianU2(w));
  bdev->last_time=GetTimeStamp;
  return TRUE;
}

BoolI8 ATAPISetMaxSpeed(LTBlkDev *bdev)
{
  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  OutU1(bdev->base0+ATAR0_LCYL,0);
  OutU1(bdev->base0+ATAR0_HCYL,0);
  ATACmd(bdev,ATA_PACKETCMD);
  ATAPIWritePktWord(bdev,0xBB00); //Set speed
  ATAPIWritePktWord(bdev,0xFFFF); //read at max
  ATAPIWritePktWord(bdev,0xFFFF); //write at max
  ATAPIWritePktWord(bdev,0);
  ATAPIWritePktWord(bdev,0);
  ATAPIWritePktWord(bdev,0);
  return ATAWaitNotBUSY(bdev,0);
}


BoolI8 ATAPISeek(LTBlkDev *bdev,U8 native_blk)
{
  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  OutU1(bdev->base0+ATAR0_LCYL,0);
  OutU1(bdev->base0+ATAR0_HCYL,0);
  ATACmd(bdev,ATA_PACKETCMD);
  ATAPIWritePktWord(bdev,0x2B00); //Seek
  ATAPIWritePktWord(bdev,native_blk>>16);
  ATAPIWritePktWord(bdev,native_blk);
  ATAPIWritePktWord(bdev,0);
  ATAPIWritePktWord(bdev,0);
  ATAPIWritePktWord(bdev,0x0000);
  return ATAWaitNotBUSY(bdev,0);
}

BoolI8 ATAPIStartStop(LTBlkDev *bdev,BoolI1 start,LTDate timeout)
{
  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  ATACmd(bdev,ATA_PACKETCMD);

  //Start/Stop
  if (!ATAPIWritePktWord(bdev,0x1B00,timeout))          return FALSE;
  if (!ATAPIWritePktWord(bdev,0x0000,timeout))          return FALSE;
  if (start) {
    if (!ATAPIWritePktWord(bdev,0x0100,timeout))        return FALSE;
  } else {
    if (!ATAPIWritePktWord(bdev,0x0000,timeout))        return FALSE;
  }
  if (!ATAPIWritePktWord(bdev,0x0000,timeout))          return FALSE;
  if (!ATAPIWritePktWord(bdev,0x0000,timeout))          return FALSE;
  if (!ATAPIWritePktWord(bdev,0x0000,timeout))          return FALSE;
  return ATAWaitNotBUSY(bdev,timeout);
}

I8 ATAGetDeviceId(LTBlkDev *bdev,U2 **_id_record,LTDate timeout)
{
  U2 *id_record;
  if (_id_record) *_id_record=NULL;
  if (bdev->type!=LTBDT_ATAPI && bdev->base1)
    OutU1(bdev->base1+ATAR1_CONTROL,0x8);
  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  ATACmd(bdev,ATA_IDENTIFY_DEVICE);
  if (!ATAWaitNotBUSY(bdev,timeout))
    return LTBDT_NULL;
  if (InU1(bdev->base0+ATAR0_STATUS)&S_ERROR)
    return LTBDT_ATAPI;
  id_record=ACAlloc(512);
  if (!ATAGetResult(bdev,id_record,512,512,FALSE,timeout)) {
    Free(id_record);
    return LTBDT_NULL;
  }
  if (_id_record) *_id_record=id_record;
  return LTBDT_ATA;
}

I8 ATAReadNativeMax(LTBlkDev *bdev)
{ //returns zero on err
  I8 blk=0;
  BoolI1 okay=TRUE;

  if (bdev->type==LTBDT_ATAPI) {
    if (bdev->flags & LTBDF_EXT_SIZE)
      OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
    else
      OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
    ATACmd(bdev,ATA_DEV_RESET);
    if (!ATAWaitNotBUSY(bdev,0))
      okay=FALSE;
  } else {
    while (InU1(bdev->base0+ATAR0_STATUS) & S_BUSY) {
      if (bdev->flags&LTBDF_LAST_WAS_WRITE)
        OutU2(bdev->base0+ATAR0_DATA,0);
      else
        InU2(bdev->base0+ATAR0_DATA);
      TaskNext;
    }

    Free(bdev->device_id_record);
    if (ATAGetDeviceId(bdev,&bdev->device_id_record,0)==LTBDT_NULL)
      okay=FALSE;
    BEqu(&bdev->flags,LTBDf_EXT_SIZE,Bt(&bdev->device_id_record[86],10));
  }
  if (okay) {
    if (bdev->flags & LTBDF_EXT_SIZE && bdev->base1) {
      OutU1(bdev->base1+ATAR1_CONTROL,0x8);
      OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
      ATACmd(bdev,ATA_READ_NATIVE_MAX_EXT);
      if (ATAWaitNotBUSY(bdev,0)) {
        blk.u1[0]=InU1(bdev->base0+ATAR0_SECTOR);
        blk.u1[1]=InU1(bdev->base0+ATAR0_LCYL);
        blk.u1[2]=InU1(bdev->base0+ATAR0_HCYL);

        OutU1(bdev->base1+ATAR1_CONTROL,0x80);
        blk.u1[3]=InU1(bdev->base0+ATAR0_SECTOR);
        blk.u1[4]=InU1(bdev->base0+ATAR0_LCYL);
        blk.u1[5]=InU1(bdev->base0+ATAR0_HCYL);
      }
    } else {
      if (bdev->type!=LTBDT_ATAPI && bdev->base1)
        OutU1(bdev->base1+ATAR1_CONTROL,0x8);
      OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
      ATACmd(bdev,ATA_READ_NATIVE_MAX);
      if (ATAWaitNotBUSY(bdev,0)) {
        blk.u1[0]=InU1(bdev->base0+ATAR0_SECTOR);
        blk.u1[1]=InU1(bdev->base0+ATAR0_LCYL);
        blk.u1[2]=InU1(bdev->base0+ATAR0_HCYL);
        blk.u1[3]=InU1(bdev->base0+ATAR0_SELECT) & 0xF;
      }
    }
  }
  return bdev->max_blk=blk;
}

U8 ATAPIReadCapacity(LTBlkDev *bdev,U8 *_blk_size=NULL)
{ //Supposedly this can return a result +/- 75 sectors.
//Error might just be for music.
  BoolI1 unlock=BlkDevLock(bdev);
  U4 buf[2];
  if (ATAWaitNotBUSY(bdev,0)) {
    if (bdev->flags & LTBDF_EXT_SIZE)
      OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
    else
      OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
    OutU1(bdev->base0+ATAR0_LCYL,8);
    OutU1(bdev->base0+ATAR0_HCYL,0);
    ATACmd(bdev,ATA_PACKETCMD);
    ATAPIWritePktWord(bdev,0x2500);  //Read Capacity
    ATAPIWritePktWord(bdev,0x0000);
    ATAPIWritePktWord(bdev,0x0000);
    ATAPIWritePktWord(bdev,0x0000);
    ATAPIWritePktWord(bdev,0x0000);
    ATAPIWritePktWord(bdev,0x0000);
    if (!ATAGetResult(bdev,buf,8,0,TRUE))
      buf[0]=buf[1]=0;
  } else
    buf[0]=buf[1]=0;

  if (unlock) BlkDevUnlock(bdev);
  if (_blk_size) *_blk_size=BigEndianU4(buf[1]);
  return BigEndianU4(buf[0]);
}

ATAPITrackInfo *ATAPIReadTrackInfo(LTBlkDev *bdev,U8 blk)
{
  ATAPITrackInfo *result=CAlloc(sizeof(ATAPITrackInfo));
  BoolI1 unlock=BlkDevLock(bdev);
  if (ATAWaitNotBUSY(bdev,0)) {
    if (bdev->flags & LTBDF_EXT_SIZE)
      OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
    else
      OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
    OutU1(bdev->base0+ATAR0_LCYL,sizeof(ATAPITrackInfo)&0xFF);
    OutU1(bdev->base0+ATAR0_HCYL,sizeof(ATAPITrackInfo)>>8);
    ATACmd(bdev,ATA_PACKETCMD);
    ATAPIWritePktWord(bdev,0x5200);  //Read Track Info
    ATAPIWritePktWord(bdev,blk.u2[1]);
    ATAPIWritePktWord(bdev,blk.u2[0]);
    ATAPIWritePktWord(bdev,(sizeof(ATAPITrackInfo)&0xFF00)>>8);
    ATAPIWritePktWord(bdev,(sizeof(ATAPITrackInfo)&0x00FF)<<8);
    ATAPIWritePktWord(bdev,0x0000);
    if (!ATAGetResult(bdev,result,sizeof(ATAPITrackInfo),0,TRUE)) {
      Free(result);
      result=NULL;
    }
  } else {
    Free(result);
    result=NULL;
  }
  if (unlock) BlkDevUnlock(bdev);
  return result;
}

BoolI8 ATAInit(LTBlkDev *bdev)
{
  BoolI1 unlock=BlkDevLock(bdev),okay=FALSE;

  if (bdev->type==LTBDT_ATAPI)
    bdev->flags&=~LTBDF_EXT_SIZE;
  else
    bdev->flags|=LTBDF_EXT_SIZE;

  if (ATAReadNativeMax(bdev)) {
    ATABlkSelect(bdev,bdev->max_blk,0);
    if (bdev->flags&LTBDF_EXT_SIZE)
      ATACmd(bdev,ATA_SET_MAX_EXT);
    else
      ATACmd(bdev,ATA_SET_MAX);
    if (ATAWaitNotBUSY(bdev,0)) {
      okay=TRUE;
      if (bdev->type==LTBDT_ATAPI) {
        if (ATAPIStartStop(bdev,TRUE,0)) {
          if(!ATAPISetMaxSpeed(bdev))
            okay=FALSE;
        } else
          okay=FALSE;
      }
    }
  }
  if (unlock) BlkDevUnlock(bdev);
  return okay;
}

BoolI8 ATAPIWaitReady(LTBlkDev *bdev,LTDate timeout)
{
  while (TRUE) {
    if (!ATAWaitNotBUSY(bdev,timeout) ||
        !ATANop(bdev,timeout) ||
        !ATAPIStartStop(bdev,TRUE,timeout))
      return FALSE;
    if (InU1(bdev->base0+ATAR0_STATUS) & S_READY &&
        !InU1(bdev->base0+ATAR0_FEATURE));
      return TRUE;
    if (timeout && BootTime>timeout)
      return FALSE;
    ATAInit(bdev);
    TaskNext;
  }
}

U0 ATAReadBlks(LTBlkDev *bdev,U1 *buf, U8 blk, U8 cnt)
{
  I8 retries=3;
  BoolI1 unlock=BlkDevLock(bdev);

retry:
  ATABlkSelect(bdev,blk,cnt);
  if (bdev->flags & LTBDF_EXT_SIZE)
    ATACmd(bdev,ATA_MULTREAD_EXT);
  else
    ATACmd(bdev,ATA_MULTREAD);
  if (!ATAGetResult(bdev,buf,cnt*bdev->blk_size,BLK_SIZE,FALSE,BootTime+LTDATE_FREQ)) {
    if (retries--) {
      ATAWaitNotBUSY(bdev,0);
      goto retry;
    } else
      throw(EXCEPT_BLKDEV,6);
  }

  disk_reads+=(cnt*bdev->blk_size)>>BLK_SIZE_BITS;
  if (unlock) BlkDevUnlock(bdev);
}

BoolI8 ATAPIReadBlks2(LTBlkDev *bdev,U1 *buf, U8 native_blk, U8 cnt,BoolI1 lock,LTDate timeout)
{
  BoolI1 result,unlock;
  if (!cnt)
    return FALSE;
  if (lock)
    unlock=BlkDevLock(bdev);
  if (!ATAPIWaitReady(bdev,timeout))                    goto arb_false_done;
  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  OutU1(bdev->base0+ATAR0_LCYL,bdev->blk_size);
  OutU1(bdev->base0+ATAR0_HCYL,bdev->blk_size.u1[1]);
  ATACmd(bdev,ATA_PACKETCMD);
  if (!ATAPIWritePktWord(bdev,0xA800,timeout))          goto arb_false_done;
  if (!ATAPIWritePktWord(bdev,native_blk.u2[1],timeout))        goto arb_false_done;
  if (!ATAPIWritePktWord(bdev,native_blk,timeout))              goto arb_false_done;
  if (!ATAPIWritePktWord(bdev,cnt.u2[1],timeout))       goto arb_false_done;
  if (!ATAPIWritePktWord(bdev,cnt,timeout))             goto arb_false_done;
  if (!ATAPIWritePktWord(bdev,0x0000,timeout))          goto arb_false_done;
  if (!ATAGetResult(bdev,buf,cnt*bdev->blk_size,0,FALSE,timeout)) {
arb_false_done:
    result=FALSE;
  } else {
    disk_reads+=(cnt*bdev->blk_size)>>BLK_SIZE_BITS;
    result=TRUE;
  }
//  ATAPIStartStop(bdev,FALSE,0);
  if (lock && unlock) BlkDevUnlock(bdev);
  return result;
}

U0 ATAPIReadBlks(LTBlkDev *bdev,U1 *buf, U8 blk, U8 cnt)
{
  LTPartition *p=Drive2Partition(bdev->drive_letter_base);
  U8 retry,spc=bdev->blk_size>>BLK_SIZE_BITS,n,blk2,
    l2=bdev->max_reads<<1+spc<<1;
  U1 *cd_buf=MAlloc(l2<<BLK_SIZE_BITS);
  if (cnt) {
    if (blk<=bdev->max_reads)
      blk2=0;
    else
      blk2=RoundDownU8(blk-bdev->max_reads,spc);
    if (blk2+l2>p->size)
      l2=p->size-blk2;
    n=(l2+spc-1)/spc;

    retry=4;
    while (--retry)
      if (ATAPIReadBlks2(bdev,cd_buf,blk2/spc,n,TRUE,
        //n is 0x800 if max_reads.  Up to 8 additional seconds
        BootTime+7*LTDATE_FREQ+n*LTDATE_FREQ/0x100))
        break;

    if (!retry)
      ATAPIReadBlks2(bdev,cd_buf,blk2/spc,n,TRUE,0);
    if (bdev->flags & LTBDF_READ_CACHE)
      LTDCAdd(p,cd_buf,blk2,n*spc,FALSE,FALSE);
    MemCpy(buf,cd_buf+(blk-blk2)<<BLK_SIZE_BITS,cnt<<BLK_SIZE_BITS);
  }
  Free(cd_buf);
}

U0 ATARBlks(LTPartition *p,U1 *buf, U8 blk, U8 cnt)
{
  U8 n;
  LTBlkDev *bdev=p->bdev;
  BlkDevChk(bdev);
  while (cnt>0) {
    n=cnt;
    if (n>bdev->max_reads)
      n=bdev->max_reads;
    if (bdev->type==LTBDT_ATAPI)
      ATAPIReadBlks(bdev,buf,blk,n);
    else
      ATAReadBlks(bdev,buf,blk,n);
    buf+=n<<BLK_SIZE_BITS;
    blk+=n;
    cnt-=n;
  }
}

//This is for low level disk access
//Call WBlks() instead
U0 ATAWriteBlks(LTBlkDev *bdev,U1 *buf, U8 blk, U8 cnt)
{
  U8 i,U4s_avail,sectors_avail,retries=3;
  LTDate timeout;
  BoolI1 unlock=BlkDevLock(bdev);
retry:
  ATABlkSelect(bdev,blk,cnt);
  if (bdev->flags&LTBDF_EXT_SIZE)
    ATACmd(bdev,ATA_MULTWRITE_EXT);
  else
    ATACmd(bdev,ATA_MULTWRITE);
  bdev->flags|=LTBDF_LAST_WAS_WRITE;
  while (cnt) {
    timeout=BootTime+LTDATE_FREQ;
    while (TRUE) {
      i=InU1(bdev->base0+ATAR0_STATUS);
      if (i & (S_READY|S_SEEK)!=S_READY|S_SEEK ||
          !(i & S_DRQ)) {
        TaskNext;
      } else
        break;
      if (/* i&S_ERROR||*/ BootTime>timeout) {
        if (retries--) {
          ATAWaitNotBUSY(bdev,0);
          goto retry;
        } else
          throw(EXCEPT_BLKDEV,7);
      }
    }
    sectors_avail=1;
    U4s_avail=sectors_avail<<BLK_SIZE_BITS>>2;
    RepOutU4(buf,U4s_avail,bdev->base0+ATAR0_DATA);
    buf+=U4s_avail<<2;
    cnt-=sectors_avail;
    retries=3;
  }
  ATAWaitNotBUSY(bdev,0);
  if (unlock) BlkDevUnlock(bdev);
}

BoolI8 ATAPISync(LTBlkDev *bdev)
{
  BoolI1 okay=TRUE;
  if (!ATAWaitNotBUSY(bdev,0))
    okay=FALSE;
  else {
    if (bdev->flags & LTBDF_EXT_SIZE)
      OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
    else
      OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
    OutU1(bdev->base0+ATAR0_LCYL,0);
    OutU1(bdev->base0+ATAR0_HCYL,0);
    ATACmd(bdev,ATA_PACKETCMD);
    ATAPIWritePktWord(bdev,0x3500);  //Sync
    ATAPIWritePktWord(bdev,0x0000);
    ATAPIWritePktWord(bdev,0x0000);
    ATAPIWritePktWord(bdev,0x0000);
    ATAPIWritePktWord(bdev,0x0000);
    ATAPIWritePktWord(bdev,0x0000);
    if (!ATAWaitNotBUSY(bdev,0))
      okay=FALSE;
  }
  return okay;
}

U0 ATAPIClose(LTBlkDev *bdev,I8 close_field=0x200,I8 track=0)
{
//0x200 CD/DVD part 1
//0x300    DVD part 2
  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  OutU1(bdev->base0+ATAR0_LCYL,0);
  OutU1(bdev->base0+ATAR0_HCYL,0);
  ATACmd(bdev,ATA_PACKETCMD);
  ATAPIWritePktWord(bdev,0x5B00);  //Close
  ATAPIWritePktWord(bdev,close_field);
  ATAPIWritePktWord(bdev,track);
  ATAPIWritePktWord(bdev,0x0000);
  ATAPIWritePktWord(bdev,0x0000);
  ATAPIWritePktWord(bdev,0x0000);
  ATAWaitNotBUSY(bdev,0);
}

U0 ATAPIWriteBlks(LTBlkDev *bdev,U1 *buf, U8 native_blk, U8 cnt)
{
  U8 U4s_avail;
  U1 *buf2;
  ATAWaitNotBUSY(bdev,0);
  ATAPISeek(bdev,native_blk);

  OutU1(bdev->base0+ATAR0_FEATURE,0);
  OutU1(bdev->base0+ATAR0_LCYL,bdev->blk_size);
  OutU1(bdev->base0+ATAR0_HCYL,bdev->blk_size.u1[1]);
  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  OutU1(bdev->base0+ATAR0_CMD,ATA_PACKETCMD);
  ATAPIWritePktWord(bdev,0x0400);  //FMT
  ATAPIWritePktWord(bdev,native_blk.u2[1]);
  ATAPIWritePktWord(bdev,native_blk);
  ATAPIWritePktWord(bdev,cnt.u2[1]);
  ATAPIWritePktWord(bdev,cnt);
  ATAPIWritePktWord(bdev,0x0000);
  bdev->flags|=LTBDF_LAST_WAS_WRITE;
  ATAWaitNotBUSY(bdev,0);

  ATAPISeek(bdev,native_blk);

  if (bdev->flags & LTBDF_EXT_SIZE)
    OutU1(bdev->base0+ATAR0_SELECT,0xEF|bdev->unit<<4);
  else
    OutU1(bdev->base0+ATAR0_SELECT,0xE0|bdev->unit<<4);
  OutU1(bdev->base0+ATAR0_LCYL,bdev->blk_size);
  OutU1(bdev->base0+ATAR0_HCYL,bdev->blk_size.u1[1]);
  ATACmd(bdev,ATA_PACKETCMD);
  ATAPIWritePktWord(bdev,0xAA00);  //Write
  ATAPIWritePktWord(bdev,native_blk.u2[1]);
  ATAPIWritePktWord(bdev,native_blk);
  ATAPIWritePktWord(bdev,cnt.u2[1]);
  ATAPIWritePktWord(bdev,cnt);
  ATAPIWritePktWord(bdev,0x0000);
  buf2=buf+bdev->blk_size*cnt;
  while (buf<buf2) {
    ATAWaitDRQ(bdev,0);
    U4s_avail=(InU1(bdev->base0+ATAR0_HCYL)<<8+InU1(bdev->base0+ATAR0_LCYL))>>2;
    if (buf+U4s_avail<<2>buf2)
      U4s_avail=(buf2-buf)>>2;
    if (U4s_avail) {
      RepOutU4(buf,U4s_avail,bdev->base0+ATAR0_DATA);
      buf+=U4s_avail<<2;
      disk_writes+=U4s_avail>>(BLK_SIZE_BITS-2);
    }
  }
  ATAWaitNotBUSY(bdev,0);
}

U0 ATAWBlks(LTPartition *p,U1 *buf, U8 blk, U8 cnt)
{
  U8 n,spc;
  LTBlkDev *bdev=p->bdev;
  BoolI1 unlock;
  BlkDevChk(bdev);
  spc=bdev->blk_size>>BLK_SIZE_BITS;
  if (bdev->type==LTBDT_ATAPI) {
    unlock=BlkDevLock(bdev);
    ATAPIWaitReady(bdev,0);
  }
  while (cnt>0) {
    n=cnt;
    if (n>bdev->max_writes)
      n=bdev->max_writes;
    if (bdev->type==LTBDT_ATAPI)
      ATAPIWriteBlks(bdev,buf,blk/spc,(n+spc-1)/spc);
    else
      ATAWriteBlks(bdev,buf,blk,n);
    buf+=n<<BLK_SIZE_BITS;
    blk+=n;
    cnt-=n;
    disk_writes+=n;
  }
  if (bdev->type==LTBDT_ATAPI) {
    ATAPISync(bdev);
//    ATAPIStartStop(bdev,FALSE,0);
    if (unlock) BlkDevUnlock(bdev);
  }
}
