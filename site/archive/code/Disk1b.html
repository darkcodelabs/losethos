BoolI8 MapDrive(U1 drive_letter,LTPartition *p)
{
  drive_letter=ToUpper(drive_letter);
  if ('A'<=drive_letter<='Z') {
    drive_to_partition[drive_letter-'A']=p;
    p->drive_letter=drive_letter;
    return TRUE;
  } else
    return FALSE;
}

BoolI8 Drive(U1 drive_letter)
{
  LTPartition *p=Drive2Partition(drive_letter);
  LTBlkDev *bdev=p->bdev;
  BlkDevChk(bdev);
  if (p!=Fs->cur_partition) {
    if ( bdev->flags & LTBDF_REMOVABLE &&
       !(bdev->flags & LTBDF_INITIALIZED) )
      ChangeDisk(Partition2Drive(p));
    if (bdev->type==LTBDT_RAM ||
        bdev->type==LTBDT_NON_LT_FILE ||
        bdev->type==LTBDT_TEMP_LT_FILE)
      BlkDevInit(bdev);
  }
  Fs->cur_partition=p;
  Free(Fs->cur_dir);
  Fs->cur_dir=StrNew("/");
  switch (p->type) {
    case PT_LT:
    case PT_FAT32:
    case PT_ISO9660:
      return TRUE;
    default:
      PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
      return FALSE;
  }
}

BoolI8 Cd(U1 *dirname,BoolI1 make_dirs=FALSE)
{
  I8 maxlen;
  BoolI1 result=TRUE;
  U1 *change_to_buf,*new_cur_dir,*buf;
  LTPartition *p;
  U8 cur_dir_cluster=0;
  if (!*dirname) return TRUE;
  if (dirname[1]==':') {
    if (*dirname==':') {
      if (Fs->cur_partition!=default_partition)
        if (!Drive(*dirname)) return FALSE;
    } else {
      if (Fs->cur_partition!=
        Drive2Partition(*dirname))
          if (!Drive(*dirname)) return FALSE;
    }
    dirname+=2;
  }
  if (*dirname=='/' || !*dirname || !Fs->cur_dir) {
    Free(Fs->cur_dir);
    Fs->cur_dir=StrNew("/");
    if (*dirname=='/')
      dirname++;
  }
  change_to_buf=MStrUtil(dirname,
    SU_REM_LEADING|SU_REM_TRAILING|
    SU_DISCARD_PARITY|SU_REM_CTRL_CHARS);
  maxlen=StrLen(Fs->cur_dir)+1+StrLen(change_to_buf)+1;
  new_cur_dir=MAlloc(maxlen);
  buf=MAlloc(maxlen);
  StrCpy(new_cur_dir,Fs->cur_dir);
  while (*change_to_buf && result) {
    StrFirstRem(change_to_buf,"/",buf);
    if (!*buf)
      StrCpy(new_cur_dir,"/");
    else if (!StrCmp(buf,"..")) {
      StrLastRem(new_cur_dir,"/");
      if (!*new_cur_dir)
        StrCpy(new_cur_dir,"/");
    } else if (!StrCmp(buf,".")) {
      ;
    } else if (*buf) {
      if (!StrCmp(buf,"HOME")) {
        result=Cd(sys_home_dir);
        Free(new_cur_dir);
        new_cur_dir=MAlloc(maxlen+StrLen(sys_home_dir));
        StrCpy(new_cur_dir,sys_home_dir+2);
      } else {
        p=Fs->cur_partition;
        cur_dir_cluster=Name2DirCluster(p,new_cur_dir);
        switch (p->type) {
          case PT_LT:
            result=LTFSCd(buf,cur_dir_cluster);
            break;
          case PT_FAT32:
            result=FAT32Cd(buf,cur_dir_cluster);
            break;
          case PT_ISO9660:
            result=ISO1Cd(buf,cur_dir_cluster);
            break;
          default:
            PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
            result=FALSE;
        }
        if (!result && make_dirs) {
          Free(Fs->cur_dir);
          Fs->cur_dir=StrNew(new_cur_dir);
          result=MkDir(buf);
        }
        if (result) {
          if (StrCmp(new_cur_dir,"/"))
            StrCat(new_cur_dir,"/");
          StrCat(new_cur_dir,buf);
        }
      }
    }
  }
  Free(Fs->cur_dir);
  Fs->cur_dir=StrNew(new_cur_dir);
  Free(buf);
  Free(change_to_buf);
  Free(new_cur_dir);
  return result;
}

U0 LTFileAccessDel(LTFileAccess *fa)
{
  LTBlkDev *bdev;
  if (!fa) return;
  bdev=fa->old_partition->bdev;
  if (!(bdev->flags & LTBDF_INIT_IN_PROGRESS)) {
    if (fa->old_dir) {
      Drive(Partition2Drive(fa->old_partition));
      Cd(fa->old_dir);
    }
  } else {
    Fs->cur_partition=fa->old_partition;
    Free(Fs->cur_dir);
    Fs->cur_dir=StrNew("/");
  }
  Free(fa->old_dir);
  Free(fa->mask);
  Free(fa);
}

LTFileAccess *LTFileAccessNew(U1 *_mask,BoolI1 make_mask=FALSE,BoolI1 make_dirs=FALSE)
{
  BoolI1 valid=TRUE;
  U1 *buf,*mask,*temp_mask;
  BoolI1 old_silent;
  LTFileAccess *fa=CAlloc(sizeof(LTFileAccess));
  mask=MStrUtil(_mask,
    SU_REM_LEADING|SU_REM_TRAILING|
    SU_DISCARD_PARITY|SU_REM_CTRL_CHARS);
  temp_mask=mask;
  fa->old_dir=StrNew(Fs->cur_dir);
  fa->old_partition=Fs->cur_partition;
  if (*mask && mask[1]==':') {
    if (Fs->cur_partition!=Drive2Partition(*mask))
      if (!Drive(*mask)) valid=FALSE;
    mask+=2;
  }
  if (StrStr(mask,"HOME")) {  //TODO
    if (Fs->cur_partition!=*sys_home_dir)
      Drive(*sys_home_dir);
  }
  fa->p=Fs->cur_partition;
  PartitionChk(fa->p);
  buf=MAlloc(StrLen(mask)+2);
  StrCpy(buf,mask);
  fa->mask=MAlloc(StrLen(mask)+2);
  StrLastRem(buf,"/",fa->mask);
  if (*mask=='/' && !*buf)
    StrCpy(buf,"/");
  if (*buf)
    if (!Cd(buf,make_dirs))
      valid=FALSE;
  if (valid && make_mask) {
    if (!*fa->mask) {
      Free(fa->mask);
      fa->mask=StrNew("*");
    } else {
      if (!make_dirs || FileNameChk(fa->mask)) {
        old_silent=Silent(ON);
        if (Cd(fa->mask,make_dirs)) {
          Free(fa->mask);
          fa->mask=StrNew("*");
        }
        Silent(old_silent);
      }
    }
  }
  Free(buf);
  Free(temp_mask);
  if (!valid) {
    LTFileAccessDel(fa);
    fa=NULL;
//    throw(EXCEPT_FILE);
  }
  return fa;
}

LTDirEntry *FilesFind(U1 *wild,U8 fuf_flags=0)
{
  LTDirEntry *result=NULL;
  LTFileAccess *fa=LTFileAccessNew(wild,TRUE);
  if (fa) {
    result=SysFilesFind(fa->mask,fuf_flags);
    LTFileAccessDel(fa);
  }
  return result;
}

BoolI8 IsDir(U1 *dir_name)
{
  BoolI1 result=FALSE;
  U1 *mask=MSPrintF("%s/*",dir_name);
  BoolI1 old_silent=Silent(ON);
  LTFileAccess *fa=LTFileAccessNew(mask);
  if (fa) {
    result=TRUE;
    LTFileAccessDel(fa);
  }
  Free(mask);
  Silent(old_silent);
  return result;
}

U0 Dir(U1 *wild="*",BoolI1 full=FALSE)
{
  LTFileAccess *fa;
  fa=LTFileAccessNew(wild,TRUE);
  if (fa) {
    DirLT(fa->mask,full);
    LTFileAccessDel(fa);
  }
}


U0 Del(U1 *wild,BoolI1 print_msg=TRUE)
{
  LTFileAccess *fa=LTFileAccessNew(wild);
  if (fa) {
    switch (fa->p->type) {
      case PT_LT:
        LTFSFilesDel(fa->p,Fs->cur_dir,fa->mask,FALSE,print_msg);
        break;
      case PT_FAT32:
        FAT32FilesDel(fa->p,Fs->cur_dir,fa->mask,FALSE,print_msg);
        break;
      case PT_ISO9660:
        PutDefineErr("ST_NOT_WRITABLE");
        break;
      default:
        PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
    }
    LTFileAccessDel(fa);
  }
}

U1 *FileRead(U1 *filename,U8 *size=NULL,I8 *attr=NULL,BoolI1 raw=FALSE)
{
  GenericHashEntry *temph;
  U1 *absname;
  U1 *result=NULL;
  I8 size2=0,attr2=0;
  LTFileAccess *fa;
  ArcCompressStruct *ac=NULL;

  if (attr)
    *attr=0;
  absname=FileNameAbs(filename);
  if (!raw && (temph=HashFind(absname,adam_task->hash_table,HTT_FILE))) {
    if (FileAttr(absname) & LT_ATTR_COMPRESSED) {
      ac=temph->user_data0;
      if (size)
        *size=ac->expanded_size;
      if (attr)
        *attr=FileAttr(absname,*attr);
      result=ExpandBuf(ac);
    } else {
      result=MAlloc(temph->user_data1+1);
      MemCpy(result,temph->user_data0,temph->user_data1);
      result[temph->user_data1]=0; //Terminate
      if (size)
        *size=temph->user_data1;
    }
  } else {
    fa=LTFileAccessNew(absname);
    if (fa) {
      switch (fa->p->type) {
        case PT_LT:
          result=LTFSFileRead(fa->p,Fs->cur_dir,fa->mask,&size2,&attr2,raw,&ac);
          break;
        case PT_FAT32:
          result=FAT32FileRead(fa->p,Fs->cur_dir,fa->mask,&size2,&attr2,raw,&ac);
          break;
        case PT_ISO9660:
          result=ISO1FileRead(fa->p,Fs->cur_dir,fa->mask,&size2,&attr2,raw,&ac);
          break;
        default:
          PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
      }
      while (!result && StrCmp(Fs->cur_dir,"/")) {
        Cd("..");
        switch (Fs->cur_partition->type) {
          case PT_LT:
            result=LTFSFileRead(fa->p,Fs->cur_dir,fa->mask,&size2,&attr2,raw,&ac);
            break;
          case PT_FAT32:
            result=FAT32FileRead(fa->p,Fs->cur_dir,fa->mask,&size2,&attr2,raw,&ac);
            break;
          case PT_ISO9660:
            result=ISO1FileRead(fa->p,Fs->cur_dir,fa->mask,&size2,&attr2,raw,&ac);
            break;
          default:
            PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
        }
      }
      if (!result) {
        cout filename," ";
        PutDefineErr("ST_FILE_NOT_FOUND");
      }
      LTFileAccessDel(fa);
    }
    if (size)
      *size=size2;
    if (attr)
      *attr=attr2;
    if (!raw && attr2 & LT_ATTR_RESIDENT)
      HashGenericAdd(absname,HTT_FILE,AMAllocIdentical(ac),size2,0,adam_task);
    Free(ac);
  }
  Free(absname);
  return result;
}

U1 *TextFileRead(U1 *filename,U8 *attr=NULL)
{
  return FileRead(filename,NULL,attr);
}


BoolI8 FileFind(U1 *filename,LTDirEntry *rde=NULL,
      BoolI1 dir_only=FALSE,BoolI1 file_only=FALSE,BoolI1 scan_parents=FALSE)
{
  U8 cur_dir_cluster;
  BoolI1 result=FALSE;
  LTDirEntry de;
  LTFileAccess *fa=LTFileAccessNew(filename);
  if (fa) {
    cur_dir_cluster=Name2DirCluster(fa->p,Fs->cur_dir);
    switch (fa->p->type) {
      case PT_LT:
        result=LTFSFileFind(fa->p,cur_dir_cluster,fa->mask,
                          &de,dir_only,file_only);
        break;
      case PT_FAT32:
        result=FAT32FileFind(fa->p,cur_dir_cluster,fa->mask,
                          &de,dir_only,file_only);
        break;
      case PT_ISO9660:
        result=ISO1FileFind(fa->p,cur_dir_cluster,fa->mask,
                           &de,dir_only,file_only);
        break;
      default:
        PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
    }
    if (scan_parents) {
      while (!result && StrCmp(Fs->cur_dir,"/")) {
        Cd("..");
        cur_dir_cluster=Name2DirCluster(fa->p,Fs->cur_dir);
        switch (fa->p->type) {
           case PT_LT:
             result=LTFSFileFind(fa->p,cur_dir_cluster,fa->mask,
                              &de,dir_only,file_only);
             break;
           case PT_FAT32:
             result=FAT32FileFind(fa->p,cur_dir_cluster,fa->mask,
                              &de,dir_only,file_only);
             break;
           case PT_ISO9660:
             result=ISO1FileFind(fa->p,cur_dir_cluster,fa->mask,
                                &de,dir_only,file_only);
             break;
           default:
             PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
        }
      }
    }
    LTFileAccessDel(fa);
  }
  if (result && rde)
    MemCpy(rde,&de,sizeof(LTDirEntry));
  return result;
}

U8 FileSize(U1 *filename)
{
  LTDirEntry de;
  if (FileFind(filename,&de))
    return de.size;
  else
    return 0;
}

U8 FileWrite(U1 *filename,U1 *fbuf,
   U8 size,LTDate date=0,U8 attr=0,BoolI1 raw=FALSE)
{
  U8 c=0;
  GenericHashEntry *temph;
  LTFileAccess *fa=LTFileAccessNew(filename);
  ArcCompressStruct *ac=NULL;
  U1 *absname=FileNameAbs(filename);
  if (fa) {
    switch (fa->p->type) {
      case PT_LT:
        c=LTFSFileWrite(fa->p,Fs->cur_dir,fa->mask,
          fbuf,size,date,attr,raw,&ac);
        break;
      case PT_FAT32:
        c=FAT32FileWrite(fa->p,Fs->cur_dir,fa->mask,
          fbuf,size,date,attr,raw,&ac);
        break;
      case PT_ISO9660:
        PutDefineErr("ST_NOT_WRITABLE");
        break;
      default:
        PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
    }
    if (temph=HashFind(absname,adam_task->hash_table,HTT_FILE)) {
      if (!raw && ac && attr & LT_ATTR_RESIDENT) {
        Free(temph->user_data0);
        temph->user_data0=AMAllocIdentical(ac);
        temph->user_data1=size;
      } else
        HashRemDel(temph,adam_task->hash_table);
    } else if (!raw && ac && attr & LT_ATTR_RESIDENT)
      HashGenericAdd(absname,HTT_FILE,AMAllocIdentical(ac),size,0,adam_task);
    Free(ac);
    LTFileAccessDel(fa);
  }
  Free(absname);
  return c;
}
