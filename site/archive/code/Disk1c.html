BoolI8 CopySingleZ(U1 *f1,U1 *f2) //Just one file
{
  U1 *file_buf=NULL;
  U8 size,attr=0,c;
  LTDirEntry de;
 
  if (FileFind(f1,&de,FALSE,TRUE)) {
    file_buf=FileRead(f1,&size,&attr);
    attr=FileAttr(f2,attr);
    if (file_buf) {
      coutln "Copying ",f1," to ",f2;
      c=FileWrite(f2,file_buf,size,de.datetime,attr);
      Free(file_buf);
      return ToBool(c);
    } else PutDefineErr("ST_FILE_NOT_FOUND");
  } else PutDefineErr("ST_FILE_NOT_FOUND");
  return FALSE;
}

#define COPY_BUF_BLKS   0x80
BoolI8 CopySingle(U1 *f1,U1 *f2) //Just one file
{
  U1 *absf1=FileNameAbs(f1),
     *absf2=FileNameAbs(f2);
  U1 *buf;
  U8 cnt,n,size,attr1=FileAttr(f1),attr2=FileAttr(f2);
  LTFile *in_file=NULL,*out_file=NULL;
  U8 xsum=LT_XSUM,i,j;
  if (!StrCmp(absf1,absf2)) {
    Free(absf1);
    Free(absf2);
    return FALSE;
  }
  Free(absf1);
  Free(absf2);
  if (attr1!=attr2)
    return CopySingleZ(f1,f2);
  buf=MAlloc(COPY_BUF_BLKS<<BLK_SIZE_BITS);
  if (attr1 & LT_ATTR_CONTIGUOUS)
    in_file=FOpen(f1,"rc");
  else
    in_file=FOpen(f1,"r");
  if (in_file) {
    size=FSize(in_file);
    cnt=(size+BLK_SIZE-1)>>BLK_SIZE_BITS;
    if (attr2 & LT_ATTR_CONTIGUOUS)
      out_file=FOpen(f2,"wc",cnt);
    else
      out_file=FOpen(f2,"w",cnt);
    if (out_file) {
      coutln "Copying ",f1," to ",f2;
      j=size;
      while (cnt) {
        if (cnt>COPY_BUF_BLKS) {
          n=COPY_BUF_BLKS;
          i=n<<BLK_SIZE_BITS;
        } else {
          n=cnt;
          i=j;
        }
        FRBlks(in_file, buf,LTFB_NEXT_BLK,n);
        xsum^=ArcXSum(buf,i);
        FWBlks(out_file,buf,LTFB_NEXT_BLK,n);
        cnt-=n;
        j-=n<<BLK_SIZE_BITS;
      }
      if (in_file->de.xsum &&
          in_file->de.xsum!=xsum)
        throw(EXCEPT_ARC_XSUM);
      out_file->de.xsum=xsum;
      out_file->flags|=LTF_USE_OLD_DATETIME;
      out_file->de.datetime=in_file->de.datetime;
      out_file->de.size=size;
      out_file->de.expanded_size=in_file->de.expanded_size;
      out_file->de.attr=FileAttr(f2,in_file->de.attr);
      FClose(out_file);
      FClose(in_file);
      Free(buf);
      return TRUE;
    } else PutDefineErr("ST_FILE_NOT_FOUND");
    FClose(in_file);
  } else PutDefineErr("ST_FILE_NOT_FOUND");
  Free(buf);
  return FALSE;
}

BoolI8 MkDir(U1 *filename)
{
  BoolI1 result=FALSE;
  U1 *name;
  if (FileFind(filename,NULL,TRUE))
    return FALSE;
  LTFileAccess *fa=LTFileAccessNew(filename);
  if (fa) {
    if (*fa->mask) {
      if (!FileNameChk(fa->mask))
        PutDefineErr("ST_INVALID_FILENAME");
      else {
        coutln "Make Directory:",filename;
        name=MStrUtil(fa->mask,
        SU_REM_LEADING|SU_REM_TRAILING|
        SU_DISCARD_PARITY|SU_REM_CTRL_CHARS);
        switch (fa->p->type) {
          case PT_LT:
            result=LTFSMkDir(fa->p,Fs->cur_dir,name);
            break;
          case PT_FAT32:
            result=FAT32MkDir(fa->p,Fs->cur_dir,name);
            break;
          case PT_ISO9660:
            PutDefineErr("ST_NOT_WRITABLE");
            break;
          default:
            PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
        }
        Free(name);
      }
    }
    LTFileAccessDel(fa);
  }
  return result;
}

U0 SysDel(U1 *wild,BoolI1 make_mask)    //this is dangerous
{
  LTFileAccess *fa=LTFileAccessNew(wild,make_mask);
  if (fa) {
    switch (fa->p->type) {
      case PT_LT:
        LTFSFilesDel(fa->p,Fs->cur_dir,fa->mask,TRUE);
        break;
      case PT_FAT32:
        FAT32FilesDel(fa->p,Fs->cur_dir,fa->mask,TRUE);
        break;
      case PT_ISO9660:
        PutDefineErr("ST_NOT_WRITABLE");
        break;
      default:
        PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
    }
    LTFileAccessDel(fa);
  }
}

U0 Format(I8 drive_letter,BoolI1 quick=TRUE,BoolI1 confirm=TRUE,U8 type=PT_LT)
{//See PT_FAT32 for a list of supported file system types.

  LTPartition *p=Drive2Partition(drive_letter);
  BoolI1 old_silent=IsSilent;
  if (confirm) {
    if (!AreYouSure)
      return;
  }
  if (!p)  {
    PutDefineErr("ST_INVALID_DRIVE");
    return;
  }
  switch (type) {
    case PT_LT:
      Silent(ON);
      Drive(drive_letter);
      Silent(old_silent);
      LTFSFormat(drive_letter,quick);
      DiskCacheInvalidate(Drive2Partition(drive_letter));
      Drive(drive_letter);
      break;
    case PT_FAT32:
      Silent(ON);
      Drive(drive_letter);
      Silent(old_silent);
      FAT32Format(drive_letter,quick);
      DiskCacheInvalidate(Drive2Partition(drive_letter));
      Drive(drive_letter);
      break;
    case PT_ISO9660:
      PutDefineErr("ST_NOT_WRITABLE");
      break;
    default:
      PutDefineErr("ST_UNSUPPORTED_FILE_SYSTEM");
  }
}


U0 CDImageRead(U1 drive_letter,U1 *out_name)
{
  LTBlkDev *bdev=Drive2BlkDev(drive_letter);
  LTPartition *p=Drive2Partition(drive_letter);
  U1 *buf=MAlloc(COPY_BUF_BLKS<<BLK_SIZE_BITS);
  U1 *out_name2=DefaultExtension(out_name,"ISO");
  LTFile *f=FOpen(out_name2,"w");
  U8 n,spc=bdev->blk_size>>BLK_SIZE_BITS,blk=0,cnt,retry;
  BlkDevInit(bdev);
  if (bdev->type!=LTBDT_ATAPI)
    throw(EXCEPT_BLKDEV,3);
  if (!out_name)
    out_name=default_iso_name;
  cnt=RoundUpU8(p->size,spc);
  while (cnt) {
    if (cnt>COPY_BUF_BLKS)
      n=COPY_BUF_BLKS;
    else
      n=cnt;
    if (n>bdev->max_reads)
      n=bdev->max_reads;
    if (!Bt(&sys_semas[SYS_SEMA_NO_IO_SND],0))
      Snd(bdev->read_freq);

    retry=4;
    while (--retry)
      if (ATAPIReadBlks2(bdev,buf,blk/spc,n/spc,TRUE,
        //n is 0x800 if max_reads.  Up to 8 additional seconds
        BootTime+7*LTDATE_FREQ+n/spc*LTDATE_FREQ/0x100))
        break;

    if (!retry)
      ATAPIReadBlks2(bdev,buf,blk/spc,n/spc,TRUE,0);

    if (!Bt(&sys_semas[SYS_SEMA_NO_IO_SND],0))
      Snd(0);
    FWBlks(f,buf,blk,n);
    cnt-=n;
    blk+=n;
  }
  FClose(f);
  Free(buf);
  Free(out_name2);
}

class LTDualBuf
{
  U1 *buf0,*buf1;
  U8 in_buf,out_buf,cnt;
  U1 *filename;
  LTBlkDev *CD_bdev;
};


U0 CDImageWriteTask(LTDualBuf *d)
{
  U1 *buf;
  U8 n,blk=0,cnt=d->cnt;
  LTFile *f;
  Preempt(ON);
  TaskLocalInit; //needed for disk access
  if (FileAttr(d->filename)&LT_ATTR_CONTIGUOUS)
    f=FOpen(d->filename,"rc");
  else
    f=FOpen(d->filename,"r");
  while (cnt) {
    if (cnt>COPY_BUF_BLKS)
      n=COPY_BUF_BLKS;
    else
      n=cnt;
    if (n>d->CD_bdev->max_writes)
      n=d->CD_bdev->max_writes;
    buf=d->in_buf&1 ? d->buf1:d->buf0;
    while (d->in_buf>d->out_buf+1)
      TaskNext;
    FRBlks(f,buf,blk,n);
    if (!Bt(&sys_semas[SYS_SEMA_NO_IO_SND],0))
      Snd(d->CD_bdev->write_freq);
    d->in_buf++;
    cnt-=n;
    blk+=n;
  }
  FClose(f);
}

U0 CDImageWrite(U1 drive_letter,U1 *in_name,I8 media_type=MT_CD)
{
  LTDualBuf *d=CAlloc(sizeof(LTDualBuf));
  U1 *buf;
  U1 *in_name2,*in_name3;
  U8 i,n,spc,blk=0,cnt;
  LTPartition *p=Drive2Partition(drive_letter);
  LTBlkDev *bdev=p->bdev,*bdev2;
  TaskStruct *task;
  BoolI1 old_preempt;
  LTFile *f;

  if (!in_name)
    in_name=default_iso_name;
  in_name3=DefaultExtension(in_name,"ISO");
  in_name2=FileNameAbs(in_name3);
  f=FOpen(in_name2,"r");
  if (!f) {
    Free(d);
    return;
  }

  cnt=(FSize(f)+BLK_SIZE-1)>>BLK_SIZE_BITS;
  FClose(f);
  if (bdev->type!=LTBDT_ATAPI)
    throw(EXCEPT_BLKDEV,4);
  bdev2=Drive2BlkDev(*in_name2);
  while (bdev2->lock_forwarding)
    bdev2=bdev2->lock_forwarding;
  if ((bdev2->type==LTBDT_ATA || bdev2->type==LTBDT_ATAPI) &&
       bdev2->base0==bdev->base0) {
    coutln "Cannot Burn CD on same ATA controller as File.\r\n";
    throw(EXCEPT_BLKDEV,5);
  }

  bdev->flags|=LTBDF_READ_ONLY_OVERRIDE;
  BlkDevInit(bdev);
  spc=bdev->blk_size>>BLK_SIZE_BITS;
  if (p->size<cnt)
    p->size=cnt;

  d->filename=in_name2;
  d->CD_bdev=bdev;
  d->buf0=MAlloc(COPY_BUF_BLKS<<BLK_SIZE_BITS);
  d->buf1=MAlloc(COPY_BUF_BLKS<<BLK_SIZE_BITS);
  d->cnt=cnt;

  old_preempt=Preempt(OFF);
  task=Spawn(&CDImageWriteTask,d,"Write CD");

  while (d->in_buf<=d->out_buf)
    TaskNext;

  if (!Bt(&sys_semas[SYS_SEMA_NO_IO_SND],0))
    Snd(bdev->write_freq);
  BlkDevLock(bdev);
  ATAPIWaitReady(bdev,0);

  progress1=0; progress1_max=cnt;
  StrCpy(progress1_desc,"Writing");
  while (cnt) {
    if (cnt>COPY_BUF_BLKS)
      n=COPY_BUF_BLKS;
    else
      n=cnt;
    if (n>bdev->max_writes)
      n=bdev->max_writes;
    buf=d->out_buf&1 ? d->buf1:d->buf0;
    while (d->in_buf<=d->out_buf)
      TaskNext;
    ATAPIWriteBlks(bdev,buf,blk/spc,(n+spc-1)/spc);
    d->out_buf++;
    cnt-=n;
    blk+=n;
    progress1+=n;
  }
  ATAPISync(bdev);

  progress1=0; progress1_max=2;
  StrCpy(progress1_desc,"Closing");
  for (i=0;i<2;i++) {
    ATAPIClose(bdev,0x100,i); //Close tracks
    progress1++;
  }

  ATAPISync(bdev);

  ATAPIClose(bdev,0x200); //close disk
  ATAPISync(bdev);
  if (media_type==MT_DVD) {
    ATAPIClose(bdev,0x300);
    ATAPISync(bdev);
  }

  *progress1_desc=0;
  progress1=progress1_max=0;

  if (!Bt(&sys_semas[SYS_SEMA_NO_IO_SND],0))
    Snd(0);
  bdev->flags&=~LTBDF_READ_ONLY_OVERRIDE;
  BlkDevUnlock(bdev);
  Free(d->buf0);
  Free(d->buf1);
  Free(in_name2);
  Free(in_name3);
  Preempt(old_preempt);
  Free(d);
}
