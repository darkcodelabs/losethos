F64 t0=tJ;

U0 DrawIt(TaskStruct *task,GrDC *dc)
{
  F64 dt=(tJ-t0)*music_tempo;
  dc->flags|=DCF_TRANSFORMATION;
  GrRotZEqu(dc->r,0.25*Sin(2*pi*tJ));
  GrScaleMatEqu(dc->r,1.5+0.5*Sin(tJ));
  GrSetRotMat(dc,dc->r);
  if (Blink(music_tempo/2,tJ))
    GrSpritePlot3b(dc,12.0*dt%task->win_pixel_width,20,0,<1>);
  else
    GrSpritePlot3b(dc,12.0*dt%task->win_pixel_width,20,0,<2>);
}

U0 Song()
{
  SettingsPush; //See SettingsPush
  Fs->draw_it=&DrawIt;
  Fs->task_end_cb=&SndTaskEndCB;

  music_tempo=3.5;
  t0=tJ;
  try {
    while (!ScanKey) {
      Play("3qG2G3DBsDCDCqRCG",
          "$CL$$BG,9$\r\n\r\n$BG,2$$CM-LE,3$$FG,14$Baa, \0the \0grass \0is \0green.\r\n\0 \0 \0 \0 \0 \0 \0");
      Play("G2G3DBsDCDCqRCG",
          "This \0must \0be \0a \0dream.\r\n\0 \0 \0 \0 \0 \0 \0");
      Play("EeGFqBBDA2eGGqR",
          "Thanks, \0my \0 \0shep\0herd.  \0You \0are \0good.\r\n\0 \0 \0");
      Play("3EeGFqBBDA2eGGqR",
          "Thanks, \0my \0 \0shep\0herd.  \0You \0are \0good.\r\n\0 \0 \0");
    }
  } catch
    CatchAll;
  SettingsPop;
}

Song;
