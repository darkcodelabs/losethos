F64 t0=tJ;
U0 DrawIt(TaskStruct *task,GrDC *dc)
{
  nounusedwarn task;
  F64 dt=(tJ-t0)*music_tempo;
  dc->flags|=DCF_TRANSFORMATION;
  GrDCAllocDepthBuf(dc);
  dc->x=320;
  dc->y=27;
  dc->z=GR_Z_ALL;
  GrRotXEqu(dc->r,0.140*pi);
  GrRotZEqu(dc->r,0.250*pi*Sin(2*pi*dt/2));
  GrRotYEqu(dc->r,0.230*pi);
  GrSpritePlot3(dc,0,0,0,<1>);
}
