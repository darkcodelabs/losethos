F64 t0=tJ;

#define WOOD_X  220
#define WOOD_Y  45

U0 DrawIt(TaskStruct *task,GrDC *dc)
{
  nounusedwarn task;
  U8 *img;
  F64 dt=(tJ-t0)*music_tempo;
  I64 r;
 
  if (dt<8)
    r=3;
  else if (dt<18)
    r=dt-5;
  else
    r=18-5;
  dc->color=BLUE;
  GrEllipse  (dc,WOOD_X,WOOD_Y,r,1.5*r,pi/8);
  GrFloodFill(dc,WOOD_X,WOOD_Y);

  if (dt<20)
    img=<1>;
  else if (dt%0.4>.2)
    img=<2>;
  else
    img=<3>;
  GrSpritePlot3(dc,WOOD_X,WOOD_Y,0,img);
}
