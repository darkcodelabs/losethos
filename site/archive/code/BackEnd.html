/*Intermediate Code
To
Machine Code

RAX,RBX,RCX and RDX can be trashed by
each intermediate code's output code.
However, intermediate codes must be
coupled together based on the arg and
result type specifications in the
IcArg.RAX is the most common register
for coupling intermediate codes.

Internal calculations take place on
64-bit values, so anything which has
found it's way into a register has been
sign or zero extended to 64-bits.

The IcMov() routine is commonly used.
It needs a register for zero-extensions
when dealing with U4's and when
shuffling things around.I try to use
the RBX register as a scratch register
available for sub-intermediate codes to
trash.Don't assume RBX will be
preserved by IcMov() or other
sub-intermediate code routines.If you
are not calling such routines, you can
use it.

You might get away with returning a
value from a subintermediate code
helper routine in RBX, but maybe not
because it has to zero-extend U4's and
I know no way besides XORing ahead and
moving a U4 in, which requires two
registers.



Code size changing during passes could
cause problems.Jumps will be
shortened (not lengthened) on
successive passes and contraction does
not hurt jumps, but contraction can
hurt a glbl displacement.I decided
all code must reside in the lowest 2Gig
allowing 32-bit relative addressing
for CALLS and JMPS.To allow code
beyond 2Gig, indirect calls and jmps
could be used.(Load a 64-bit value in
a register and call the register.)

*/

U0 IcU1(IntermediateCode *tempi,U1 b)
{
  if (tempi->ic_cnt>=IC_BODY_SIZE-1|| tempi->ic_ext_body) {
    if (!tempi->ic_ext_body) {
      tempi->ic_ext_body=MAlloc(256);
      MemCpy(tempi->ic_ext_body,tempi->ic_body,tempi->ic_cnt);
    }
    tempi->ic_ext_body[tempi->ic_cnt++]=b;
  } else
    tempi->ic_body[tempi->ic_cnt++]=b;
}

BoolI8 IcLastIns(IntermediateCode *tempi,IntermediateCode **tempi1,U1 **body)
{
  IntermediateCode *tempil1;
  if (tempi->ic_last_cnt>=0) {
    if (tempi->ic_ext_body)
      *body=tempi->ic_ext_body;
    else
      *body=tempi->ic_body;
    *tempi1=tempi;
    return TRUE;
  } else {
    if (!tempi->ic_cnt &&
      (tempil1=OptLag1(tempi)) && tempil1->ic_last_cnt>=0) {
      if (tempil1->ic_ext_body)
        *body=tempil1->ic_ext_body;
      else
        *body=tempil1->ic_body;
      *tempi1=tempil1;
      return TRUE;
    } else {
      *body=NULL;
      *tempi1=NULL;
      return FALSE;
    }
  }
}


U0 IcAddRSP(IntermediateCode *tempi,I8 i,BoolI1 optimize=TRUE)
{
  I8 j,last_cnt;
  U1 *body;
  IntermediateCode *tempil1;
  if (optimize && IcLastIns(tempi,&tempil1,&body)) {
    j=tempil1->ic_cnt;
    if (tempil1->ic_last_cnt==j-4 && body[j-3]==0x83 && body[j-4]==0x48) {
      if (body[j-2]==0xEC)
        j=-body[j-1]><(I1);
      else if (body[j-2]==0xC4)
        j=body[j-1]><(I1);
      else
        j=0;
    } else if (tempil1->ic_last_cnt==j-7 && body[j-6]==0x81 && body[j-7]==0x48) {
      if (body[j-5]==0xEC)
        j=-body[j-4]><(I4);
      else if (body[j-5]==0xC4)
        j=body[j-4]><(I4);
      else
        j=0;
    } else
      j=0;
    if (j) {
      if (tempi==tempil1) {
        tempi->ic_cnt=tempi->ic_last_cnt;
        i+=j;
      } else if (!(tempi->ic_flags&ICF_PREVIOUS_DELETED)) {
        tempil1->ic_flags|=ICF_DEL_PREVIOUS_INS;
        tempi->ic_flags=tempi->ic_flags&~ICF_CODE_FINAL|ICF_PREVIOUS_DELETED;
        i+=j;
      }
    }
  }
  last_cnt=tempi->ic_cnt;
  if (i>0) {
    if (i<=MAX_I1)
      IcU4(tempi,0xC48348+i<<24);
    else if (i<=MAX_I4) {
      IcU3(tempi,0xC48148);
      IcU4(tempi,i);
    } else
      throw(EXCEPT_COMPILER,31);
  } else if (i<0) {
    i=-i;
    if (i<=MAX_I1)
      IcU4(tempi,0xEC8348+i<<24);
    else if (i<=MAX_I4) {
      IcU3(tempi,0xEC8148);
      IcU4(tempi,i);
    } else
      throw(EXCEPT_COMPILER,31);
  }
  if (optimize && tempi->ic_cnt>last_cnt)
    tempi->ic_last_cnt=last_cnt;
}

U0 CmpNoteFloatOp(LexStruct *lx,IntermediateCode *tempi,BoolI1 dont_pushable,BoolI1 dont_popable,I8 pos)
{
  BoolI1 link=FALSE;
  if (lx->pass==7 && lx->last_float_op_ic && lx->last_dont_popable && dont_pushable) {
    nobound_switch (pos) {
      case CN_P3:
        if (lx->last_float_op_ic!=tempi && lx->dont_push_float)
          link=TRUE;
        break;
      case CN_P2:
        if (lx->last_float_op_ic!=tempi && lx->dont_push_float)
          link=TRUE;
        break;
      case CN_P1:
        if (lx->last_float_op_ic!=tempi && lx->dont_push_float)
          link=TRUE;
        break;
      case CN_MAIN:
        if (lx->last_float_op_ic!=tempi) {
          if (lx->dont_push_float) {
            if (!(tempi->ic_flags & ICF_P1_FIRST))
              tempi->ic_flags|=ICF_ALTERNATE_TEMPLATE;
            else
              tempi->ic_flags&=~ICF_ALTERNATE_TEMPLATE;
            link=TRUE;
          }
        } else {
          if (lx->last_float_op_pos==CN_P1 ^^ !(tempi->ic_flags & ICF_P1_FIRST))
            tempi->ic_flags|=ICF_ALTERNATE_TEMPLATE;
          else
            tempi->ic_flags&=~ICF_ALTERNATE_TEMPLATE;
          link=TRUE;
        }
        break;
      case CN_R:
        if (lx->last_float_op_ic==tempi && lx->last_float_op_pos==CN_MAIN)
          link=TRUE;
        break;
    }
    if (link) {
      if (!Bts(&lx->last_float_op_ic->ic_flags,ICf_DONT_POP_FLOAT0+lx->last_ic_float_op_num))
        lx->last_float_op_ic->ic_flags&=~ICF_CODE_FINAL;
      if (!Bts(&tempi->ic_flags,ICf_DONT_PUSH_FLOAT0+lx->cur_ic_float_op_num))
        tempi->ic_flags&=~ICF_CODE_FINAL;
    }
  }
  lx->last_float_op_ic=tempi;
  lx->last_dont_pushable=dont_pushable;
  lx->last_dont_popable=dont_popable;
  lx->last_ic_float_op_num=lx->cur_ic_float_op_num++;
  lx->last_float_op_pos=pos;
  if (lx->cur_ic_float_op_num>4)
    throw(EXCEPT_COMPILER,29);
}

U0 CmpSetFloatOpPushPop(LexStruct *lx,IntermediateCode *tempi,BoolI1 *dont_push_float,BoolI1 *dont_pop_float)
{
  if (lx->pass==7) {
    *dont_push_float=FALSE;
    *dont_pop_float =FALSE;
    tempi->ic_flags&=~ICF_CODE_FINAL;
  } else {
    *dont_push_float=Bt(&tempi->ic_flags,ICf_DONT_PUSH_FLOAT0+lx->cur_ic_float_op_num);
    *dont_pop_float=Bt(&tempi->ic_flags,ICf_DONT_POP_FLOAT0+lx->cur_ic_float_op_num);
  }
}

U0 IcCopyTemplate(LexStruct *lx,IntermediateCode *tempi,U8 op,
  BoolI1 off_the_record,BoolI1 dont_pushable,BoolI1 dont_popable,I8 pos)
{
  BoolI1 dont_push_float,dont_pop_float,alt;
  U1 *ptr;
  I8 i=0;
  if (!off_the_record) {
    if (tempi->ic_flags&ICF_ALTERNATE_TEMPLATE)
      alt=TRUE;
    else
      alt=FALSE;
    CmpSetFloatOpPushPop(lx,tempi,&dont_push_float,&dont_pop_float);
  } else {
    dont_push_float=FALSE;
    dont_pop_float=FALSE;
    alt=FALSE;
  }
  if (alt && dont_push_float && !dont_pop_float) {
    ptr=cmp_templates_dont_push2[op];
    i=cmp_templates_dont_push2[op+1]-ptr;
  }
  if (!i) {
    if (dont_push_float) {
      if (dont_pop_float) {
        ptr=cmp_templates_dont_push_pop[op];
        i=cmp_templates_dont_push_pop[op+1]-ptr;
      } else {
        ptr=cmp_templates_dont_push[op];
        i=cmp_templates_dont_push[op+1]-ptr;
      }
    } else {
      if (dont_pop_float) {
        ptr=cmp_templates_dont_pop[op];
        i=cmp_templates_dont_pop[op+1]-ptr;
      } else {
        ptr=cmp_templates[op];
        i=cmp_templates[op+1]-ptr;
      }
    }
  }
  if (tempi->ic_cnt+i>=IC_BODY_SIZE) {
    if (!tempi->ic_ext_body) {
      tempi->ic_ext_body=MAlloc(256);
      MemCpy(tempi->ic_ext_body,tempi->ic_body,tempi->ic_cnt);
    }
    MemCpy(&tempi->ic_ext_body[tempi->ic_cnt],ptr,i);
  } else
    MemCpy(&tempi->ic_body[tempi->ic_cnt],ptr,i);
  if (!off_the_record)
    CmpNoteFloatOp(lx,tempi,dont_pushable,dont_popable,pos);
  tempi->ic_cnt+=i;
}

U8 IcModr(U8 r,U8 t2,U8 r2,I8 d2)
{
  //result.u[0] is type 0=[REG],1=disp8[REG],2=disp32[REG],3=REG,4=RIP_REL
  //result.u[1] is REX
  //result.u[2] is ModR
  //result.u[3] is SIB
  U8 result=0;
  if (t2&TY_MASK>TY_RIP_DISP32) {
    PrintF("Callers:%P;%P\r\n",Caller,Caller(2));
    throw(EXCEPT_COMPILER,20);
  }
  if (t2&IT_MASK<IT_I8)
    result.u1[1]=0x40;
  else
    result.u1[1]=0x48;
  if (r>7) {
    result.u1[1]+=4;
    r&=7;
  }
  switch (t2>>4) {
    case TY_RIP_DISP32>>4:
      result.u1[2]=0x05+r<<3;
      result.u1[0]=4;
      if (result.u1[1]==0x40 &&
          (t2&IT_MASK>=IT_I2 ||
           r<4))
        result.u1[1]=0;
      break;
    case TY_REG>>4:
      if (r2>7) {
        result.u1[1]++;
        r2&=7;
      }
      result.u1[2]=0xC0+r<<3+r2;
      result.u1[0]=3;
      if (result.u1[1]==0x40 &&
          (t2&IT_MASK>=IT_I2 ||
           r<4 && r2<4))
        result.u1[1]=0;
      break;
    case TY_DISP>>4:
      if (r2>7) {
        result.u1[1]++;
        r2&=7;
      }
      if (!d2) {
        result.u1[2]=r<<3+r2;
        result.u1[0]=0;
      } else if (MIN_I1<=d2<=MAX_I1) {
        result.u1[2]=0x40+r<<3+r2;
        result.u1[0]=1;
      } else {
        result.u1[2]=0x80+r<<3+r2;
        result.u1[0]=2;
      }
      if (result.u1[1]==0x40 &&
          (t2&IT_MASK>=IT_I2 ||
           r<4))
        result.u1[1]=0;
      break;
    case TY_SIB>>4:
      if (7<r2.u1[0]<CREG_NONE)
        result.u1[1]++;
      if (r2.u1[1]&15>7)
        result.u1[1]+=2;
      if (r2.u1[0]==CREG_NONE) {
        result.u1[3]=5+(r2.u1[1]&7)<<3+r2.u1[1]&0xC0;
        result.u1[2]=4+r<<3;
        result.u1[0]=2;
      } else {
        result.u1[3]=r2.u1[0]&7+(r2.u1[1]&7)<<3+r2.u1[1]&0xC0;
        if (!d2 && r2.u1[0]&7!=CREG_RBP) {
          result.u1[2]=4+r<<3;
          result.u1[0]=0;
        } else if (MIN_I1<=d2<=MAX_I1) {
          result.u1[2]=0x44+r<<3;
          result.u1[0]=1;
        } else {
          result.u1[2]=0x84+r<<3;
          result.u1[0]=2;
        }
      }
      if (result.u1[1]==0x40 &&
          (t2&IT_MASK>=IT_I2 ||
           r<4))
        result.u1[1]=0;
      break;
  }
  return result;
}

U0 IcSlashOp(IntermediateCode *tempi,U8 t1,U8 r1,I8 d1,U8 op,I8 ip)
{
  I8 i=IcModr(op.u1[0],t1,r1,d1);
  if (tempi->ic_flags&ICF_LOCK && op!=SLASH_OP_MOV)
    IcU1(tempi,0xF0);
  switch (t1&IT_MASK) {
    case IT_I1:
    case IT_U1:
      IcRex(tempi,i.u1[1]);
      IcU2(tempi,i.u1[2]<<8+op.u1[1]);
      break;
    case IT_I2:
    case IT_U2:
      Ic66Rex(tempi,i.u1[1]);
      IcU2(tempi,i.u1[2]<<8+op.u1[2]);
      break;
    default:
      if (i.u1[1]!=0x48 || !op.u1[6])
        IcRex(tempi,i.u1[1]);
      IcU2(tempi,i.u1[2]<<8+op.u1[2]);
  }
  if (t1&TY_MASK==TY_SIB)
    IcU1(tempi,i.u1[3]);
  if (i.u1[0]==1)
    IcU1(tempi,d1);
  else if (i.u1[0]==2)
    IcU4(tempi,d1);
  else if (i.u1[0]==4) {
    IcU4(tempi,d1-(ip+tempi->ic_cnt+4));
    tempi->ic_flags&=~ICF_CODE_FINAL;
  }
}

U0 IcPush(IntermediateCode *tempi,U8 t1,U8 r1,I8 d1,I8 ip)
{
  U8 i;
  switch (t1>>4) {
    case TY_REG>>4:
      if (r1>7)
        IcU2(tempi,0x5049+(r1&7)<<8);
      else
        IcU1(tempi,0x50+r1);
      break;
    case TY_IMM>>4:
      if (MIN_I1<=d1<=MAX_I1)
        IcU2(tempi,0x6A+d1<<8);
      else if (MIN_I4<=d1<=MAX_I4) {
        IcU1(tempi,0x68);
        IcU4(tempi,d1);
      } else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RBX,0,t1,r1,d1,ip);
        IcU1(tempi,0x50+CREG_RBX);
      }
      break;
    case TY_DISP>>4:
      i=0x48;
      switch (t1&IT_MASK) {
        case IT_I8:
        case IT_U8:
        case IT_F8:
          if (r1>7) {
            i++;
            r1&=7;
          }
          if (!d1)
            IcU3(tempi,0x30FF00+r1<<16+i);
          else if (MIN_I1<=d1<=MAX_I1) {
            IcU3(tempi,0x70FF00+r1<<16+i);
            IcU1(tempi,d1);
          } else {
            IcU3(tempi,0xB0FF00+r1<<16+i);
            IcU4(tempi,d1);
          }
          break;
        default:
          IcMov(tempi,TY_REG+IT_I8,CREG_RBX,0,t1,r1,d1,ip);
          IcU2(tempi,0x5048+CREG_RBX<<8);
      }
      break;
    case TY_SIB>>4:
      i=0x48;
      switch (t1&IT_MASK) {
        case IT_I8:
        case IT_U8:
        case IT_F8:
          if (7<r1.u1[0]<CREG_NONE)
            i++;
          if (r1.u1[1]&15>7)
            i+=2;
          if (r1.u1[0]==CREG_NONE) {
            r1=CREG_RBP+(r1.u1[1]&7)<<3+r1.u1[1]&0xC0;
            IcU4(tempi,0x34FF00+r1<<24+i);
            IcU4(tempi,d1);
          } else {
            r1=r1.u1[0]&7+(r1.u1[1]&7)<<3+r1.u1[1]&0xC0;
            if (!d1 && r1.u1[0]&7!=CREG_RBP)
              IcU4(tempi,0x34FF00+r1<<24+i);
            else if (MIN_I1<=d1<=MAX_I1) {
              IcU4(tempi,0x74FF00+r1<<24+i);
              IcU1(tempi,d1);
            } else {
              IcU4(tempi,0xB4FF00+r1<<24+i);
              IcU4(tempi,d1);
            }
          }
          break;
        default:
          IcMov(tempi,TY_REG+IT_I8,CREG_RBX,0,t1,r1,d1,ip);
          IcU2(tempi,0x5048+CREG_RBX<<8);
      }
      break;
    case TY_RIP_DISP32>>4:
      switch (t1&IT_MASK) {
        case IT_I8:
        case IT_U8:
        case IT_F8:
          IcU3(tempi,0x35FF48);
          IcU4(tempi,d1-(ip+tempi->ic_cnt+4));
          tempi->ic_flags&=~ICF_CODE_FINAL;
          break;
        default:
          IcMov(tempi,TY_REG+IT_I8,CREG_RBX,0,t1,r1,d1,ip);
          IcU2(tempi,0x5048+CREG_RBX<<8);
      }
      break;
    case TY_STK>>4:
      break;
    default:
      IcMov(tempi,TY_REG+IT_I8,CREG_RBX,0,t1,r1,d1,ip);
      IcU2(tempi,0x5048+CREG_RBX<<8);
  }
}

U0 IcPushRegs(IntermediateCode *tempi,U8 mask)
{
  I8 i;
  for (i=0;i<NUM_REGS;i++) {
    if (Bt(&mask,i)) {
      if (i>7)
        IcU2(tempi,0x5049+(i&7)<<8);
      else
        IcU1(tempi,0x50+i);
    }
  }
}

U0 IcPop(IntermediateCode *tempi,U8 t1,U8 r1,I8 d1,I8 ip)
{
  switch (t1>>4) {
    case TY_NULL>>4:
      IcAddRSP(tempi,8);
      break;
    case TY_REG>>4:
      if (r1>7)
        IcU2(tempi,0x5849+(r1&7)<<8);
      else
        IcU1(tempi,0x58+r1);
      break;
    case TY_DISP>>4:
    case TY_RIP_DISP32>>4:
    case TY_SIB>>4:
      if (t1&IT_MASK<IT_I8) {
        IcU1(tempi,0x58+CREG_RBX);
        IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,CREG_RBX,0,ip);
      } else
        IcSlashOp(tempi,t1,r1,d1,SLASH_OP_POP,ip);
      break;
    default:
      IcU1(tempi,0x58+CREG_RBX);
      IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,CREG_RBX,0,ip);
  }
}

U0 IcPopRegs(IntermediateCode *tempi,U8 mask)
{
  I8 i;
  for (i=NUM_REGS-1;i>=0;i--) {
    if (Bt(&mask,i)) {
      if (i>7)
        IcU2(tempi,0x5849+(i&7)<<8);
      else
        IcU1(tempi,0x58+i);
    }
  }
}

U0 IcZero(IntermediateCode *tempi,U8 r)
{
  if (r>7) {
    r&=7;
    IcU3(tempi,0xC0334D+r<<16+r<<19);
  } else
    IcU2(tempi,0xC033+r<<8+r<<11);
}

U0 IcOr(IntermediateCode *tempi,U8 r)
{
  U8 i=0xC00B48; //OR R,R
  if (r>7) {
    i+=5;
    r&=7;
  }
  IcU3(tempi,i+r<<16+r<<19);
}

U8 IcBuiltInFloatConst(F8 d)
{ //Returns 2-byte opcode for FLD const or zero
  if (!d)
    return 0xEED9;
  else if (d==1.0)
    return 0xE8D9;
  else if (GetOpt(OPTf_NO_BUILTIN_CONST))
    return 0;
  else if (d==pi)
    return 0xEBD9;
  else if (d==log2_10)
    return 0xE9D9;
  else if (d==log2_e)
    return 0xEAD9;
  else if (d==log10_2)
    return 0xECD9;
  else if (d==loge_2)
    return 0xEDD9;
  else
    return 0;
}


U0 IcMov(IntermediateCode *tempi,U8 t1,U8 r1,I8 d1,U8 t2,U8 r2,I8 d2,I8 ip)
{
  I8 i,last_cnt=tempi->ic_cnt;
  IntermediateCode *tempil1;
  U1 *body;
  BoolI1 copy_from_rbx,
         old_lock=Btr(&tempi->ic_flags,ICf_LOCK);
  switch (t1>>4) {
    case TY_REG>>4:
      if (t2&TY_MASK==TY_IMM) {
        if (!d2)
          IcZero(tempi,r1);
        else if (0<=d2<=MAX_U1) {
          IcZero(tempi,r1);
          if (r1>7)
            IcU3(tempi,d2<<16+(0xB0+r1&7)<<8+0x41);
          else if (r1>3)
            IcU3(tempi,d2<<16+(0xB0+r1)<<8+0x40);
          else
            IcU2(tempi,d2<<8+0xB0+r1);
        } else if (MIN_I1<=d2<=MAX_I1) {
          if (r1>7) {
            r1&=7;
            IcU3(tempi,d2<<16+(0xB0+r1)<<8+0x41);
            IcU4(tempi,0xC0BE0F4D+r1<<24+r1<<27);
          } else {
            if (r1>3)
              IcU3(tempi,d2<<16+(0xB0+r1)<<8+0x40);
            else
              IcU2(tempi,d2<<8+0xB0+r1);
            IcU4(tempi,0xC0BE0F48+r1<<24+r1<<27);
          }
        } else if (MIN_I4<=d2<=MAX_I4) {
          if (r1>7) {
            r1&=7;
            IcU2(tempi,(0xB8+r1)<<8+0x41);
            IcU4(tempi,d2);
            IcU3(tempi,0xC0634D+r1<<16+r1<<19);
          } else {
            IcU1(tempi,0xB8+r1);
            IcU4(tempi,d2);
            IcU3(tempi,0xC06348+r1<<16+r1<<19);
          }
        } else {
          i=0xB848;
          if (r1>7) {
            i++;
            r1&=7;
          }
          IcU2(tempi,i+r1<<8);
          IcU8(tempi,d2);
        }
      } else if (t2&TY_MASK==TY_STK)
        IcPop(tempi,t1,r1,d1,ip);
      else {
        if (r1==r2 && t2&TY_MASK==TY_REG)
          goto move_done;
        if (t2&TY_MASK==TY_REG) {
          t2=TY_REG+IT_I8;
          if (IcLastIns(tempi,&tempil1,&body)) {
            i=tempil1->ic_cnt;
            if (tempil1->ic_last_cnt==i-3 &&
                body[i-2]==0x8B &&
                body[i-1]==0xC0+r1&7+(r2&7)<<3 &&
                body[i-3]==0x48+ (r2>7 ? 4:0)+(r1>7) ) {
              goto move_done;
            }
          }
        }
        i=IcModr(r1,t2,r2,d2);
        if (t2&IT_MASK==IT_U4) {
          if (r1!=r2) {
            IcZero(tempi,r1);
            copy_from_rbx=FALSE;
          } else {
            if (r2!=CREG_RBX)
              IcZero(tempi,CREG_RBX);
            else
              throw(EXCEPT_COMPILER,19); //internal error, fatal, out of registers
            copy_from_rbx=TRUE;
            i=IcModr(CREG_RBX,t2,r2,d2);
          }
        } else
          i|=0x4800;
        IcRex(tempi,i.u1[1]);
        switch (t2&IT_MASK) {
          case IT_I1:
            IcU3(tempi,i.u1[2]<<16+0xBE0F);
            break;
          case IT_I2:
            IcU3(tempi,i.u1[2]<<16+0xBF0F);
            break;
          case IT_I4:
            IcU2(tempi,i.u1[2]<<8+0x63);
            break;

          case IT_U1:
            IcU3(tempi,i.u1[2]<<16+0xB60F);
            break;
          case IT_U2:
            IcU3(tempi,i.u1[2]<<16+0xB70F);
            break;
          case IT_U4:
            IcU2(tempi,i.u1[2]<<8+0x8B);
            if (t2&TY_MASK==TY_SIB)
              IcU1(tempi,i.u1[3]);
            if (i.u1[0]==1)
              IcU1(tempi,d2);
            else if (i.u1[0]==2)
              IcU4(tempi,d2);
            else if (i.u1[0]==4) {
              IcU4(tempi,d2-(ip+tempi->ic_cnt+4));
              tempi->ic_flags&=~ICF_CODE_FINAL;
            }
            if (copy_from_rbx) {
              if (r1>7)
                IcU3(tempi,0xC38B4C+(r1&7)<<19);
              else
                IcU3(tempi,0xC38B48+r1<<19);
            }
            goto move_done1;
          default:
            IcU2(tempi,i.u1[2]<<8+0x8B);
        }
        if (t2&TY_MASK==TY_SIB)
          IcU1(tempi,i.u1[3]);
        if (i.u1[0]==1)
          IcU1(tempi,d2);
        else if (i.u1[0]==2)
          IcU4(tempi,d2);
        else if (i.u1[0]==4) {
          IcU4(tempi,d2-(ip+tempi->ic_cnt+4));
          tempi->ic_flags&=~ICF_CODE_FINAL;
        }
move_done1:
      }
      break;
    case TY_STK>>4:
      if (t1&IT_MASK<t2&IT_MASK)
        IcPush(tempi,t2&TY_MASK+t1&IT_MASK,r2,d2,ip);
      else
        IcPush(tempi,t2,r2,d2,ip);
      break;
    case TY_DISP>>4:
    case TY_RIP_DISP32>>4:
    case TY_SIB>>4:
      if (t2&TY_MASK==TY_IMM &&
          (t1&IT_MASK<IT_I8 || (MIN_I4<=d2<=MAX_I4))) {
        i=IcModr(0,t1,r1,d1);
        switch (t1&IT_MASK) {
          case IT_I1:
          case IT_U1:
            IcRex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0xC6);
            break;
          case IT_U2:
          case IT_I2:
            Ic66Rex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0xC7);
            break;
          default:
            IcRex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0xC7);
        }
        if (t1&TY_MASK==TY_SIB)
          IcU1(tempi,i.u1[3]);
        if (i.u1[0]==1)
          IcU1(tempi,d1);
        else if (i.u1[0]==2)
          IcU4(tempi,d1);
        else if (i.u1[0]==4) {
          i=d1-(ip+tempi->ic_cnt+4);
          switch (t1&IT_MASK) {
            case IT_I1:
            case IT_U1:
              i--;
              break;
            case IT_I2:
            case IT_U2:
              i-=2;
              break;
            default:
              i-=4;
          }
          IcU4(tempi,i);
          tempi->ic_flags&=~ICF_CODE_FINAL;
        }
        switch (t1&IT_MASK) {
          case IT_I1:
          case IT_U1:
            IcU1(tempi,d2);
            break;
          case IT_I2:
          case IT_U2:
            IcU2(tempi,d2);
            break;
          default:
            IcU4(tempi,d2);
        }
      } else {
        if (t2&TY_MASK==TY_REG)
          IcSlashOp(tempi,t1,r1,d1,r2+SLASH_OP_MOV,ip);
        else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RBX,0,t2,r2,d2,ip);
          IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,CREG_RBX,0,ip);
        }
      }
      break;
  }
move_done:
  BEqu(&tempi->ic_flags,ICf_LOCK,old_lock);
  if (tempi->ic_cnt>last_cnt>tempi->ic_last_cnt)
    tempi->ic_last_cnt=last_cnt;
}

U0 IcLea(IntermediateCode *tempi,U8 t1,U8 r1,I8 d1,U8 t2,U8 r2,I8 d2,LexStruct *lx,U1 *buf,I8 ip)
{
  I8 i;
  ExeAbsoluteAddressStruct *tempa;
  switch (t1>>4) {
    case TY_REG>>4:
      i=IcModr(r1,t2,r2,d2);
      i.u1[1]|=0x48;
      IcU3(tempi,i.u1[2]<<16+0x8D00+i.u1[1]);
      if (t2&TY_MASK==TY_SIB)
        IcU1(tempi,i.u1[3]);
      if (i.u1[0]==1)
        IcU1(tempi,d2);
      else if (i.u1[0]==2)
        IcU4(tempi,d2);
      else if (i.u1[0]==4) {
        IcU4(tempi,d2-(ip+tempi->ic_cnt+4));
        tempi->ic_flags&=~ICF_CODE_FINAL;
      }
      break;
    case TY_STK>>4:
      if (t2&TY_MASK==TY_RIP_DISP32) {
        IcU1(tempi,0x68);
        IcU4(tempi,d2);
        if (lx->flags&LF_STATIC_COMPILE && buf && !(lx->flags&LF_NO_ABSOLUTES)) {
          tempa=CAlloc(sizeof(ExeAbsoluteAddressStruct));
          tempa->next=lx->a->absolutes;
          tempa->type=EAA_ADD_U4;
          lx->a->absolutes=tempa;
          tempa->ip=ip+tempi->ic_cnt-4;
        }
        tempi->ic_flags&=~ICF_CODE_FINAL;
      } else {
        IcLea(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,lx,buf,ip);
        IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,CREG_RCX,0,ip);
      }
      break;
    default:
      IcLea(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,lx,buf,ip);
      IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,CREG_RCX,0,ip);
  }
}

U0 IcDeref(IntermediateCode *tempi,I8 ip)
{
  I8 ty;
  ty=tempi->r.type&IT_MASK;
  if (ty>tempi->p1.ptype)
    ty=tempi->p1.ptype;

  if (tempi->p1.type&TY_MASK==TY_REG)
    IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
          TY_DISP+ty,tempi->p1.reg,tempi->p1.disp,ip);
  else {
    IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,
      tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip);
    IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
          TY_DISP+ty,CREG_RCX,0,ip);
  }
}

U0 IcAddEct(IntermediateCode *tempi,U8 t1,U8 r1,I8 d1,
           U8 t2,U8 r2,I8 d2,U8 t3,U8 r3,I8 d3,U8 op,I8 ip)
{
  I8 i,temp,result_reg=CREG_RAX;
  BoolI1 swap=FALSE;
  if (r3!=result_reg) {
    swap^=TRUE;
    SwapU8(&t2,&t3);
    SwapU8(&r2,&r3);
    SwapU8(&d2,&d3);
  }
  if (t2&IT_MASK>=IT_I8 && r2!=result_reg &&
      TY_REG<=t2&TY_MASK<=TY_RIP_DISP32) {
    if (t1&TY_MASK==TY_REG && !(r2==r1 && TY_REG<=t2&TY_MASK<=TY_SIB))
      result_reg=r1;
    IcMov(tempi,TY_REG+IT_I8,result_reg,0,t3,r3,d3,ip);
    i=IcModr(result_reg,t2,r2,d2);
    if (tempi->ic_flags&ICF_LOCK)
      IcU1(tempi,0xF0);
    IcRex(tempi,i.u1[1]);
    IcU2(tempi,i.u1[2]<<8+op);
    if (t2&TY_MASK==TY_SIB)
      IcU1(tempi,i.u1[3]);
    if (i.u1[0]==1)
      IcU1(tempi,d2);
    else if (i.u1[0]==2)
      IcU4(tempi,d2);
    else if (i.u1[0]==4) {
      IcU4(tempi,d2-(ip+tempi->ic_cnt+4));
      tempi->ic_flags&=~ICF_CODE_FINAL;
    }
  } else {
    if (t2&TY_MASK==TY_REG)
      temp=r2;
    else
      temp=CREG_RCX;

    if (t1&TY_MASK==TY_REG)
      result_reg=r1;

    if (temp==result_reg)
      result_reg=CREG_RDX;
    if (swap ^^ tempi->ic_flags & ICF_P1_FIRST) {
      if (r3==temp && TY_REG<=t3&TY_MASK<=TY_SIB)
        temp=CREG_RCX;
      IcMov(tempi,TY_REG+IT_I8,temp,0,t2,r2,d2,ip);
      IcMov(tempi,TY_REG+IT_I8,result_reg,0,t3,r3,d3,ip);
    } else {
      if (r2==result_reg && TY_REG<=t2&TY_MASK<=TY_SIB)
        result_reg=CREG_RDX;
      IcMov(tempi,TY_REG+IT_I8,result_reg,0,t3,r3,d3,ip);
      IcMov(tempi,TY_REG+IT_I8,temp,0,t2,r2,d2,ip);
    }
    i=0x48;
    if (result_reg>7)
      i+=4;
    if (temp>7)
      i++;
    if (tempi->ic_flags&ICF_LOCK)
      IcU1(tempi,0xF0);
    IcU3(tempi,0xC00000+i+(temp&7)<<16+(result_reg&7)<<19+op<<8);
  }
  IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,result_reg,0,ip);
}

U0 IcAddConst(IntermediateCode *tempi,U8 t1,U8 r1,I8 d1,
           U8 t2,U8 r2,I8 d2,I8 disp,U8 op,I8 ip)
{
  I8 i,r;
  if (op.u1[0]==0x2B) {
    op=0x0003;
    disp=-disp;
  }
  if (t1&TY_MASK==TY_REG) {
    if (t2&TY_MASK!=TY_REG) {
      IcMov(tempi,t1,r1,d1,t2,r2,d2,ip);
      t2=t1;
      r2=r1;
      d2=d1;
    }
    if (r1==r2) {
      if (r1>7)
        i=0x49;
      else
        i=0x48;
      if (!disp &&
        (op.u1[0]==0x03||op.u1[0]==0x2B||
         op.u1[0]==0x33||op.u1[0]==0x0B))
        return;
      else if (disp==1 && op.u1[0]==0x03) {
        IcU3(tempi,0xC0FF00+op.u1[1]<<19+i+(r1&7)<<16);
        return;
      } else if (disp==-1 && op.u1[0]==0x03) {
        IcU3(tempi,0xC8FF00+i+(r1&7)<<16);
        return;
      } else if (MIN_I1<=disp<=MAX_I1) {
        IcU3(tempi,0xC08300+op.u1[1]<<19+i+(r1&7)<<16);
        IcU1(tempi,disp);
        return;
      } else if (MIN_I4<=disp<=MAX_I4) {
        IcU3(tempi,0xC08100+op.u1[1]<<19+i+(r1&7)<<16);
        IcU4(tempi,disp);
        return;
      }
    }
    if (op.u1[0]==0x03 && MIN_I4<=disp<=MAX_I4 &&
      !Bt(&non_ptr_vars_mask,r2)) {
      i=IcModr(r1,TY_DISP+IT_I8,r2,disp);
      i.u1[1]|=0x48;
      IcU3(tempi,i.u1[2]<<16+0x8D00+i.u1[1]);
      if (i.u1[0]==1)
        IcU1(tempi,disp);
      else if (i.u1[0]==2)
        IcU4(tempi,disp);
      else if (i.u1[0]==4) {
        IcU4(tempi,disp-(ip+tempi->ic_cnt+4));
        tempi->ic_flags&=~ICF_CODE_FINAL;
      }
      return;
    }
  }
  switch (t1>>4) {
    case TY_REG>>4:
    case TY_DISP>>4:
    case TY_SIB>>4:
    case TY_RIP_DISP32>>4:
      if (t1!=t2 || r1!=r2 || d1!=d2) {
        IcMov(tempi,t1,r1,d1,t2,r2,d2,ip);
        t2=t1;
        r2=r1;
        d2=d1;
      }

      if (!disp &&
        (op.u1[0]==0x03||op.u1[0]==0x2B||
         op.u1[0]==0x33||op.u1[0]==0x0B))
        return;
      r=op.u1[1]; //not a register but a slash value
      if (op.u1[0]==0x03 && disp==-1) //add -1
        r=1; //Decrement slash value

      if (op.u1[0]==0x03 && (disp==1 || disp==-1)) {  //Add
        i=IcModr(r,t1,r1,d1);
        if (t1!=TY_REG && tempi->ic_flags&ICF_LOCK)
          IcU1(tempi,0xF0);
        switch (t1&IT_MASK) {
          case IT_I1:
          case IT_U1:
            IcRex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0xFE);
            break;
          case IT_I2:
          case IT_U2:
            Ic66Rex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0xFF);
            break;
          default:
            IcRex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0xFF);
        }
        if (t1&TY_MASK==TY_SIB)
          IcU1(tempi,i.u1[3]);
        if (i.u1[0]==1)
          IcU1(tempi,d1);
        else if (i.u1[0]==2)
          IcU4(tempi,d1);
        else if (i.u1[0]==4) {
          IcU4(tempi,d1-(ip+tempi->ic_cnt+4));
          tempi->ic_flags&=~ICF_CODE_FINAL;
        }
        return;
      }
      if (MIN_I1<=disp<=MAX_I1 ||
          t1&(IT_MASK-1)==IT_I1) {
        i=IcModr(r,t1,r1,d1);
        if (tempi->ic_flags&ICF_LOCK)
          IcU1(tempi,0xF0);
        switch (t1&IT_MASK) {
          case IT_I1:
          case IT_U1:
            IcRex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0x80);
            break;
          case IT_I2:
          case IT_U2:
            Ic66Rex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0x83);
            break;
          default:
            IcRex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0x83);
        }
        if (t1&TY_MASK==TY_SIB)
          IcU1(tempi,i.u1[3]);
        if (i.u1[0]==1)
          IcU1(tempi,d1);
        else if (i.u1[0]==2)
          IcU4(tempi,d1);
        else if (i.u1[0]==4) {
          IcU4(tempi,d1-(ip+tempi->ic_cnt+5));
          tempi->ic_flags&=~ICF_CODE_FINAL;
        }
        IcU1(tempi,disp);
        return;
      }
      if (MIN_I4<=disp<=MAX_I4 ||
          t1&IT_MASK<IT_I8) {
        i=IcModr(r,t1,r1,d1);
        if (tempi->ic_flags&ICF_LOCK)
          IcU1(tempi,0xF0);
        switch (t1&IT_MASK) {
          case IT_I2:
          case IT_U2:
            Ic66Rex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0x81);
            break;
          default:
            IcRex(tempi,i.u1[1]);
            IcU2(tempi,i.u1[2]<<8+0x81);
        }
        if (t1&TY_MASK==TY_SIB)
          IcU1(tempi,i.u1[3]);
        if (i.u1[0]==1)
          IcU1(tempi,d1);
        else if (i.u1[0]==2)
          IcU4(tempi,d1);
        else if (i.u1[0]==4) {
          if (t1&(IT_MASK-1)==IT_I2)
            IcU4(tempi,d1-(ip+tempi->ic_cnt+6));
          else
            IcU4(tempi,d1-(ip+tempi->ic_cnt+8));
          tempi->ic_flags&=~ICF_CODE_FINAL;
        }
        if (t1&(IT_MASK-1)==IT_I2)
          IcU2(tempi,disp);
        else
          IcU4(tempi,disp);
        return;
      }
      break;
    case TY_STK>>4:
      IcAddConst(tempi,TY_REG+IT_I8,CREG_RAX,0,t2,r2,d2,disp,op,ip);
      IcPushRegs(tempi,1<<CREG_RAX);
      return;
  }
  IcAddEct(tempi,t1,r1,d1,TY_IMM+IT_I8,0,disp,t2,r2,d2,op.u1[0],ip);
}

U0 IcSub(IntermediateCode *tempi,U8 t1,U8 r1,I8 d1,
           U8 t2,U8 r2,I8 d2,U8 t3,U8 r3,I8 d3,I8 ip)
{
  I8 i=0x48,op=0x2B;
  BoolI1 swap=FALSE;
  if (r3!=CREG_RAX) {
    swap=TRUE;
    SwapU8(&t2,&t3);
    SwapU8(&r2,&r3);
    SwapU8(&d2,&d3);
  }
  if (t2&IT_MASK>=IT_I8 && r2.u1[0]!=CREG_RAX && (t2&TY_MASK!=TY_SIB || r2.u1[1]&15!=CREG_RAX) &&
      TY_REG<=t2&TY_MASK<=TY_RIP_DISP32) {
    IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
    if (!swap) {
      op=0x03;
      IcU3(tempi,0xD8F748);
    }
    i=IcModr(CREG_RAX,t2,r2,d2);
    if (tempi->ic_flags&ICF_LOCK)
      IcU1(tempi,0xF0);
    IcRex(tempi,i.u1[1]);
    IcU2(tempi,i.u1[2]<<8+op);
    if (t2&TY_MASK==TY_SIB)
      IcU1(tempi,i.u1[3]);
    if (i.u1[0]==1)
      IcU1(tempi,d2);
    else if (i.u1[0]==2)
      IcU4(tempi,d2);
    else if (i.u1[0]==4) {
      IcU4(tempi,d2-(ip+tempi->ic_cnt+4));
      tempi->ic_flags&=~ICF_CODE_FINAL;
    }
    IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,CREG_RAX,0,ip);
  } else {
    if (tempi->ic_flags & ICF_P1_FIRST) {
      if (t3&TY_MASK!=TY_REG || t3&IT_MASK<IT_I8) {
        if (swap) {
          swap=FALSE;
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
          r3=CREG_RCX;
          r2=CREG_RAX;
        } else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
          r2=CREG_RCX;
          r3=CREG_RAX;
        }
      } else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
        r2=CREG_RCX;
      }
    } else {
      if (t3&TY_MASK!=TY_REG || t3&IT_MASK<IT_I8) {
        if (swap) {
          swap=FALSE;
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
          r2=CREG_RAX;
          r3=CREG_RCX;
        } else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
          r3=CREG_RAX;
          r2=CREG_RCX;
        }
      } else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
        r2=CREG_RCX;
      }
    }
    if (swap) {
      op=0x03;
      IcU3(tempi,0xD9F748);
    }
    if (r3>7)
      i++;
    if (r2>7)
      i+=4;
    if (tempi->ic_flags&ICF_LOCK)
      IcU1(tempi,0xF0);
    IcU3(tempi,0xC00000+i+(r3&7)<<16+(r2&7)<<19+op<<8);
    IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,r2,0,ip);
  }
}

  if (to_int) {
    op1=SLASH_OP_FLD;
    op2=SLASH_OP_FISTTP;
  } else {
    op1=SLASH_OP_FILD;
    op2=SLASH_OP_FSTP;
  }

  CmpSetFloatOpPushPop(lx,tempi,&dont_push_float,&dont_pop_float);
  if (!dont_push_float) {

        (t2&TY_MASK==TY_DISP ||

        rsp_size=8;
        IcAddRSP(tempi,-8);
      }

    IcSlashOp(tempi,t2,r2,d2,op1,ip);

    if (!dont_pop_float) {
      rsp_size=8;
      IcAddRSP(tempi,-8);
    }
  }
  if (to_int)
    CmpNoteFloatOp(lx,tempi,TRUE,FALSE,pos);
  else
    CmpNoteFloatOp(lx,tempi,FALSE,TRUE,pos);
  if (dont_pop_float) {
    if (rsp_size)
      IcAddRSP(tempi,rsp_size);
  } else {

            t1=TY_SIB+IT_I8; r1=CREG_RSP+CREG_RSP<<8; d1=0;
            rsp_size+=8;
          }

            rsp_size+=8;

              d1+=8;
          }
        }
      } else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,
            p1->type,p1->reg,p1->disp,ip);
        IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
            p2->type,p2->reg,p2->disp,ip);
        IcU2(tempi,0x5052);     //PUSH EDX PUSH EAX
        rsp_size=16;
        t1=TY_SIB+IT_I8; r1=CREG_RSP+CREG_RSP<<8; d1=8;
        t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
      }
    } else {
      if (p1->type&TY_MASK!=TY_STK) {
        if (p2->type&IT_MASK>=IT_I8 &&
            (p2->type&TY_MASK==TY_DISP ||
            p2->type&TY_MASK==TY_RIP_DISP32 ||
            p2->type&TY_MASK==TY_SIB)) {
          t2=p2->type;
          r2=p2->reg;
          d2=p2->disp;
        } else {

            if (!(builtin2=IcBuiltInFloatConst(p2->disp><(F8)))) {
              t2=TY_RIP_DISP32+IT_I8;
              r2=CREG_RIP;
              d2=CbFloatConstFind(lx,p2->disp><(F8))+buf2;
            }
          } else {
            IcPush(tempi,p2->type,p2->reg,p2->disp,ip);
            t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
            rsp_size+=8;
          }
        }
        if (p1->type&IT_MASK>=IT_I8 &&
            (p1->type&TY_MASK==TY_DISP ||
            p1->type&TY_MASK==TY_RIP_DISP32 ||
            p1->type&TY_MASK==TY_SIB)) {
          t1=p1->type;
          r1=p1->reg;
          d1=p1->disp;
        } else {
          if (p1->type&TY_MASK==TY_IMM) {
            if (!(builtin1=IcBuiltInFloatConst(p1->disp><(F8)))) {
              t1=TY_RIP_DISP32+IT_I8;
              r1=CREG_RIP;
              d1=CbFloatConstFind(lx,p1->disp><(F8))+buf2;

            if (r2==CREG_RSP+CREG_RSP<<8)

      alt=!alt;
      SwapU8(&t1,&t2);

        break;
      case 6: //FDIV
        op=SLASH_OP_FDIVR;
        break;
    }
  if (builtin2) {
    IcU2(tempi,builtin2);
    IcU2(tempi,op.u2[2]);
  } else
    IcSlashOp(tempi,t2,r2,d2,op,ip);
  CmpNoteFloatOp(lx,tempi,TRUE,TRUE,CN_MAIN);
  if (dont_pop_float) {
    if (rsp_size)
      IcAddRSP(tempi,rsp_size);
  } else {
    if (rsp_size==8)
      IcSlashOp(tempi,TY_SIB+IT_I8,CREG_RSP+CREG_RSP<<8,0,SLASH_OP_FSTP,ip);
    else if (rsp_size>8) {
      IcSlashOp(tempi,TY_SIB+IT_I8,CREG_RSP+CREG_RSP<<8,rsp_size-8,SLASH_OP_FSTP,ip);
      IcAddRSP(tempi,rsp_size-8);

    IcPop(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,ip);
  }
}

      t1,r1,d1,t2,r2,d2;
  BoolI1 short_jmp=FALSE;
  CbMiscStruct *lb,*lb1;
  U1 *buf2;

  if (tempi->ic_flags&ICF_ALTERNATE_TEMPLATE) {
    p1=&tempi->p2;
    p2=&tempi->p1;
    alt=TRUE;
  } else {
    p1=&tempi->p1;
    p2=&tempi->p2;
    alt=FALSE;
  }

  if (lx->flags&LF_STATIC_COMPILE)
    buf2=lx->a->ip;
  else
    buf2=buf;

  CmpSetFloatOpPushPop(lx,tempi,&dont_push_float,&dont_pop_float);
  if (dont_push_float) {
    if (tempi->ic_flags&ICF_POP_CMP && alt) {
      t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
      rsp_size+=8;
    } else {
      if (p2->type&IT_MASK>=IT_I8 &&
          (p2->type&TY_MASK==TY_DISP ||
          p2->type&TY_MASK==TY_RIP_DISP32 ||
          p2->type&TY_MASK==TY_SIB)) {
        t2=p2->type;
        r2=p2->reg;
        d2=p2->disp;
      } else {
        if (p2->type&TY_MASK==TY_IMM) {
          if (!(builtin2=IcBuiltInFloatConst(p2->disp><(F8))) ||
            tempi->ic_flags&ICF_PUSH_CMP) {
            t2=TY_RIP_DISP32+IT_I8;
            r2=CREG_RIP;
            d2=CbFloatConstFind(lx,p2->disp><(F8))+buf2;
          }
        } else {
          IcPush(tempi,p2->type,p2->reg,p2->disp,ip2);
          t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
          rsp_size+=8;
        }
      }
    }
  } else {
    if (tempi->ic_flags & ICF_P1_FIRST^^alt) {
      if (p2->type&TY_MASK!=TY_STK) {
        if (tempi->ic_flags&ICF_POP_CMP) {
          t1=TY_SIB+IT_I8; r1=CREG_RSP+CREG_RSP<<8; d1=0;
          rsp_size+=8;
        } else {
          if (p1->type&IT_MASK>=IT_I8 &&
              (p1->type&TY_MASK==TY_DISP ||
              p1->type&TY_MASK==TY_RIP_DISP32 ||
              p1->type&TY_MASK==TY_SIB)) {
            t1=p1->type;
            r1=p1->reg;
            d1=p1->disp;
          } else {
            if (p1->type&TY_MASK==TY_IMM) {
              if (!(builtin1=IcBuiltInFloatConst(p1->disp><(F8)))) {
                t1=TY_RIP_DISP32+IT_I8;
                r1=CREG_RIP;
                d1=CbFloatConstFind(lx,p1->disp><(F8))+buf2;
              }
            } else {
              IcPush(tempi,p1->type,p1->reg,p1->disp,ip2);
              t1=TY_SIB+IT_I8; r1=CREG_RSP+CREG_RSP<<8; d1=0;
              rsp_size+=8;
            }
          }
        }
        if (p2->type&IT_MASK>=IT_I8 &&
            (p2->type&TY_MASK==TY_DISP ||
            p2->type&TY_MASK==TY_RIP_DISP32 ||
            p2->type&TY_MASK==TY_SIB)) {
          t2=p2->type;
          r2=p2->reg;
          d2=p2->disp;
        } else {
          if (p2->type&TY_MASK==TY_IMM) {
            if (!(builtin2=IcBuiltInFloatConst(p2->disp><(F8))) ||
              tempi->ic_flags&ICF_PUSH_CMP) {
              t2=TY_RIP_DISP32+IT_I8;
              r2=CREG_RIP;
              d2=CbFloatConstFind(lx,p2->disp><(F8))+buf2;
            }
          } else {
            IcPush(tempi,p2->type,p2->reg,p2->disp,ip2);
            t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
            rsp_size+=8;
            if (r1==CREG_RSP+CREG_RSP<<8)
              d1+=8;
          }
        }
      } else {
        if (tempi->ic_flags&ICF_POP_CMP)
          IcPopRegs(tempi,1<<CREG_RDX);
        else
          IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,
              p1->type,p1->reg,p1->disp,ip2);
        IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
              p2->type,p2->reg,p2->disp,ip2);
        IcU2(tempi,0x5052);     //PUSH EDX PUSH EAX
        rsp_size=16;
        t1=TY_SIB+IT_I8; r1=CREG_RSP+CREG_RSP<<8; d1=8;
        t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
      }
    } else {
      if (p1->type&TY_MASK!=TY_STK) {
        if (p2->type&IT_MASK>=IT_I8 &&
            (p2->type&TY_MASK==TY_DISP ||
            p2->type&TY_MASK==TY_RIP_DISP32 ||
            p2->type&TY_MASK==TY_SIB)) {
          t2=p2->type;
          r2=p2->reg;
          d2=p2->disp;
        } else {
          if (p2->type&TY_MASK==TY_IMM) {
            if (!(builtin2=IcBuiltInFloatConst(p2->disp><(F8))) ||
              tempi->ic_flags&ICF_PUSH_CMP) {
              t2=TY_RIP_DISP32+IT_I8;
              r2=CREG_RIP;
              d2=CbFloatConstFind(lx,p2->disp><(F8))+buf2;
            }
          } else {
            IcPush(tempi,p2->type,p2->reg,p2->disp,ip2);
            t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
            rsp_size+=8;
          }
        }
        if (tempi->ic_flags&ICF_POP_CMP) {
          t1=TY_SIB+IT_I8; r1=CREG_RSP+CREG_RSP<<8; d1=0;
          rsp_size+=8;
          if (r2==CREG_RSP+CREG_RSP<<8)
            d2+=8;
        } else {
          if (p1->type&IT_MASK>=IT_I8 &&
              (p1->type&TY_MASK==TY_DISP ||
              p1->type&TY_MASK==TY_RIP_DISP32 ||
              p1->type&TY_MASK==TY_SIB)) {
            t1=p1->type;
            r1=p1->reg;
            d1=p1->disp;
          } else {
            if (p1->type&TY_MASK==TY_IMM) {
              if (!(builtin1=IcBuiltInFloatConst(p1->disp><(F8)))) {
                t1=TY_RIP_DISP32+IT_I8;
                r1=CREG_RIP;
                d1=CbFloatConstFind(lx,p1->disp><(F8))+buf2;
              }
            } else {
              IcPush(tempi,p1->type,p1->reg,p1->disp,ip2);
              t1=TY_SIB+IT_I8; r1=CREG_RSP+CREG_RSP<<8; d1=0;
              rsp_size+=8;
              if (r2==CREG_RSP+CREG_RSP<<8)
                d2+=8;
            }
          }
        }
      } else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
            p2->type,p2->reg,p2->disp,ip2);
        if (tempi->ic_flags&ICF_POP_CMP)
          IcPopRegs(tempi,1<<CREG_RDX);
        else
          IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,
              p1->type,p1->reg,p1->disp,ip2);
        IcU2(tempi,0x5052);     //PUSH EDX PUSH EAX
        rsp_size=16;
        t1=TY_SIB+IT_I8; r1=CREG_RSP+CREG_RSP<<8; d1=8;
        t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
      }
    }
  }
  if (!dont_push_float) {
    if (builtin1)
      IcU2(tempi,builtin1);
    else
      IcSlashOp(tempi,t1,r1,d1,SLASH_OP_FLD,ip2);
  }
  if (!alt)
    us=not_us;
  if (builtin2)
    IcU2(tempi,builtin2);
  else
    IcSlashOp(tempi,t2,r2,d2,SLASH_OP_FLD,ip2);
  if (tempi->ic_flags&ICF_PUSH_CMP) {
    t2=TY_SIB+IT_I8; r2=CREG_RSP+CREG_RSP<<8; d2=0;
    if (!rsp_size) {
      rsp_size=8;
      IcAddRSP(tempi,-8);
    } else if (rsp_size==16)
      d2=8;
  }
  if (tempi->ic_flags&ICF_PUSH_CMP) {
    if (alt) {
      IcU2(tempi,0xF1DF);         //FCOMIP
      IcSlashOp(tempi,t2,r2,d2,SLASH_OP_FSTP,ip2);
    } else {
      IcU2(tempi,0xF1DB);         //FCOMI
      IcSlashOp(tempi,t2,r2,d2,SLASH_OP_FSTP,ip2);
      IcU4(tempi,0xF7D9C0DD); //FFREE,FINCSTP
    }
  } else {
    IcU2(tempi,0xF1DF);   //FCOMIP
    IcU4(tempi,0xF7D9C0DD); //FFREE,FINCSTP
  }
  CmpNoteFloatOp(lx,tempi,TRUE,FALSE,CN_MAIN);

  if (tempi->ic_flags&ICF_PUSH_CMP) {
    if (r2.u1[0]==CREG_RSP) {
      while (d2 && rsp_size) {
        IcU1(tempi,0x5B); //POP RBX (Dont change flags)
        rsp_size-=8;
        d2-=8;
      }
    } else {
      while (rsp_size) {
        IcU1(tempi,0x5B); //POP RBX (Dont change flags)
        rsp_size-=8;
      }
      IcPush(tempi,t2,r2,d2,ip2);
    }
  } else {
    while (rsp_size) {
      IcU1(tempi,0x5B); //POP RBX (Dont change flags)
      rsp_size-=8;
    }
  }

  ip+=tempi->ic_cnt;
  lb=tempi->ic_data;
  while (lb1=lb->forward)
    lb=lb1;
  if (lb->address!=INVALID_PTR) {
    i=lb->address-ip;
    if (lb->flags&CBMF_POP_CMP)
      if(tempi->ic_flags&ICF_PUSH_CMP)
        i+=4;
      else
        i+=8;
        if (MIN_I1+2<i<MAX_I1+2)
      short_jmp=TRUE;
  }
  if (buf)
    short_jmp=ToBool(tempi->ic_flags&ICF_SHORT_JMP);

  if (short_jmp) {
    tempi->ic_flags|=ICF_SHORT_JMP;
    i=lb->address-(ip+2);
    if (lb->flags&CBMF_POP_CMP)
      if(tempi->ic_flags&ICF_PUSH_CMP)
        i+=4;
      else
        i+=8;
    IcU2(tempi,i<<8+us.u1[2]);
  } else {
    tempi->ic_flags&=~ICF_SHORT_JMP;
    i=lb->address-(ip+6);
    if (lb->flags&CBMF_POP_CMP)
      if(tempi->ic_flags&ICF_PUSH_CMP)
        i+=4;
      else
        i+=8;
    IcU2(tempi,us.u2[0]);
    IcU4(tempi,i);
  }
}

U0 IcMul(IntermediateCode *tempi,I8 ip)
{
  I8 i,r2,r=CREG_RAX,j;
  IcArg *p1,*p2;
  if (tempi->p1.type&TY_MASK==TY_IMM) {
    p1=&tempi->p2;
    p2=&tempi->p1;
    tempi->ic_flags|=ICF_P1_FIRST;
  } else {
    p1=&tempi->p1;
    p2=&tempi->p2;
  }
  i=p2->disp;
  if (!(tempi->ic_class->sub_type&1) &&
      p2->type&TY_MASK==TY_IMM && MIN_I4<=i<=MAX_I4) {
    if (tempi->r.type==TY_REG+IT_I8) {
      IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,p1->type,p1->reg,p1->disp,ip);
      r=tempi->r.reg;
    } else
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,p1->type,p1->reg,p1->disp,ip);
    if (r>7)
      j=0xC0004D;
    else
      j=0xC00048;
    if (MIN_I1<=i<=MAX_I1)
      IcU4(tempi,i<<24+0x6B00+j+(r&7)<<16+(r&7)<<19);
    else {
      IcU3(tempi,0x6900+j+(r&7)<<16+(r&7)<<19);
      IcU4(tempi,i);
    }
  } else {
    if (tempi->ic_class->sub_type&1)
      i=0xE0F748;
    else
      i=0xE8F748;
    if (tempi->ic_flags & ICF_P1_FIRST) {
      IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,p1->type,p1->reg,p1->disp,ip);
      r2=CREG_RCX;
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,p2->type,p2->reg,p2->disp,ip);
    } else {
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,p2->type,p2->reg,p2->disp,ip);
      if (p1->type&TY_MASK!=TY_REG ||
          p1->type&IT_MASK<IT_I8) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,p1->type,p1->reg,p1->disp,ip);
        r2=CREG_RCX;
      } else
        r2=p1->reg;
    }
    if (r2>7) {
      i++;
      r2&=7;
    }
    IcU3(tempi,i+r2<<16);
  }
  IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,TY_REG+IT_I8,r,0,ip);
}

U0 IcAddSubEctEqual(IntermediateCode *tempi,U8 ptype,U8 t1,U8 r1,I8 d1,
           U8 t2,U8 r2,I8 d2,U8 t3,U8 r3,I8 d3,U8 op,I8 ip)
{
  BoolI1 done;
  I8 result_reg,temp,i;
  if (tempi->ic_flags & ICF_NOT_ADDRESS) {
    if (t3&TY_MASK==TY_IMM) {
      IcAddConst(tempi,
        t2,r2,d2,t2,r2,d2,d3,op,ip);
      if (t1&TY_MASK)
        IcMov(tempi,t1,r1,d1,t2,r2,d2,ip);
      return;
    } else {
      done=FALSE;
      if (ptype>=IT_I8) {
        if (!(t1&TY_MASK) &&
            TY_REG<=t2&TY_MASK<=TY_RIP_DISP32) {
          if (t3&TY_MASK==TY_REG)
            temp=r3;
          else {
            temp=CREG_RCX;
            IcMov(tempi,TY_REG+IT_I8,temp,0,t3,r3,d3,ip);
          }
          i=IcModr(temp,t2&TY_MASK+ptype,r2,d2);
          if (tempi->ic_flags&ICF_LOCK)
            IcU1(tempi,0xF0);
          IcRex(tempi,i.u1[1]);
          IcU2(tempi,i.u1[2]<<8+op.u1[5]);
          if (t2&TY_MASK==TY_SIB)
            IcU1(tempi,i.u1[3]);
          if (i.u1[0]==1)
            IcU1(tempi,d2);
          else if (i.u1[0]==2)
            IcU4(tempi,d2);
          else if (i.u1[0]==4) {
            IcU4(tempi,d2-(ip+tempi->ic_cnt+4));
            tempi->ic_flags&=~ICF_CODE_FINAL;
          }
          return;
        }
        if (t3&IT_MASK>=IT_I8 &&
            TY_REG<=t3&TY_MASK<=TY_RIP_DISP32) {
          if (t2&TY_MASK==TY_REG)
            result_reg=r2;
          else {
            result_reg=CREG_RCX;
            IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
          }
          i=IcModr(result_reg,t3&TY_MASK+ptype,r3,d3);
          if (tempi->ic_flags&ICF_LOCK)
            IcU1(tempi,0xF0);
          IcRex(tempi,i.u1[1]);
          IcU2(tempi,i.u1[2]<<8+op.u1[0]);
          if (t3&TY_MASK==TY_SIB)
            IcU1(tempi,i.u1[3]);
          if (i.u1[0]==1)
            IcU1(tempi,d3);
          else if (i.u1[0]==2)
            IcU4(tempi,d3);
          else if (i.u1[0]==4) {
            IcU4(tempi,d3-(ip+tempi->ic_cnt+4));
            tempi->ic_flags&=~ICF_CODE_FINAL;
          }
          IcMov(tempi,t2&TY_MASK+ptype,r2,d2,TY_REG+IT_I8,result_reg,0,ip);
          done=TRUE;
        }
      }
      if (!done) {
        if (tempi->ic_flags & ICF_P1_FIRST) {
          if (t2&TY_MASK==TY_REG && r2!=CREG_RAX)
            result_reg=r2;
          else {
            result_reg=CREG_RCX;
            IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2&TY_MASK+ptype,r2,d2,ip);
          }
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
        } else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
          if (t2&TY_MASK==TY_REG && r2!=CREG_RAX)
            result_reg=r2;
          else {
            result_reg=CREG_RCX;
            IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2&TY_MASK+ptype,r2,d2,ip);
          }
        }
        if (tempi->ic_flags&ICF_LOCK)
          IcU1(tempi,0xF0);
        if (result_reg>7)
          IcU1(tempi,0x4C);
        else
          IcU1(tempi,0x48);
        IcU2(tempi,0xC000+op.u1[0]+(result_reg&7)<<11);
        IcMov(tempi,t2&TY_MASK+ptype,r2,d2,TY_REG+IT_I8,result_reg,0,ip);
      }
    }
  } else {
    done=FALSE;
    if (t3&TY_MASK==TY_IMM && op.u1[2]) {
      if (!d3.u4[1]) {
        if (tempi->ic_flags&ICF_NO_DEPEND_RESULT &&
            t2&TY_MASK==TY_REG && d3><(U8)<=MAX_I1) {
          IcSlashOp(tempi,TY_DISP+ptype,r2,0,0x838000+op.u1[4],ip);
          IcU1(tempi,d3);
          done=TRUE;
        } else if (op.u1[2]==0x24) { //AND
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,TY_DISP+ptype,CREG_RCX,0,ip);
          result_reg=CREG_RAX;
          if (tempi->ic_flags&ICF_LOCK)
            IcU1(tempi,0xF0);
          IcU2(tempi,op.u1[3]<<8+0x40);
          IcU4(tempi,d3);
          IcMov(tempi,TY_DISP+ptype,CREG_RCX,0,TY_REG+IT_I8,result_reg,0,ip);
          done=TRUE;
        } else if (ptype&IT_MASK<IT_I8) { //OR/XOR
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,TY_DISP+ptype,CREG_RCX,0,ip);
          result_reg=CREG_RAX;
          if (tempi->ic_flags&ICF_LOCK)
            IcU1(tempi,0xF0);
          if (d3.u2[1]) {
            IcU2(tempi,op.u1[3]<<8+0x40);
            IcU4(tempi,d3);
          } else if (d3.u1[1]) {
            IcU3(tempi,op.u1[3]<<16+0x4066);
            IcU2(tempi,d3);
          } else {
            IcU2(tempi,op.u1[2]<<8+0x40);
            IcU1(tempi,d3);
          }
          IcMov(tempi,TY_DISP+ptype,CREG_RCX,0,TY_REG+IT_I8,result_reg,0,ip);
          done=TRUE;
        }
      }
    }
    if (!done) {
      if (tempi->ic_flags & ICF_P1_FIRST) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
        IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
      } else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,t3,r3,d3,ip);
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
      }
      IcMov(tempi,TY_REG+IT_I8,CREG_RBX,0,TY_DISP+ptype,CREG_RCX,0,ip);
      result_reg=CREG_RBX;
      if (tempi->ic_flags&ICF_LOCK)
        IcU1(tempi,0xF0);
      IcU1(tempi,0x48);
      IcU2(tempi,0xC000+op.u1[0]+(result_reg&7)<<11);
      IcMov(tempi,TY_DISP+ptype,CREG_RCX,0,TY_REG+IT_I8,result_reg,0,ip);
    }
  }
  if (t1&TY_MASK)
    IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,result_reg,0,ip);
}

    IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,TY_DISP+ptype,CREG_RDX,0,ip);
    result_reg=CREG_RAX;
    IcU1(tempi,0x48);
    if (t3&TY_MASK==TY_IMM) {
      IcU2(tempi,0x00C1+(is+result_reg&7)<<8);
      IcU1(tempi,d3);
    } else

    IcMov(tempi,TY_DISP+ptype,CREG_RDX,0,TY_REG+IT_I8,result_reg,0,ip);
  }
  if (t1&TY_MASK)
    IcMov(tempi,t1,r1,d1,TY_REG+IT_I8,result_reg,0,ip);
}

U0 IcNot(IntermediateCode *tempi,I8 ip)
{
  I8 i;

    TY_DISP<=tempi->p1.type&TY_MASK<=TY_RIP_DISP32) {

    i=IcModr(tempi,tempi->p1.type,tempi->p1.reg,tempi->p1.disp);
    IcRex(tempi,i.u1[1]);
    IcU2(tempi,i.u1[2]<<8+0xF6);

      IcU1(tempi,i.u1[3]);

      IcU1(tempi,tempi->p1.disp);
    else if (i.u1[0]==2)
      IcU4(tempi,tempi->p1.disp);
    else if (i.u1[0]==4) {
      IcU4(tempi,tempi->p1.disp-(ip+tempi->ic_cnt+5));
      tempi->ic_flags&=~ICF_CODE_FINAL;

    IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip);
    IcZero(tempi,CREG_RAX);

  IcU3(tempi,0xC0FF48);
  IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
    TY_REG+IT_U8,CREG_RAX,0,ip);
}

U0 IcCmp(IntermediateCode *tempi,U8 us,U8 is,I8 ip)
{
  I8 r1,d1,r2,i=0x48,j=tempi->p2.disp;
  if (tempi->p2.type&TY_MASK==TY_IMM && MIN_I4<=j<=MAX_I4) {
    if (!(tempi->ic_flags&(ICF_POP_CMP|ICF_PUSH_CMP)) &&
        tempi->p1.type&TY_MASK==TY_DISP && //TODO
        tempi->p1.type&IT_MASK>=IT_I8 && tempi->p1.reg!=CREG_RAX) {
      r1=tempi->p1.reg;
      d1=tempi->p1.disp;
      IcZero(tempi,CREG_RAX);
      if (r1>7)
        i++;
      if (MIN_I1<=j<=MAX_I1)
        i+=0x388300;
      else
        i+=0x388100;
      if (!d1) {
        IcU3(tempi,0x000000+i+(r1&7)<<16);
      } else if (MIN_I1<=d1<=MAX_I1) {
        IcU3(tempi,0x400000+i+(r1&7)<<16);
        IcU1(tempi,d1);
      } else {
        IcU3(tempi,0x800000+i+(r1&7)<<16);
        IcU4(tempi,d1);
      }
      if (MIN_I1<=j<=MAX_I1)
        IcU1(tempi,j);
      else
        IcU4(tempi,j);
    } else {
      if (tempi->ic_flags & ICF_POP_CMP) {
        IcPopRegs(tempi,1<<CREG_RCX);
        r1=CREG_RCX;
      } else {
        if (tempi->p1.type&TY_MASK==TY_REG &&
            tempi->p1.reg!=CREG_RAX)
          r1=tempi->p1.reg;
        else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip);
          r1=CREG_RCX;
        }
      }
      IcZero(tempi,CREG_RAX);
      if (r1>7)
        i++;
      if (MIN_I1<=j<=MAX_I1) {
        IcU3(tempi,0xF88300+i+(r1&7)<<16);
        IcU1(tempi,j);
      } else {
        IcU3(tempi,0xF88100+i+(r1&7)<<16);
        IcU4(tempi,j);
      }
    }
    if (tempi->ic_flags & ICF_PUSH_CMP)
      IcPush(tempi,TY_IMM+IT_I8,0,j,ip);
    if (tempi->ic_class->sub_type&1 || tempi->ic_flags & ICF_USE_UNSIGNED)
      is=us;
    IcU2(tempi,0x300+is);
    IcU3(tempi,0xC0FF48);
    IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
      TY_REG+IT_I8,CREG_RAX,0,ip);
  } else {
    if (tempi->ic_flags & ICF_P1_FIRST) {
      if (tempi->ic_flags & ICF_POP_CMP) {
        IcPopRegs(tempi,1<<CREG_RCX);
        r1=CREG_RCX;
      } else {
        if (tempi->p1.type&TY_MASK==TY_REG &&
            tempi->p1.reg!=CREG_RAX)
          r1=tempi->p1.reg;
        else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip);
          r1=CREG_RCX;
        }
      }
      if (tempi->p2.type&TY_MASK==TY_REG &&
          tempi->p2.reg!=CREG_RAX)
        r2=tempi->p2.reg;
      else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip);
        r2=CREG_RDX;
      }
    } else {
      if (tempi->p2.type&TY_MASK==TY_REG &&
          tempi->p2.reg!=CREG_RAX)
        r2=tempi->p2.reg;
      else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip);
        r2=CREG_RCX;
      }
      if (tempi->ic_flags & ICF_POP_CMP) {
        IcPopRegs(tempi,1<<CREG_RDX);
        r1=CREG_RDX;
      } else {
        if (tempi->p1.type&TY_MASK==TY_REG &&
            tempi->p1.reg!=CREG_RAX)
          r1=tempi->p1.reg;
        else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip);
          r1=CREG_RDX;
        }
      }
    }
    IcZero(tempi,CREG_RAX);
    if (r2>7)
      i++;
    if (r1>7)
      i+=4;
    if (tempi->ic_flags & ICF_PUSH_CMP)
      IcPushRegs(tempi,1<<r2);
    IcU3(tempi,0xC03B00+i+(r2&7)<<16+(r1&7)<<19);
    if (tempi->ic_class->sub_type&1 || tempi->ic_flags & ICF_USE_UNSIGNED)
      is=us;
    IcU2(tempi,0x300+is);
    IcU3(tempi,0xC0FF48);
    IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
      TY_REG+IT_I8,CREG_RAX,0,ip);
  }
}

U0 IcBitOps(IntermediateCode *tempi,IcArg *p1,IcArg *p2,IntermediateCode *tempi2,
              U8 op,U8 op_imm,I8 ip)
{ //TODO:not fully utilizing Modr
  BoolI1 no_depend=ToBool(tempi2->ic_flags&ICF_NO_DEPEND_RESULT);
  I8 r1,t2,r2,d2,i=0x48;
  if (tempi->ic_flags & ICF_NOT_ADDRESS) {
    t2=p2->type&TY_MASK+IT_I8; //TODO: check overflow
    r2=p2->reg;
    d2=p2->disp;
    if (tempi->ic_flags & ICF_P1_FIRST) {
      if (p1->type&TY_MASK==TY_REG &&
          (p1->reg!=CREG_RAX||no_depend))
        r1=p1->reg;
      else if (p1->type&TY_MASK!=TY_IMM || p1->disp>63) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,p1->type,p1->reg,p1->disp,ip);
        r1=CREG_RCX;
      } else
        r1=0;
      if (!(TY_REG<=t2&TY_MASK<=TY_RIP_DISP32) ||
          !(r2.u1[0]!=CREG_RAX&&(t2&TY_MASK!=TY_SIB||r2.u1[1]&15!=CREG_RAX)||no_depend)) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,t2,r2,d2,ip);
        t2=TY_REG+IT_I8;
        r2=CREG_RDX;
        d2=0;
      }
    } else {
      if (!(TY_REG<=t2&TY_MASK<=TY_RIP_DISP32) ||
          !(r2.u1[0]!=CREG_RAX&&(t2&TY_MASK!=TY_SIB||r2.u1[1]&15!=CREG_RAX)||no_depend)) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,t2,r2,d2,ip);
        t2=TY_REG+IT_I8;
        r2=CREG_RCX;
        d2=0;
      }
      if (p1->type&TY_MASK==TY_REG &&
          (p1->reg!=CREG_RAX||no_depend))
        r1=p1->reg;
      else if (p1->type&TY_MASK!=TY_IMM || p1->disp>63) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,p1->type,p1->reg,p1->disp,ip);
        r1=CREG_RDX;
      } else
        r1=0;
    }
  } else {
    t2=TY_DISP+IT_I8;
    d2=0;
    if (tempi->ic_flags & ICF_P1_FIRST) {
      if (p1->type&TY_MASK==TY_REG &&
          (p1->reg!=CREG_RAX||no_depend))
        r1=p1->reg;
      else if (p1->type&TY_MASK!=TY_IMM || p1->disp>63) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,p1->type,p1->reg,p1->disp,ip);
        r1=CREG_RCX;
      } else
        r1=0;
      if (p2->type&TY_MASK==TY_REG &&
          (p2->reg!=CREG_RAX||no_depend))
        r2=p2->reg;
      else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,p2->type,p2->reg,p2->disp,ip);
        r2=CREG_RDX;
      }
    } else {
      if (p2->type&TY_MASK==TY_REG &&
          (p2->reg!=CREG_RAX||no_depend))
        r2=p2->reg;
      else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,p2->type,p2->reg,p2->disp,ip);
        r2=CREG_RCX;
      }
      if (p1->type&TY_MASK==TY_REG &&
          (p1->reg!=CREG_RAX||no_depend))
        r1=p1->reg;
      else if (p1->type&TY_MASK!=TY_IMM || p1->disp>63) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,p1->type,p1->reg,p1->disp,ip);
        r1=CREG_RDX;
      } else
        r1=0;
    }
  }
  if (!no_depend)
    IcZero(tempi,CREG_RAX);
  if (tempi->ic_flags&ICF_LOCK) {
    if (op!=0xA30F)
      IcU1(tempi,0xF0);
  }
  if ((p1->type&TY_MASK!=TY_IMM || p1->disp<32) && t2&TY_MASK!=TY_REG)
    t2=t2&TY_MASK+IT_U4;
  i=IcModr(r1,t2,r2,d2);
  IcRex(tempi,i.u1[1]);
  if (p1->type&TY_MASK==TY_IMM && p1->disp<64) {
    IcU3(tempi,i.u1[2]<<16+op_imm);
    if (t2&TY_MASK==TY_SIB)
      IcU1(tempi,i.u1[3]);
    if (i.u1[0]==1)
      IcU1(tempi,d2);
    else if (i.u1[0]==2)
      IcU4(tempi,d2);
    else if (i.u1[0]==4) {
      IcU4(tempi,d2-(ip+tempi->ic_cnt+5));
      tempi->ic_flags&=~ICF_CODE_FINAL;
    }
    IcU1(tempi,p1->disp);
  } else {
    IcU3(tempi,i.u1[2]<<16+op);
    if (t2&TY_MASK==TY_SIB)
      IcU1(tempi,i.u1[3]);
    if (i.u1[0]==1)
      IcU1(tempi,d2);
    else if (i.u1[0]==2)
      IcU4(tempi,d2);
    else if (i.u1[0]==4) {
      IcU4(tempi,d2-(ip+tempi->ic_cnt+4));
      tempi->ic_flags&=~ICF_CODE_FINAL;
    }
  }
  if (!no_depend) {
    IcU4(tempi,0x00D08348);
    if (tempi->r.type&TY_MASK)
      IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,TY_REG+IT_I8,CREG_RAX,0,ip);
  }
}

U0 IcBEqu(IntermediateCode *tempi,IntermediateCode *tempi2,I8 ip)
{
  I8 i,cnt;

//This might be paranoid.
  if (tempi->p2.type&TY_MASK==TY_REG &&
      tempi->p2.reg==CREG_RAX)
    tempi->ic_flags|=ICF_P1_FIRST; //p2 is the p1 of Bts or Btr

  if (tempi->p1.type&TY_MASK==TY_REG)
    IcOr(tempi,tempi->p1.reg);
  else {
    if (tempi->p1.type&IT_MASK<=IT_U1 &&
      TY_DISP<=tempi->p1.type&TY_MASK<=TY_RIP_DISP32) {
      i=IcModr(tempi,tempi->p1.type,tempi->p1.reg,tempi->p1.disp);
      IcRex(tempi,i.u1[1]);
      IcU2(tempi,i.u1[2]<<8+0xF6);
      if (tempi->p1.type&TY_MASK==TY_SIB)
        IcU1(tempi,i.u1[3]);
      if (i.u1[0]==1)
        IcU1(tempi,tempi->p1.disp);
      else if (i.u1[0]==2)
        IcU4(tempi,tempi->p1.disp);
      else if (i.u1[0]==4) {
        IcU4(tempi,tempi->p1.disp-(ip+tempi->ic_cnt+5));
        tempi->ic_flags&=~ICF_CODE_FINAL;
      }
      IcU1(tempi,0xFF);
    } else {
      IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip);
      IcOr(tempi,CREG_RDX);
    }
  }
  //Perform trial op, to get byte cnt
  cnt=tempi->ic_cnt;
  IcBitOps(tempi,&tempi->p2,&tempi->p3,tempi2,
           0xAB0F,0x28BA0F,ip);
  i=tempi->ic_cnt+2-cnt;

  //back-up
  tempi->ic_cnt=cnt;
  IcU2(tempi,i<<8+0x74);
  IcBitOps(tempi,&tempi->p2,&tempi->p3,tempi2,
           0xAB0F,0x28BA0F,ip);
  cnt=tempi->ic_cnt;
  IcBitOps(tempi,&tempi->p2,&tempi->p3,tempi2,
           0xB30F,0x30BA0F,ip);
  i=tempi->ic_cnt-cnt;
  tempi->ic_cnt=cnt;
  IcU2(tempi,i<<8+0xEB);
  IcBitOps(tempi,&tempi->p2,&tempi->p3,tempi2,
           0xB30F,0x30BA0F,ip);
}

    if (!(tempi2->ic_flags&ICF_NO_DEPEND_RESULT))

    IcU4(tempi,0x1B8B4864);

    IcU2(tempi,0x0774);
    IcU3(tempi,0x6BBA0F);

    IcU3(tempi,0x73BA0F);
    IcU2(tempi,TASK_TASK_FLAGS+TASKf_PREEMPT<<8);
  }
  if (!(tempi2->ic_flags&ICF_NO_DEPEND_RESULT))
    IcU4(tempi,0x00D08348);
}

U0 IcCmpAndBranch(IntermediateCode *tempi,BoolI1 has_result,U8 ip,
   U8 us,U8 is,U8 not_us,U8 not_is,
   U1 *buf,I8 ip2)
{
  I8 r1,d1,r2,i=0x48,j,t1,t2,result_reg;
  BoolI1 short_jmp=FALSE,done;
  CbMiscStruct *lb,*lb1;
  IcArg *p1=&tempi->p1,*p2=&tempi->p2,*swap;

  j=p1->disp;
  if (p1->type&TY_MASK==TY_IMM && MIN_I4<=j<=MAX_I4) {
    swap=p1; p1=p2; p2=swap;
    us=not_us;
    is=not_is;
  }
  if (tempi->ic_class->sub_type&1 || tempi->ic_flags & ICF_USE_UNSIGNED)
    is=us;

  j=p2->disp;
  if (p2->type&TY_MASK==TY_IMM && MIN_I4<=j<=MAX_I4) {
    if (!has_result &&
        p1->type&TY_MASK==TY_DISP && //TODO, other modes
        p1->type&IT_MASK>=IT_I8) {
      r1=p1->reg;
      d1=p1->disp;
      if (r1>7) {
        i++;
        r1&=7;
      }
          if (MIN_I1<=j<=MAX_I1)
        i+=0x388300;
      else
        i+=0x388100;
      if (!d1) {
        IcU3(tempi,0x000000+i+r1<<16);
      } else if (MIN_I1<=d1<=MAX_I1) {
        IcU3(tempi,0x400000+i+r1<<16);
        IcU1(tempi,d1);
      } else {
        IcU3(tempi,0x800000+i+r1<<16);
        IcU4(tempi,d1);
      }
          if (MIN_I1<=j<=MAX_I1)
        IcU1(tempi,j);
      else
        IcU4(tempi,j);
    } else {
      if (p1->type&TY_MASK==TY_REG)
        r1=p1->reg;
      else {
        IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,p1->type,p1->reg,p1->disp,ip2);
        r1=CREG_RDX;
      }
      if (!j) {
        if (is.u1[2]==0x7C) {
          IcOr(tempi,r1);
          is=0x78880F;
        } else if (is.u1[2]==0x7D) {
          IcOr(tempi,r1);
          is=0x79890F;
        } else if (is.u1[2]==0x74 || is.u1[2]==0x75)
          IcOr(tempi,r1);
        else {
          if (r1>7)
            i++;
          IcU3(tempi,0xF88300+i+(r1&7)<<16);
          IcU1(tempi,j);
        }
      } else {
        if (r1>7)
          i++;
            if (MIN_I1<=j<=MAX_I1) {
          IcU3(tempi,0xF88300+i+(r1&7)<<16);
          IcU1(tempi,j);
        } else {
          IcU3(tempi,0xF88100+i+(r1&7)<<16);
          IcU4(tempi,j);
        }
      }
    }
    if (has_result) {
      if (!swap) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,TY_IMM+IT_I8,0,j,ip2);
        result_reg=CREG_RCX;
      } else
        result_reg=r1;
    }
  } else {
    done=FALSE;
    t1=p1->type;
    r1=p1->reg;
    r2=p2->reg;
    t2=p2->type;
    if (t2&IT_MASK>=IT_I8 && !has_result &&
        TY_DISP<=t2&TY_MASK<=TY_RIP_DISP32) {
      if (t1&TY_MASK!=TY_REG || t1&IT_MASK<IT_I8) {
        IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,p1->type,p1->reg,p1->disp,ip2);
        r1=CREG_RAX;
      }
      i=IcModr(r1,t2,r2,p2->disp);
      IcRex(tempi,i.u1[1]);
      IcU2(tempi,i.u1[2]<<8+0x3B);
      if (t2&TY_MASK==TY_SIB)
        IcU1(tempi,i.u1[3]);
      if (i.u1[0]==1)
        IcU1(tempi,p2->disp);
      else if (i.u1[0]==2)
        IcU4(tempi,p2->disp);
      else if (i.u1[0]==4) {
        IcU4(tempi,p2->disp-(ip2+tempi->ic_cnt+4));
        tempi->ic_flags&=~ICF_CODE_FINAL;
      }
      done=TRUE;
    } else if (t1&IT_MASK>=IT_I8 &&
        TY_REG<=t1&TY_MASK<=TY_RIP_DISP32) {
      if (t2&TY_MASK!=TY_REG || t2&IT_MASK<IT_I8) {
        if (t1&TY_MASK==TY_REG && r1==CREG_RAX) {
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,p2->type,p2->reg,p2->disp,ip2);
          r2=CREG_RCX;
        } else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,p2->type,p2->reg,p2->disp,ip2);
          r2=CREG_RAX;
        }
      }
      i=IcModr(r2,t1,r1,p1->disp);
      IcRex(tempi,i.u1[1]);
      IcU2(tempi,i.u1[2]<<8+0x39);
      if (t1&TY_MASK==TY_SIB)
        IcU1(tempi,i.u1[3]);
      if (i.u1[0]==1)
        IcU1(tempi,p1->disp);
      else if (i.u1[0]==2)
        IcU4(tempi,p1->disp);
      else if (i.u1[0]==4) {
        IcU4(tempi,p1->disp-(ip2+tempi->ic_cnt+4));
        tempi->ic_flags&=~ICF_CODE_FINAL;
      }
      if (has_result)
        result_reg=r2;
      done=TRUE;
    }
    if (!done) {
      if (tempi->ic_flags & ICF_P1_FIRST) {
        if (p1->type&TY_MASK==TY_REG)
          r1=p1->reg;
        else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,p1->type,p1->reg,p1->disp,ip2);
          r1=CREG_RAX;
        }
        if (p2->type&TY_MASK==TY_REG)
          r2=p2->reg;
        else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,p2->type,p2->reg,p2->disp,ip2);
          r2=CREG_RCX;
        }
      } else {
        if (p2->type&TY_MASK==TY_REG)
          r2=p2->reg;
        else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,p2->type,p2->reg,p2->disp,ip2);
          r2=CREG_RAX;
        }
        if (p1->type&TY_MASK==TY_REG)
          r1=p1->reg;
        else {
          IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,p1->type,p1->reg,p1->disp,ip2);
          r1=CREG_RCX;
        }
      }
      if (r2>7)
        i++;
      if (r1>7)
        i+=4;
      IcU3(tempi,0xC03B00+i+(r2&7)<<16+(r1&7)<<19);
      if (has_result)
        result_reg=r2;
    }
  }
  ip+=tempi->ic_cnt;
  lb=tempi->ic_data;
  while (lb1=lb->forward)
    lb=lb1;
  if (lb->address!=INVALID_PTR) {
    i=lb->address-ip;
    if (lb->flags&CBMF_POP_CMP)
      if(tempi->ic_flags&ICF_PUSH_CMP)
        i+=4;
      else
        i+=8;
        if (MIN_I1+2<i<MAX_I1+2)
      short_jmp=TRUE;
  }
  if (buf)
    short_jmp=ToBool(tempi->ic_flags&ICF_SHORT_JMP);

  if (short_jmp) {
    tempi->ic_flags|=ICF_SHORT_JMP;
    i=lb->address-(ip+2);
    if (lb->flags&CBMF_POP_CMP)
      if(tempi->ic_flags&ICF_PUSH_CMP)
        i+=4;
      else
        i+=8;
    IcU2(tempi,i<<8+is.u1[2]);
  } else {
    tempi->ic_flags&=~ICF_SHORT_JMP;
    i=lb->address-(ip+6);
    if (lb->flags&CBMF_POP_CMP)
      if(tempi->ic_flags&ICF_PUSH_CMP)
        i+=4;
      else
        i+=8;
    IcU2(tempi,is.u2[0]);
    IcU4(tempi,i);
  }
  if (has_result)
    IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,TY_REG+IT_I8,result_reg,0,ip2);
}

U0 IcTestAndBranch(IntermediateCode *tempi,U8 ip,U8 is,U1 *buf,I8 ip2)
{
  I8 i;
  BoolI1 short_jmp=FALSE;
  CbMiscStruct *lb,*lb1;

  if (tempi->p1.type&TY_MASK!=TY_REG) {
    if (tempi->p1.type&IT_MASK<=IT_U1 &&
      TY_DISP<=tempi->p1.type&TY_MASK<=TY_RIP_DISP32) {
      i=IcModr(tempi,tempi->p1.type,tempi->p1.reg,tempi->p1.disp);
      IcRex(tempi,i.u1[1]);
      IcU2(tempi,i.u1[2]<<8+0xF6);
      if (tempi->p1.type&TY_MASK==TY_SIB)
        IcU1(tempi,i.u1[3]);
      if (i.u1[0]==1)
        IcU1(tempi,tempi->p1.disp);
      else if (i.u1[0]==2)
        IcU4(tempi,tempi->p1.disp);
      else if (i.u1[0]==4) {
        IcU4(tempi,tempi->p1.disp-(ip2+tempi->ic_cnt+5));
        tempi->ic_flags&=~ICF_CODE_FINAL;
      }
      IcU1(tempi,0xFF);
    } else {
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip2);
      IcOr(tempi,CREG_RAX);
    }
  } else
    IcOr(tempi,tempi->p1.reg);
  ip+=tempi->ic_cnt;

  lb=tempi->ic_data;
  while (lb1=lb->forward)
    lb=lb1;
  if (lb->address!=INVALID_PTR) {
    i=lb->address-ip;
    if (lb->flags&CBMF_POP_CMP)
      i+=8;
    if (MIN_I1+2<i<MAX_I1+2)
      short_jmp=TRUE;
  }
  if (buf)
    short_jmp=ToBool(tempi->ic_flags&ICF_SHORT_JMP);
  if (short_jmp) {
    tempi->ic_flags|=ICF_SHORT_JMP;
    i=lb->address-(ip+2);
    if (lb->flags&CBMF_POP_CMP)
      i+=8;
    IcU2(tempi,i<<8+is.u1[2]);
  } else {
    tempi->ic_flags&=~ICF_SHORT_JMP;
    i=lb->address-(ip+6);
    if (lb->flags&CBMF_POP_CMP)
      i+=8;
    IcU2(tempi,is.u2[0]);
    IcU4(tempi,i);
  }
}

U0 IcAndBranch(IntermediateCode *tempi,U8 ip,U8 is,U1 *buf,I8 ip2)
{
  U8 i,it,t1,r1,d1,r2;
  BoolI1 short_jmp=FALSE,swap,override;
  CbMiscStruct *lb,*lb1;
  IcArg *p1,*p2;
 
  if (tempi->p1.type&TY_MASK==TY_IMM) {
    swap=TRUE;
    p1=&tempi->p2;
    p2=&tempi->p1;
  } else {
    swap=FALSE;
    p1=&tempi->p1;
    p2=&tempi->p2;
  }

  if (p2->type&TY_MASK==TY_IMM && p2->disp>MAX_U4)
    override=TRUE;
  else
    override=FALSE;

  if (p1->type&IT_MASK<p2->type&IT_MASK)
    it=p1->type&IT_MASK;
  else
    it=p2->type&IT_MASK;

  i=p2->disp;
  if (p2->type&TY_MASK==TY_IMM && i<=MAX_U4) {
    IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,p1->type,p1->reg,p1->disp,ip2);
    if (i<=MAX_U1)
      IcU2(tempi,i<<8+0xA8);
    else if (i<=MAX_U2)
      IcU4(tempi,i<<16+0xA966);
    else {
      IcU1(tempi,0xA9);
      IcU4(tempi,i);
    }
  } else {
    t1=TY_REG+it;
    d1=0;
    if ((tempi->ic_flags&ICF_P1_FIRST ^^swap)&&!override) {
      if (p1->type&TY_MASK==TY_REG) {
        r1=p1->reg;
        swap=TRUE;
      } else {
        r1=CREG_RCX;
        swap=FALSE;
      }
      if (p2->type&TY_MASK==TY_REG) {
        r2=p2->reg;
        swap=FALSE;
      } else
        r2=CREG_RDX;
      if (swap) {
        if (p1->type&TY_MASK!=TY_REG || r1!=p1->reg)
          IcMov(tempi,TY_REG+IT_I8,r1,0,p1->type,p1->reg,p1->disp,ip2);
        if (TY_REG<=p2->type&TY_MASK<=TY_RIP_DISP32) {
          t1=p2->type&TY_MASK+it;
          r2=p2->reg;
          d1=p2->disp;
        } else
          IcMov(tempi,TY_REG+IT_I8,r2,0,p2->type,p2->reg,p2->disp,ip2);
        i=IcModr(r1,t1,r2,d1);
      } else {
        if (TY_REG<=p1->type&TY_MASK<=TY_RIP_DISP32) {
          t1=p1->type&TY_MASK+it;
          r1=p1->reg;
          d1=p1->disp;
        } else
          IcMov(tempi,TY_REG+IT_I8,r1,0,p1->type,p1->reg,p1->disp,ip2);
        if (p2->type&TY_MASK!=TY_REG || r2!=p2->reg)
          IcMov(tempi,TY_REG+IT_I8,r2,0,p2->type,p2->reg,p2->disp,ip2);
        i=IcModr(r2,t1,r1,d1);
      }
    } else {
      if (p2->type&TY_MASK==TY_REG) {
        r2=p2->reg;
        swap=FALSE;
      } else {
        r2=CREG_RDX;
        swap=TRUE;
      }
      if (p1->type&TY_MASK==TY_REG) {
        r1=p1->reg;
        swap=TRUE;
      } else
        r1=CREG_RCX;
      if (override)
        swap=FALSE;
      if (swap) {
        if (TY_REG<=p2->type&TY_MASK<=TY_RIP_DISP32) {
          t1=p2->type&TY_MASK+it;
          r2=p2->reg;
          d1=p2->disp;
        } else
          IcMov(tempi,TY_REG+IT_I8,r2,0,p2->type,p2->reg,p2->disp,ip2);
        if (p1->type&TY_MASK!=TY_REG || r1!=p1->reg)
          IcMov(tempi,TY_REG+IT_I8,r1,0,p1->type,p1->reg,p1->disp,ip2);
        i=IcModr(r1,t1,r2,d1);
      } else {
        if (p2->type&TY_MASK!=TY_REG || r2!=p2->reg)
          IcMov(tempi,TY_REG+IT_I8,r2,0,p2->type,p2->reg,p2->disp,ip2);
        if (TY_REG<=p1->type&TY_MASK<=TY_RIP_DISP32) {
          t1=p1->type&TY_MASK+it;
          r1=p1->reg;
          d1=p1->disp;
        } else
          IcMov(tempi,TY_REG+IT_I8,r1,0,p1->type,p1->reg,p1->disp,ip2);
        i=IcModr(r2,t1,r1,d1);
      }
    }
    switch (it) {
      case IT_I1:
      case IT_U1:
        IcRex(tempi,i.u1[1]);
        IcU2(tempi,i.u1[2]<<8+0x84);
        break;
      case IT_U2:
      case IT_I2:
        Ic66Rex(tempi,i.u1[1]);
        IcU2(tempi,i.u1[2]<<8+0x85);
        break;
      default:
        IcRex(tempi,i.u1[1]);
        IcU2(tempi,i.u1[2]<<8+0x85);
    }
    if (t1&TY_MASK==TY_SIB)
      IcU1(tempi,i.u1[3]);
    if (i.u1[0]==1)
      IcU1(tempi,d1);
    else if (i.u1[0]==2)
      IcU4(tempi,d1);
    else if (i.u1[0]==4) {
      IcU4(tempi,d1-(ip2+tempi->ic_cnt+4));
      tempi->ic_flags&=~ICF_CODE_FINAL;
    }
  }
  ip+=tempi->ic_cnt;

  lb=tempi->ic_data;
  while (lb1=lb->forward)
    lb=lb1;
  if (lb->address!=INVALID_PTR) {
    i=lb->address-ip;
    if (lb->flags&CBMF_POP_CMP)
      i+=8;
    if (MIN_I1+2<i<MAX_I1+2)
      short_jmp=TRUE;
  }
  if (buf)
    short_jmp=ToBool(tempi->ic_flags&ICF_SHORT_JMP);
  if (short_jmp) {
    tempi->ic_flags|=ICF_SHORT_JMP;
    i=lb->address-(ip+2);
    if (lb->flags&CBMF_POP_CMP)
      i+=8;
    IcU2(tempi,i<<8+is.u1[2]);
  } else {
    tempi->ic_flags&=~ICF_SHORT_JMP;
    i=lb->address-(ip+6);
    if (lb->flags&CBMF_POP_CMP)
      i+=8;
    IcU2(tempi,is.u2[0]);
    IcU4(tempi,i);
  }
}

U0 IcAssign(IntermediateCode *tempi,I8 ip)
{
  if (tempi->ic_flags & ICF_NOT_ADDRESS) {
    IcMov(tempi,tempi->p1.type&TY_MASK+tempi->p1.ptype,tempi->p1.reg,tempi->p1.disp,
          tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip);
    if (tempi->r.type&TY_MASK) {
      if (tempi->p1.type&TY_MASK==TY_REG)
        IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
          tempi->p1.type&TY_MASK+tempi->p1.ptype,tempi->p1.reg,tempi->p1.disp,ip);
      else
        IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
          tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip);
    }
  } else {
    if (tempi->p1.type&TY_MASK==TY_REG) {
      IcMov(tempi,TY_DISP+tempi->p1.ptype,tempi->p1.reg,0,
            tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip);

        IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
          tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip);

            tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip);
      IcMov(tempi,TY_DISP+tempi->p1.ptype,CREG_RCX,0,
            tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip);
      if (tempi->r.type&TY_MASK)
        IcMov(tempi,tempi->r.type,tempi->r.reg,tempi->r.disp,
          tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip);
    }
  }
}

      } else {
        t2=p2->type;
        if (t2&TY_MASK==TY_IMM && p2->disp<64)
          r2=0;
        else
          r2=p2->reg;
        d2=p2->disp;
      }
      if (!(tempi->ic_flags&ICF_SWAP) && p1->type&TY_MASK!=TY_REG && (p1->type&TY_MASK!=TY_IMM || p1->disp>63) ||
          tempi->ic_flags&ICF_SWAP && p1->type&TY_MASK==TY_IMM && p1->disp<64||
          p1->type&TY_MASK==TY_STK) {

  } else {

      tempi->ic_flags&=~ICF_CODE_FINAL;
    }
  }

  ip+=tempi->ic_cnt;
  lb=tempi->ic_data;
  while (lb1=lb->forward)
    lb=lb1;
  if (lb->address!=INVALID_PTR) {

  }
}

U0 IcInsQue(IntermediateCode *tempi,I8 ip2,BoolI1 atomic)
{
  I8 r1,r2;
  if (atomic) {
    if (tempi->p1.type&TY_MASK==TY_STK || tempi->p2.type&TY_MASK==TY_STK) {
      PutErr;
      PutDefine("ST_ATOMIC_QUEUE");
      throw(EXCEPT_COMPILER,0);
    }
    IcU2(tempi,0xFA9C);
  }
  if (tempi->ic_flags & ICF_P1_FIRST) {
    if (tempi->p1.type==TY_REG+IT_I8 && tempi->p1.reg!=CREG_RAX)
      r1=tempi->p1.reg;
    else {
      IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,
        tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip2);
      r1=CREG_RDX;
    }
    if (tempi->p2.type==TY_REG+IT_I8)
      r2=tempi->p2.reg;
    else {
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
        tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip2);
      r2=CREG_RAX;
    }
  } else {
    if (tempi->p2.type==TY_REG+IT_I8 && tempi->p2.reg!=CREG_RDX)
      r2=tempi->p2.reg;
    else {
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
        tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip2);
      r2=CREG_RAX;
    }
    if (tempi->p1.type==TY_REG+IT_I8)
      r1=tempi->p1.reg;
    else {
      IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,
        tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip2);
      r1=CREG_RDX;
    }
  }
  IcMov(tempi,TY_REG+IT_I8,CREG_RBX,0,TY_DISP+IT_I8,r1,0,ip2);
  IcMov(tempi,TY_DISP+IT_I8,r1,0,TY_REG+IT_I8,r2,0,ip2);
  IcMov(tempi,TY_DISP+IT_I8,r2,0,TY_REG+IT_I8,CREG_RBX,0,ip2);
  IcMov(tempi,TY_DISP+IT_I8,r2,sizeof(U0 *),TY_REG+IT_I8,r1,0,ip2);
  IcMov(tempi,TY_DISP+IT_I8,CREG_RBX,sizeof(U0 *),TY_REG+IT_I8,r2,0,ip2);
  if (atomic)
    IcU1(tempi,0x9D);
}

U0 IcMemSet(IntermediateCode *tempi,I8 ip2)
{
  I8 i,cnt;
  if (tempi->p1.type&TY_MASK==TY_IMM &&
      tempi->p2.type&TY_MASK==TY_IMM && !tempi->p2.disp ) {
    if (!(cnt=tempi->p1.disp)) return;
    IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
        tempi->p3.type,tempi->p3.reg,tempi->p3.disp,ip2);

    if (!(cnt&7))
      i=3;
    else if (!(cnt&3))
      i=2;
    else if (!(cnt&1))
      i=1;
    else
      i=0;
    cnt>>=i;

    IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,
        tempi->p1.type,tempi->p1.reg,cnt,ip2);

    IcPushRegs(tempi,1<<CREG_RDI);
    IcMov(tempi,TY_REG+IT_I8,CREG_RDI,0,
        TY_REG+IT_I8,CREG_RAX,0,ip2);
    IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
        tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip2);
    switch (i) {
      case 0: IcU2(tempi,0xAAF3); break;
      case 1: IcU3(tempi,0xABF366); break;
      case 2: IcU2(tempi,0xABF3); break;
      case 3: IcU4(tempi,0xAB48F348); break;
    }
    IcPopRegs(tempi,1<<CREG_RDI);
  } else {
    if (tempi->p1.type&TY_MASK==TY_REG &&
        tempi->p1.reg==CREG_RAX) {
      IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,
          tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip2);
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
          tempi->p3.type,tempi->p3.reg,tempi->p3.disp,ip2);
      IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,
          tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip2);
    } else if (tempi->p2.type&TY_MASK==TY_REG &&
        tempi->p2.reg==CREG_RAX) {
      IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,
          tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip2);
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
          tempi->p3.type,tempi->p3.reg,tempi->p3.disp,ip2);
      IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,
          tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip2);
    } else {
      IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
          tempi->p3.type,tempi->p3.reg,tempi->p3.disp,ip2);
      IcMov(tempi,TY_REG+IT_I8,CREG_RDX,0,
          tempi->p2.type,tempi->p2.reg,tempi->p2.disp,ip2);
      IcMov(tempi,TY_REG+IT_I8,CREG_RCX,0,
          tempi->p1.type,tempi->p1.reg,tempi->p1.disp,ip2);
    }
    IcPushRegs(tempi,1<<CREG_RDI);
    IcMov(tempi,TY_REG+IT_I8,CREG_RDI,0,
        TY_REG+IT_I8,CREG_RAX,0,ip2);
    IcMov(tempi,TY_REG+IT_I8,CREG_RAX,0,
        TY_REG+IT_I8,CREG_RDX,0,ip2);
    IcU2(tempi,0xAAF3);
    IcPopRegs(tempi,1<<CREG_RDI);
  }
}
