F64 t0=tJ;

#define PHASE1  9.5
#define PHASE2  10.0
#define PHASE3  10.5
#define PHASE4  16.0
#define PHASE5  16.5

#define MAN_X   250
#define MAN_Y   30

U0 DrawIt(TaskStruct *task,GrDC *dc)
{
  nounusedwarn task;
  F64 dt=(tJ-t0)*music_tempo;
  U8 *inter;
         if (dt<PHASE1)
    GrSpritePlot3(dc,MAN_X,MAN_Y,0,<1>);
  else if (dt<PHASE2) {
    inter=GrSpriteInterpolate(<1>,<2>,(dt-PHASE1)/(PHASE2-PHASE1));
    GrSpritePlot3(dc,MAN_X,MAN_Y,0,inter);
    Free(inter);
  } else if (dt<PHASE3) {
    inter=GrSpriteInterpolate(<2>,<3>,(dt-PHASE2)/(PHASE3-PHASE2));
    GrSpritePlot3(dc,MAN_X,MAN_Y,0,inter);
    Free(inter);
  } else if (dt<PHASE4)
    GrSpritePlot3(dc,MAN_X,MAN_Y,0,<3>);
  else if (dt<PHASE5) {
    inter=GrSpriteInterpolate(<3>,<1>,(dt-PHASE4)/(PHASE5-PHASE4));
    GrSpritePlot3(dc,MAN_X-100*(dt-PHASE4),MAN_Y-5*(dt-PHASE4),0,inter);
    Free(inter);
  } else
    GrSpritePlot3(dc,MAN_X-100*(PHASE5-PHASE4),MAN_Y-5*(PHASE5-PHASE4),0,<1>);
}

U0 Song()
{
  TaskStruct *task;
  MusicSettingsRst;
  music_tempo= 2.350;
  music_stacatto_factor= 0.900;
  try {
    task=PopUpViewStr(
    "Ecclesiastes,2:24\r\n"
        "Ecclesiastes,3:13\r\n"
        );
    TaskWaitIdle(task);
    task->win_right=Fs->win_right;
    task->win_left=Fs->win_left;
    task->win_top=Fs->win_bottom+3;
    task->win_bottom=task->win_top+12;
    TaskDerivedValsUpdate(task);
    SettingsPush; //See SettingsPush
    Fs->draw_it=&DrawIt;
    Fs->task_end_cb=&SndTaskEndCB;
    PutS("$FG,GREEN$I did this melody.  The others are random.\r\n\r\n$FG$");
    while (!ScanKey) {
      t0=tJ;
      Play("M5/4W13qEAeDDCCqBEAeDD",
          "Up \0a\0bove \0He \0gen\0tly \0smiles\r\n\0as \0we \0meet \0and \0");

          "con\0quer \0trials.\r\n\0$FG,RED$You \0can \0do \0it \0lit\0tle \0one.\r\n\0With\0");
      Play("BeEEDDqC",
          "out \0work \0there \0is \0no \0fun.\r\n\0");
    }
  } catch
    CatchAll;
  SettingsPop;
}
