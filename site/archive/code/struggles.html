F64 t0=tJ;
#define CENTER_X        200

U0 DrawIt(TaskStruct *task,GrDC *dc)
{
  nounusedwarn task;
  F64 dt=(tJ-t0)*music_tempo;
  dc->flags|=DCF_SYMMETRY;
  GrSetSymmetry(dc,CENTER_X,0,CENTER_X,1);
  GrSpritePlot3(dc,CENTER_X-40,75+dt,0,<1>);
}
