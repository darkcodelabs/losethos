asm {
#include_text "/LT/OSMain/Disk/ATADef.HPZ"
#include_text "/LT/OSMain/BootDefs.HPZ"

#define MODULE_SIZE             1*BLK_SIZE
#define BOOT_HIGH_LOC   ((BOOT_RAM_LIMIT-(BOOT_STK_SIZE+MODULE_SIZE))>>4)
USE16
BHDE_START::
        DU1     0x5A DUP(0);
BHDE_CODE::
        CLI
        CLD

        MOV     AX,VGA_TEXT_MEM_BASE/16
        MOV     FS,AX
        MOV     U4 FS:[4],0
        MOV     U4 FS:[0],'B'+0x2000+('3'+0x2000)<<16

        MOV     AX,BOOT_HIGH_LOC
        MOV     SS,AX
        MOV     ES,AX
        MOV     SP,BOOT_STK_SIZE+MODULE_SIZE
        CALL    GET_IP
GET_IP:
        POP     BX
        SUB     BX,GET_IP
        SHR     BX,4
//Copy this bootloader code to just under 0x80000
        MOV     AX,CS
        ADD     AX,BX
        MOV     DS,AX
        MOV     CX,MODULE_SIZE
        XOR     SI,SI
        XOR     DI,DI
        REP_MOVSB

        MOV     AX,BOOT_HIGH_LOC
        MOV     DS,AX

        DU1     0xEA;
        DU2     BHDE_HISTART,BOOT_HIGH_LOC;

BHDE_WAIT_BUSY:
        MOV     DX,U2 [BHDE_BASE0]
        ADD     DX,ATAR0_STATUS
        IN      AL,DX
        AND     AL,S_BUSY
        JNZ     BHDE_WAIT_BUSY
        RET

BHDE_ATA_CMD:
        PUSH    AX
        MOV     DX,U2 [BHDE_BASE0]
        ADD     DX,ATAR0_FEATURE
        XOR     AL,AL
        OUT     DX,AL
        POP     AX
        ADD     DX,ATAR0_CMD-ATAR0_FEATURE
        OUT     DX,AL
        RET

BHDE_BASE0::    DU2     0x1F0;
BHDE_BASE1::    DU2     0x3F4;
BHDE_UNIT::     DU1     0xE0;
BHDE_BLK::      DU8     0;

BHDE_HISTART:
//Init Drive
        MOV     DX,U2 [BHDE_BASE0]
        ADD     DX,ATAR0_SELECT
        MOV     AL,U1 [BHDE_UNIT]
        OUT     DX,AL
        MOV     AL,ATA_DEV_RESET
        CALL    BHDE_ATA_CMD
        CALL    BHDE_WAIT_BUSY

        MOV     AL,ATA_READ_NATIVE_MAX_EXT
        CALL    BHDE_ATA_CMD
        CALL    BHDE_WAIT_BUSY

        MOV     AL,ATA_SET_MAX_EXT
        CALL    BHDE_ATA_CMD
        CALL    BHDE_WAIT_BUSY

BHDE_RESTART::
        MOV     CX,512         //Gets patched
        MOV     AX,BOOT_RAM_BASE/16
        MOV     ES,AX

@@1:    MOV     DX,U2 [BHDE_BASE0]
        ADD     DX,ATAR0_STATUS
@@11:   IN      AL,DX
        AND     AL,S_DRQ
        JNZ     @@11

        XOR     DI,DI
        PUSH    CX

//READ BLK
        MOV     DX,U2 [BHDE_BASE1]
        ADD     DX,ATAR1_CONTROL
        MOV     AL,8
        OUT     DX,AL

        MOV     DX,U2 [BHDE_BASE0]

        ADD     DX,ATAR0_NSECTOR
        MOV     AL,0    //Blk Cnt
        OUT     DX,AL

        ADD     DX,ATAR0_SECTOR-ATAR0_NSECTOR
        MOV     AL,U1 [BHDE_BLK+3]
        OUT     DX,AL

        ADD     DX,ATAR0_LCYL-ATAR0_SECTOR
        MOV     AL,U1 [BHDE_BLK+4]
        OUT     DX,AL

        ADD     DX,ATAR0_HCYL-ATAR0_LCYL
        MOV     AL,U1 [BHDE_BLK+5]
        OUT     DX,AL

        ADD     DX,ATAR0_NSECTOR-ATAR0_HCYL
        MOV     AL,1    //Blk Cnt
        OUT     DX,AL

        ADD     DX,ATAR0_SECTOR-ATAR0_NSECTOR
        MOV     AL,U1 [BHDE_BLK]
        OUT     DX,AL

        ADD     DX,ATAR0_LCYL-ATAR0_SECTOR
        MOV     AL,U1 [BHDE_BLK+1]
        OUT     DX,AL

        ADD     DX,ATAR0_HCYL-ATAR0_LCYL
        MOV     AL,U1 [BHDE_BLK+2]
        OUT     DX,AL

        ADD     DX,ATAR0_SELECT-ATAR0_HCYL
        MOV     AL,U1 [BHDE_UNIT]
        OUT     DX,AL

        MOV     AL,ATA_MULTREAD_EXT
        CALL    BHDE_ATA_CMD

        MOV     DX,U2 [BHDE_BASE0]
        ADD     DX,ATAR0_STATUS
@@10:   IN      AL,DX
        AND     AL,S_DRQ
        JZ      @@10
        MOV     CX,BLK_SIZE/4
        ADD     DX,ATAR0_DATA-ATAR0_STATUS
        REP_INSD
        CALL    BHDE_WAIT_BUSY

        POP     CX
        INC     U4 [BHDE_BLK]
        JNZ     @@20
        INC     U4 [BHDE_BLK+4]
@@20:   MOV     AX,ES
        ADD     AX,BLK_SIZE/16
        MOV     ES,AX
        DEC     CX
        JNZ     U2 @@1

        MOV     EBX,BOOT_CODE_HARDDRIVE //See sys_boot_code

        DU1     0xEA;
        DU2     0,BOOT_RAM_BASE/16;
//Continues here ::/LT/OSMain/OSStartUp.CPZ
BHDE_END::
#assert $<MODULE_SIZE-2
        END
}
