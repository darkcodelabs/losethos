U0 MousePktRead()
{
  U1 j;
  if (GetTimeStamp>mouse_evt_time+time_stamp_freq>>3)
    U1FifoFlush(mouse_fifo);
  mouse_evt_time=GetTimeStamp;
  LowPower(FALSE);
  if (!U1FifoCnt(mouse_fifo))
    U1FifoInsert(mouse_fifo,0x0C);
  U1FifoInsert(mouse_fifo,InU1(KEYB_PORT));
  if (U1FifoCnt(mouse_fifo)==mouse_pkt_size+1) {
    while (U1FifoRem(mouse_fifo,&j))
      U1FifoInsert(kbdmouse_fifo,j);
  }
}

interrupt U0 MouseIntHandler()
{
  Cld;
  OutU1(0xA0,0x20);
  OutU1(0x20,0x20);
  mouse_irqs_working=TRUE;
  if (attempting_mouse_install || !mouse_installed) {
    reset_kbdmouse=TRUE;
    return;
  }
  MousePktRead;
}

U0 MouseGetDeviceType()
{
  U1 b;
  KbdMouseCmdAck(0xF2);
  b=KbdCmdRead;
  if (b==3)
    mouse_has_wheel=TRUE;
  else if (b==4)
    mouse_has_ext_buttons=TRUE;
}

BoolI8 MouseReset()
{
  U1 *bb;
  U2 *ww;
  U1 b;
  I8 i;
  BoolI1 result=FALSE,timeout,present;

  mouse_has_wheel=FALSE;
  mouse_has_ext_buttons=FALSE;

//Somehow the BIOS Extended Data Area
//enables my mouse!
  ww=0x40E;
  present=MemMarkPresent(ww,TRUE);
  if (*ww==0x9FC0) {
    bb=0x9FC00+0x30;
    *bb=1; //This enables my mouse
    //Must be some kind of config
    //register!!        No clue.  Legacy PS/2
    //must use it, somehow.
  }
  if (!present)
    MemMarkPresent(ww,FALSE);

  try {
    KbdCmdFlush;
    KbdCmdSend(KEYB_CTRL,0xAD); //Disable Kbd
    KbdCmdSend(KEYB_CTRL,0xA8); //Enable Mouse

    KbdMouseCmdAck(0xFF); //Reset
    for (i=0;i<250;i++) {
      try {
        KbdCmdRead;
        timeout=FALSE;
      } catch {
        Fs->catch_except=TRUE;
        timeout=TRUE;
      }
      if (!timeout) break;
    }
    try
      KbdCmdRead;
    catch
      Fs->catch_except=TRUE;

    KbdMouseCmdAck(0xF3);
    KbdMouseCmdAck(200);
    KbdMouseCmdAck(0xF3); // enable Mouse
    KbdMouseCmdAck(100); // Set Rate
    KbdMouseCmdAck(0xF3);
    KbdMouseCmdAck(80); //Resolution

    MouseGetDeviceType; // enable Mouse

    KbdMouseCmdAck(0xF3);
    KbdMouseCmdAck(10);
    MouseGetDeviceType; // enable Mouse

    KbdMouseCmdAck(0xE8); //Resolution
    KbdMouseCmdAck(0x03);
    KbdMouseCmdAck(0xE6);

    KbdMouseCmdAck(0xF3); // Set Rate
    KbdMouseCmdAck(100);

    KbdMouseCmdAck(0xF4); // enable Mouse
    result=TRUE;

//Enable IRQ 12
    KbdCmdSend(KEYB_CTRL,0x20);
    b=KbdCmdRead;
    KbdCmdSend(KEYB_CTRL,0x60);
    KbdCmdSend(KEYB_PORT,(b|2)&~0x20);

  } catch
    Fs->catch_except=TRUE;

//This is been added to override failure
//because the mouse sometimes still works.
        result=TRUE;

  try
    KbdCmdSend(KEYB_CTRL,0xAE); //Enable Keyboard
  catch
    Fs->catch_except=TRUE;
  if (mouse_has_wheel || mouse_has_ext_buttons)
    mouse_pkt_size=4;
  else
    mouse_pkt_size=3;
  if (!result)
    try
      KbdCmdSend(KEYB_CTRL,0xA7); //Disable Mouse
    catch
      Fs->catch_except=TRUE;
  return result;
}


U0 SetMouseSpeed()
{
  U8 dd,temp;
  if (dd=(old_mxx-mxx)*(old_mxx-mxx)+
         (old_myy-myy)*(old_myy-myy))
    if (temp=mouse_evt_time-last_mouse_evt_time)
      mouse_speed=Sqrt(dd)*time_stamp_freq/temp;
  last_mouse_evt_time=mouse_evt_time;
}

U0 MouseUpdatePre(BoolI1 *old_buttons)
{
  I8 i;

  old_mxx=mxx;
  old_myy=myy;
  old_mzz=mzz;
  for (i=0;i<5;i++)
    old_buttons[i]=mouse_buttons[i];
}

U0 MouseUpdatePost(BoolI1 *old_buttons)
{
  I8 i;

  mxx=mxx_prescale*mxx_scale;
  myy=myy_prescale*myy_scale;
  mzz=mzz_prescale*mzz_scale;

  i=ToI8(ipx_scale*mxx/ipx_snap)*ipx_snap+ipx_offset;
  if (i<0)
    ipx_offset-=i;
  else if (i>=gr_width)
    ipx_offset+=gr_width-1-i;

  i=ToI8(ipy_scale*myy/ipy_snap)*ipy_snap+ipy_offset;
  if (i<0)
    ipy_offset-=i;
  else if (i>=gr_height)
    ipy_offset+=gr_height-1-i;

  if (mxx!=old_mxx||myy!=old_myy||mzz!=old_mzz) {
    mouse_evt=TRUE;
    SetMouseSpeed;
  } else
    for (i=0;i<5;i++)
      if (mouse_buttons[i]!=old_buttons[i]) {
        mouse_evt=TRUE;
        break;
      }
}

U0 MouseHandler()
{
  I8 i,dx,dy,dz;
  U1 mouse_buf[4];
  BoolI1 old_buttons[5];

  MouseUpdatePre(old_buttons);

  for (i=0;i<4;i++)
    mouse_buf[i]=0;
  for (i=0;i<mouse_pkt_size;i++)
    if (!U1FifoRem(kbdmouse_fifo,&mouse_buf[i]))
      mouse_buf[i]=0;

  mouse_buttons[0] = mouse_buf[0] & 1;
  mouse_buttons[1] = (mouse_buf[0] & 2) >> 1;
  mouse_buttons[2] = (mouse_buf[0] & 4) >> 2;
  mouse_buttons[3] = (mouse_buf[3] & 0x10) >> 4;
  mouse_buttons[4] = (mouse_buf[3] & 0x20) >> 5;
  if (mouse_buf[0] & 0x10)
    dx=mouse_buf[1]-256;
  else
    dx=mouse_buf[1];
  if (mouse_buf[0] & 0x20)
    dy=256-mouse_buf[2];
  else
    dy=-mouse_buf[2];
  if (mouse_buf[3] & 0x08)
    dz=mouse_buf[3]&7-8;
  else
    dz=mouse_buf[3]&7;

  mxx_prescale+=dx;
  myy_prescale+=dy;
  mzz_prescale+=dz;

  MouseUpdatePost(old_buttons);
}

U0 SetMouseIP(I8 x,I8 y,I8 z,I8 l,I8 r,I8 c)
{
  BoolI1 old_buttons[5];

  mouse_evt_time=GetTimeStamp;
  LowPower(FALSE);

  MouseUpdatePre(old_buttons);

  mxx_prescale=x/mxx_scale;
  myy_prescale=y/myy_scale;
  mzz_prescale=z/mzz_scale;
  mouse_buttons[0]=l;
  mouse_buttons[1]=r;
  mouse_buttons[2]=c; //unsure which one

  MouseUpdatePost(old_buttons);
}

U0 KbdMouseReset()
{
  KbdCmdFlush;
  U1FifoFlush(kbdmouse_fifo);
  U8FifoFlush(scan_code_fifo);
  sys_cur_scan_code=0;
  reset_kbdmouse=FALSE;
}

BoolI8 InstallMouseDriver()
{
  I8 i;
  BoolI1 old_preempt=Preempt(ON);
  attempting_mouse_install=TRUE;
  OutU1(0xA1,InU1(0xA1)|0x10);
  mouse_installed=mouse_irqs_working=FALSE;
  SetIDTEntry(0x2C,&MouseIntHandler,0);
  for(i=0;i<5;i++)
    mouse_buttons[i]=0;
  if (i=MouseReset)
    OutU1(0xA1,InU1(0xA1)&~0x10);
  KbdMouseReset;
  mouse_install_attempts++;
  mouse_installed=mouse_evt=i;
  attempting_mouse_install=FALSE;
  Preempt(old_preempt);
  return mouse_installed;
}

U0 KbdMouseHandler(BoolI1 poll_kbd,BoolI1 poll_mouse)
{
  U1 j;
  if (attempting_mouse_install) {
    TaskNext;
    return;
  }
  if (reset_kbdmouse)
    KbdMouseReset;
  else {
    if (poll_mouse &&
        mouse_installed &&
        !mouse_irqs_working) {
      PushFD;
      Cli;
      while (InU1(KEYB_CTRL)&1)
        MousePktRead;
      PopFD;
    }

    if (poll_kbd)
      while (InU1(KEYB_CTRL)&1)
        KbdPktRead;

    if (reset_kbdmouse)
      KbdMouseReset;
    else {
      while (U1FifoCnt(kbdmouse_fifo)) {
        U1FifoRem(kbdmouse_fifo,&j);
        if (j==0x01)
          KbdHandler;
        else if (j==0x0C && mouse_installed)
          MouseHandler;
        else
          KbdMouseReset;
      }
    }
  }
}

U0 KbdMouseVariablesInit()
{
  sys_key_cnt=0;
  scan_code_fifo=U8FifoNew(128);
  kbd_fifo=U1FifoNew(8);
  mouse_fifo=U1FifoNew(8);
  MemSet(key_down_bitmap,0,256/8);
  MemSet(key_down_bitmap2,0,256/8);
  last_down_scan_code=0;
  sys_cur_scan_code=0;
  reset_kbdmouse=FALSE;
  mouse_irqs_working=FALSE;
  mxx_scale=0.5;
  myy_scale=0.5;
  mzz_scale=1.0;
  mxx_prescale=gr_width/mxx_scale/2.0;
  myy_prescale=gr_height/myy_scale/2.0;
  mzz_prescale=0/mzz_scale;
  mxx=gr_width>>1;
  myy=gr_height>>1;
  mzz=0;
  mouse_speed=0.0;
  old_mxx=mxx;
  old_myy=myy;
  old_mzz=mzz;
  mouse_installed=FALSE;
  mouse_install_attempts=0;
  last_mouse_evt_time=0;
  mouse_has_wheel=FALSE;
  mouse_has_ext_buttons=FALSE;
  mouse_evt=FALSE;
}
