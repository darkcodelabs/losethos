public U0 PutBibleVerse(U8 *verse,I64 lines)
{
  I64 i;
  Ltf *l;
  LtfEntry *cl;
  U8 *st=StrNew(verse),
      *st2=StrNew(verse);
  if (StrOcc(st,','))
    StrLastRem(st,",",st2);
  else
    *st2=0;

  i=A2I(DefineSub(DefineMatch(st,"ST_BIBLE_BOOKS",LEF_IGNORE_CASE),
      "ST_BIBLE_BOOK_LINES"));
  l=LtfRead(Define("ST_BIBLE_FILE"),LTFF_PLAIN_TEXT);
  LtfGoToLineNum(l,i); //one based
  if (*st2) {
    StrCpy(l->find_replace->find_text,st2);
    EdFindNext(l);
  }
  cl=l->cur_entry;
  while (lines) {
    if (cl->btype==LTFT_TEXT)
      cout cl->tag;
    else if (cl->btype==LTFT_CR) {
      CrLf;
      lines--;
    }
    cl=cl->next;
  }
  LtfDel(l);
  Free(st);
  Free(st2);
}
