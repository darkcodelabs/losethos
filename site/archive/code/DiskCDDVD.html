      n=CDFileCreateDo(out_file,tempm1,de,de2,cur_blk,tempm,write);
      de><(U1 *)+=sizeof(ISO1DirEntry)-1+n;

    if ((de1><(U1 *)-dir_blk_buf><(U1 *))/CD_BLK_SIZE!=

      MemSet(de12,0,i);
      de2><(U1 *)=dir_blk_buf2><(U1 *)+(de2><(U1 *)-dir_blk_buf2><(U1 *))/CD_BLK_SIZE*CD_BLK_SIZE;
      MemCpy(de2,buf,i);
      de2><(U1 *)+=i;
    }
    tempm1=tempm2;
  }

  tempc=tempm->user_data;

  tempi->len=sizeof(ISO1DirEntry)-1;
  tempi->ext_attr_len=0;
  tempi->flags=ISO1_ATTR_DIR;
  if (!tempm->name[0]) {

    StrCpy(&tempi->name,tempm->name);
  }
  tempi->len+=tempi->name_len;

  n=de><(U1 *)+1-dir_blk_buf><(U1 *);

  } else {

      *ptr1++=*ptr2++;

  n=de2><(U1 *)+1-dir_blk_buf2><(U1 *);
  n=(n+CD_BLK_SIZE-1)/CD_BLK_SIZE;
  FillU4Palindrome(&tempi2->size,n*CD_BLK_SIZE);
  FillU4Palindrome(&tempi2->loc,*cur_blk);
  tempc->long_dir_blks=n;
  FillU4Palindrome(&dir_blk_buf2->size,n*CD_BLK_SIZE);
  FillU4Palindrome(&dir_blk_buf2->loc,*cur_blk);

  Free(dir_blk_buf2);
  Free(buf);
  return tempi->name_len;
}

U8 CDTableLen(LTDirEntry *tempm,U8 *size1,U8 *size2,U8 cur_depth)
//returns depth
{
  LTDirEntry *tempm1=tempm->sub;
  U8 max_depth=cur_depth,i;
  while (tempm1) {
    if (tempm1->attr & LT_ATTR_DIR) {
      *size1+=sizeof(ISO1PathTableEntry)-2+ (StrLen(tempm1->name)+1)&-0x2;
      *size2+=sizeof(ISO1PathTableEntry)-2+StrLen(tempm1->name)<<1;
      i=CDTableLen(tempm1,size1,size2,cur_depth+1);
      if (i>max_depth) max_depth=i;
    }
    tempm1=tempm1->next;
  }
  return max_depth;
}

U0 CDFillPathTable(LTDirEntry *tempm,
  ISO1PathTableEntry **itabbuf,ISO1PathTableEntry **itabbuf2,
  U8 parent_entry_num,BoolI1 big_endian,U8 *first_free,I8 cur_level,I8 output_level)
{
  U1 *ptr1,*ptr2;
  U8 i;
  ISO1PathTableEntry *tabbuf=*itabbuf;
  ISO1PathTableEntry *tabbuf2=*itabbuf2;
  LTDirEntry *tempm1=tempm->sub,*tempm2;
  CCdUserData *tempc;
 
  if (cur_level==output_level) {
    while (tempm1) {
      if (tempm1->attr & LT_ATTR_DIR) {
        tempc=tempm1->user_data;
        tempc->path_entry_num=*first_free;
        tabbuf->name_len=StrLen(tempm1->name);
        if (big_endian) {
          tabbuf->blk=BigEndianU4(tempc->loc);
          tabbuf->parent_entry_num=BigEndianU2(parent_entry_num);
        } else {
          tabbuf->blk=tempc->loc;
          tabbuf->parent_entry_num=parent_entry_num;
        }
        StrCpy(&tabbuf->name,tempm1->name);

        tabbuf><(U1 *)+=sizeof(ISO1PathTableEntry)-2+
          (StrLen(tempm1->name)+1) & -0x2;

        tabbuf2->name_len=StrLen(tempm1->name)<<1;
        if (big_endian) {
          tabbuf2->blk=BigEndianU4(tempc->loc+tempc->short_dir_blks);
          tabbuf2->parent_entry_num=BigEndianU2(parent_entry_num);
        } else {
          tabbuf2->blk=tempc->loc+tempc->short_dir_blks;
          tabbuf2->parent_entry_num=parent_entry_num;
        }
        ptr1=&tabbuf2->name;
        ptr2=&tempm1->name;
        for (i=0;i<tabbuf2->name_len;i=i+2) {
          ptr1++;
          *ptr1++=*ptr2++;
        }
        tabbuf2><(U1 *)+=sizeof(ISO1PathTableEntry)-2+
          StrLen(tempm1->name)<<1;
        *first_free+=1;
      }
      tempm1=tempm1->next;
    }
    *itabbuf=tabbuf;
    *itabbuf2=tabbuf2;
  }
  tempm1=tempm->sub;
  while (tempm1) {
    tempm2=tempm1->next;
    if (tempm1->attr & LT_ATTR_DIR) {
      tempc=tempm1->user_data;
      CDFillPathTable(tempm1,itabbuf,itabbuf2,tempc->path_entry_num,
         big_endian,first_free,cur_level+1,output_level);
    }
    tempm1=tempm2;
  }
}
