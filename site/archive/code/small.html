#define MAN_FRAMES 8

U8 *man_imgs[MAN_FRAMES]=
{
<1>,<2>,<3>,<4>,
<5>,<4>,<3>,<1>
};

F64 t0=tJ;

U0 DrawIt(TaskStruct *task,GrDC *dc)
{
  U8 *inter;
  F64 tt=4.0*tJ,dt=(tJ-t0)*music_tempo;
  inter=GrSpriteInterpolate(man_imgs[tt%MAN_FRAMES],man_imgs[(tt+1)%MAN_FRAMES],tt%1.0);
  GrSpritePlot3(dc,180,40,0,inter);
  Free(inter);
  if (dt>24.0)
    GrSpritePlot3(dc,task->win_pixel_width-12.0*(dt-24.0),40,0,<6>);
}
